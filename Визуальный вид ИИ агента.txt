function applyTheme(name) {

  const c = th.color;
  document.documentElement.style.setProperty('--neon', c);

function renderThemeGrid() {
  const wrap = $('#theme-grid');

  wrap.innerHTML = '';
  Object.entries(THEMES).forEach(([key, th]) => {

    el.onclick = () => {
      applyTheme(key);
      vibrate(10);
      toast('🎨 ' + th.name);
    };
    wrap.appendChild(el);
  });
}

  const wrap = $('#chat-messages');

  el.className = 'msg ' + role;
  if (typing) {
    el.innerHTML = `<div class="bubble"><span class="typing"><span></span><span></span><span></span></span></div>`;

function sendMessage(text) {
  text = (text || '').trim();
  if (!text) return;
  appendMsg('me', text);
  state.messages.push({ role:'me', text });
  $('#chat-text').value = '';
  vibrate(10);

  setThinking(true);
  const typingEl = appendMsg('bot', '', true);
  const delay = 700 + Math.random() * 700;

  setTimeout(() => {
    typingEl.remove();
    const arr = AI_RESPONSES[state.lang] || AI_RESPONSES.ru;
    const reply = arr[Math.floor(Math.random() * arr.length)];
    appendMsg('bot', reply);
    state.messages.push({ role:'bot', text: reply });
    setThinking(false);
    vibrate([15, 30, 15]);

    if (/задач|task|создал|created/i.test(reply)) {
      state.notes.unshift({
        id: 'n' + (state.nextNoteId++),
        title: text.slice(0, 40),
        body: '',
        tag: 'ai',
        status: 'pending',
      });
      renderNotes();
    }
  }, delay);
}

}

/* ============ WORKFLOW ============ */
function renderWorkflow() {
  const wrap = $('#timeline');
  if (!wrap) return;
  wrap.innerHTML = '';
  state.workflow.forEach(w => {
    const el = document.createElement('div');
    el.className = 'timeline-item ' +
      (w.status === 'done' ? 'done' : w.status === 'active' ? 'active' : w.status === 'paused' ? 'paused' : '') +
      (w.expanded ? ' expanded' : '') +
      (!w.expanded ? ' hidden-detail' : '');
    let dotContent = '';
    if (w.status === 'done') dotContent = '✓';
    else if (w.status === 'active') dotContent = '<svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><path d="M21 12a9 9 0 1 1-6.219-8.56"/></svg>';
    else if (w.status === 'paused') dotContent = '❚❚';
    else dotContent = '·';

    el.innerHTML = `
      <div class="timeline-dot">${dotContent}</div>
      <div class="timeline-content">
        <div class="timeline-head">
          <h4>${escapeHtml(w.title)}</h4>
          <svg class="expand-icon" width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5">
            <polyline points="6 9 12 15 18 9"/>
          </svg>
        </div>
        <div class="timeline-detail">
          <p>${escapeHtml(w.desc)}</p>
          ${w.status === 'active' ? `
            <div class="wave">${Array(8).fill('<span></span>').join('')}</div>
          ` : ''}
          ${w.status === 'paused' ? `
            <div class="wave paused">${Array(8).fill('<span></span>').join('')}</div>
          ` : ''}
          ${w.status === 'active' || w.status === 'paused' ? `
            <div class="timeline-progress">
              <div class="timeline-progress-fill" style="width:${w.progress}%"></div>
            </div>
            <div class="timeline-meta">
              <span class="live" style="color: ${w.status === 'paused' ? 'var(--muted)' : 'var(--orange)'}">
                ${w.status === 'paused' ? t('paused') : (state.lang === 'ru' ? 'выполняется' : 'running')}
              </span>
              <span>${w.progress}%</span>
            </div>
            <div class="timeline-controls">
              ${w.status === 'active' ? `
                <button class="ctrl-btn warn" data-act="pause" data-id="${w.id}">
                  <svg viewBox="0 0 24 24" fill="currentColor"><rect x="6" y="4" width="4" height="16"/><rect x="14" y="4" width="4" height="16"/></svg>
                  ${t('pause')}
                </button>
              ` : ''}
              ${w.status === 'paused' ? `
                <button class="ctrl-btn" data-act="resume" data-id="${w.id}">
                  <svg viewBox="0 0 24 24" fill="currentColor"><polygon points="5 3 19 12 5 21"/></svg>
                  ${t('resume')}
                </button>
              ` : ''}
              <button class="ctrl-btn" data-act="skip" data-id="${w.id}">
                ${t('skip')} →
              </button>
              <button class="ctrl-btn danger" data-act="stop" data-id="${w.id}">
                ${t('stop')}
              </button>
            </div>
          ` : ''}
          ${w.status === 'done' ? `
            <div class="timeline-meta">
              <span style="color: var(--neon);">✓ ${t('done')}</span>
              <button class="ctrl-btn" data-act="restart" data-id="${w.id}" style="margin-left:auto;">
                ↻ ${t('restart')}
              </button>
            </div>
          ` : ''}
          ${w.status === 'queued' ? `
            <div class="timeline-controls">
              <button class="ctrl-btn" data-act="start" data-id="${w.id}">
                ▶ ${state.lang === 'ru' ? 'запустить' : 'start'}
              </button>
            </div>
          ` : ''}
        </div>
      </div>
    `;
    el.querySelector('.timeline-head').onclick = () => {
      w.expanded = !w.expanded;
      renderWorkflow();
      vibrate(5);
    };
    wrap.appendChild(el);
  });

  wrap.querySelectorAll('.ctrl-btn').forEach(b => {
    b.onclick = (e) => {
      e.stopPropagation();
      const act = b.dataset.act;
      const id = b.dataset.id;
      const w = state.workflow.find(x => x.id === id);
      if (!w) return;
      if (act === 'pause') { w.status = 'paused'; toast('⏸ ' + t('paused')); }
      else if (act === 'resume') { w.status = 'active'; toast('▶ ' + t('resume')); }
      else if (act === 'stop') {
        w.status = 'queued'; w.progress = 0;
        toast('⏹ ' + t('stop'), true);
      } else if (act === 'skip') {
        w.status = 'done'; w.progress = 100;
        const next = state.workflow.find(x => x.status === 'queued');
        if (next) next.status = 'active';
        toast('⏭ ' + t('skip'));
      } else if (act === 'restart') {
        w.status = 'active'; w.progress = 0;
        toast('↻ ' + t('restart'));
      } else if (act === 'start') {
        w.status = 'active';
        toast('▶ start');
      }
      renderWorkflow();
      vibrate(10);
    };
  });

  // Auto-progress active task
  const active = state.workflow.find(w => w.status === 'active');
  if (active && active.progress < 100) {
    setTimeout(() => {
      if (active.status !== 'active') return;
      active.progress = Math.min(100, active.progress + Math.ceil(Math.random()*3));
      if (active.progress >= 100) {
        active.status = 'done';
        const next = state.workflow.find(w => w.status === 'queued');
        if (next) next.status = 'active';
        vibrate([10, 30, 10]);
      }
      renderWorkflow();
    }, 900);
  }
}

function renderPeriodTabs() {
  const labels = { day: t('p_day'), week: t('p_week'), month: t('p_month'), year: t('p_year') };
  $$('.period-tab').forEach(b => {
    if (b.dataset.p && labels[b.dataset.p]) {
      if (!b.closest('#note-status-tabs') && !b.closest('#limit-period-tabs')) {
        b.textContent = labels[b.dataset.p];
      }
    }
  });
}

function renderFinance() {
  const period = state.period;
  const data = state.finance[period];
  const diff = data.income - data.expense;
  const sign = diff >= 0 ? '+' : '−';
  const cls = diff >= 0 ? 'up' : 'down';
  const periodLabels = {
    day: t('p_day').toUpperCase(),
    week: t('p_week').toUpperCase(),
    month: t('p_month').toUpperCase(),
    year: t('p_year').toUpperCase(),
  };

  const mainEl = $('#metric-main');
  mainEl.textContent = sign + money(Math.abs(diff));
  mainEl.className = 'metric-value ' + cls;

  $('#metric-period-label').textContent = (state.lang === 'ru' ? 'ЧИСТЫЙ ПОТОК' : 'NET FLOW') + ' · ' + periodLabels[period];

  $('#metric-income').textContent = '+' + money(data.income);
  $('#metric-expense').textContent = '−' + money(data.expense);

  const wrap = $('#tx-history');
  wrap.innerHTML = data.txs.map(tx => `
    <div class="tx-row ${tx.type}">
      <div class="tx-avatar">${tx.type === 'in' ? '↓' : '↑'}</div>
      <div class="tx-info">
        <div class="tx-name">${escapeHtml(tx.name)}</div>
        <div class="tx-time">${escapeHtml(tx.time)}</div>
      </div>
      <div class="tx-amount ${tx.type}">${tx.type === 'in' ? '+' : '−'}${money(tx.amount)}</div>
    </div>
  `).join('');
  $('#tx-count').textContent = data.txs.length;
}

$$('.period-tab').forEach(b => {
  if (b.closest('#limit-period-tabs') || b.closest('#note-status-tabs')) return;
  b.onclick = () => {
    $$('.period-tab').forEach(x => {

    });
    b.classList.add('active');
    state.period = b.dataset.p;
    renderFinance();
    vibrate(6);
  };
});

function updateLimitSlider() {
  const track = $('#limit-slider');
  const thumb = $('#limit-thumb');
  const fill = $('#limit-fill');
  if (!track) return;
  const pct = state.limit.value / state.limit.max;
  const w = track.clientWidth;
  const tw = thumb.offsetWidth;
  const pad = 6;
  const maxLeft = w - tw - pad * 2;
  const left = pad + pct * maxLeft;
  thumb.style.left = left + 'px';
  fill.style.width = (left + tw/2) + 'px';
  $$('#limit-presets .limit-preset').forEach(b => {
    b.classList.toggle('active', parseInt(b.dataset.v) === state.limit.value);
  });
}

$$('#limit-presets .limit-preset').forEach(b => {
  b.onclick = () => {
    state.limit.value = parseInt(b.dataset.v);
    renderUserLimit();
    vibrate(6);
  };
});

let limitDragging = false;
function onLimitDown(e) {
  limitDragging = true;
  $('#limit-thumb').classList.add('dragging');
  e.preventDefault();
  window.addEventListener('pointermove', onLimitMove);
  window.addEventListener('pointerup', onLimitUp);
  onLimitMove(e);
}
function onLimitMove(e) {
  if (!limitDragging) return;
  const track = $('#limit-slider');
  const rect = track.getBoundingClientRect();
  const tw = 74;
  const pad = 6;
  const maxLeft = rect.width - tw - pad*2;
  let x = e.clientX - rect.left - tw/2 - pad;
  x = Math.max(0, Math.min(maxLeft, x));
  const pct = x / maxLeft;
  const rawVal = pct * state.limit.max;
  // Round to nearest 50
  state.limit.value = Math.max(50, Math.round(rawVal / 50) * 50);
  renderUserLimit();
  if (navigator.vibrate && Math.random() > 0.6) navigator.vibrate(3);
}
function onLimitUp() {
  if (!limitDragging) return;
  limitDragging = false;
  $('#limit-thumb').classList.remove('dragging');
  window.removeEventListener('pointermove', onLimitMove);
  window.removeEventListener('pointerup', onLimitUp);
}
$('#limit-slider').addEventListener('pointerdown', onLimitDown);

/* ============ HELPERS ============ */
function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
function money(n) {
  return '$' + n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
}
