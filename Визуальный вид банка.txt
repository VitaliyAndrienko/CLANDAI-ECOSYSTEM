function customSelect(container, options, initialValue, onChange) {
  container.innerHTML = `
    <button type="button" class="cselect-btn">
      <span class="cselect-label"></span>
      <svg class="cselect-arrow" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" width="14" height="14"><polyline points="6 9 12 15 18 9"/></svg>
    </button>
    <div class="cselect-menu"></div>`;
  const menu = container.querySelector('.cselect-menu');
  const labelEl = container.querySelector('.cselect-label');
  const btn = container.querySelector('.cselect-btn');
  let value = initialValue;
  function render() {
    menu.innerHTML = options.map(o =>
      `<div class="cselect-opt${String(o.value) === String(value) ? ' active' : ''}" data-v="${o.value}">${o.label}</div>`
    ).join('');
    const cur = options.find(o => String(o.value) === String(value));
    labelEl.textContent = cur ? cur.label : '';
  }
  btn.addEventListener('click', e => {
    e.stopPropagation();
    document.querySelectorAll('.cselect.open').forEach(s => { if (s !== container) s.classList.remove('open'); });
    container.classList.toggle('open');
  });
  menu.addEventListener('click', e => {
    const opt = e.target.closest('.cselect-opt');
    if (!opt) return;
    value = opt.dataset.v;
    container.classList.remove('open');
    render();
    if (onChange) onChange(value);
  });
  render();
  return { get value() { return value; }, set(v) { value = v; render(); } };
}

function initPinKeypad(keypadEl, displayEl, max, onComplete) {
  displayEl.innerHTML = '';
  const slots = [];
  for (let i = 0; i < max; i++) { const s = document.createElement('div'); s.className = 'pin-slot'; displayEl.appendChild(s); slots.push(s); }
  let v = '';
  const render = (err = false) => slots.forEach((s, i) => {
    s.classList.toggle('filled', i < v.length);
    s.classList.toggle('err', err);
  });
  const press = k => {
    if (k === 'del') { v = v.slice(0, -1); render(); return; }
    if (k === 'ok') { if (v.length === max) onComplete(v); return; }
    if (v.length >= max) return;
    v += k; render(); vibrate(6);
    if (v.length === max) setTimeout(() => onComplete(v), 130);
  };
  keypadEl.querySelectorAll('button').forEach(b => { b.onclick = () => press(b.dataset.k); });
  return { reset() { v = ''; render(); }, setError() { render(true); setTimeout(() => { v = ''; render(false); }, 420); } };
}

  [...wrap.children].forEach((c, i) => {
    const cc = c.offsetLeft + c.offsetWidth / 2;
    const d = Math.abs(cc - center);
    if (d < bestD) { bestD = d; best = i; }
  });
  if (best !== state.activeCard) {
    state.activeCard = best;
    resetSlider();
  }
  [...wrap.children].forEach((el, k) => el.classList.toggle('active', k === best));
  $$('#card-dots .card-dot').forEach((d, k) => d.classList.toggle('active', k === best));
  updateBalancePanel();
  renderTx();
  $('#block-card-name').textContent = cardName(CARDS[best]) + ' · ' + CARDS[best].num.slice(-4);
}

}

function applySlider() {
  const track = $('#slider-track');

let dragStartX = 0, dragStartLeft = 0, lastVibPct = -1;

function onSliderDown(e) {
  if (e.button !== undefined && e.button !== 0) return;
  state.sliderDragging = true;
  $('#slider-thumb').classList.add('dragging');
  dragStartX = e.clientX;
  const track = $('#slider-track');
  const thumb = $('#slider-thumb');
  const trackW = track.clientWidth;
  const thumbW = thumb.offsetWidth;
  const pad = 6;
  const maxLeft = trackW - thumbW - pad * 2;
  dragStartLeft = (state.sliderPct / 100) * maxLeft;
  e.preventDefault();
  window.addEventListener('pointermove', onSliderMove);
  window.addEventListener('pointerup', onSliderUp);
  window.addEventListener('pointercancel', onSliderUp);
  vibrate(10);
}
function onSliderMove(e) {
  if (!state.sliderDragging) return;
  const track = $('#slider-track');
  const thumb = $('#slider-thumb');
  const trackW = track.clientWidth;
  const thumbW = thumb.offsetWidth;
  const pad = 6;
  const maxLeft = trackW - thumbW - pad * 2;
  let dx = e.clientX - dragStartX;
  let newLeft = Math.max(0, Math.min(maxLeft, dragStartLeft + dx));
  state.sliderPct = (newLeft / maxLeft) * 100;
  applySlider();
  const p = Math.floor(state.sliderPct / 5) * 5;
  if (p !== lastVibPct) {
    lastVibPct = p;
    if (navigator.vibrate) navigator.vibrate(4);
  }
}
function onSliderUp() {
  if (!state.sliderDragging) return;
  state.sliderDragging = false;
  $('#slider-thumb').classList.remove('dragging');
  window.removeEventListener('pointermove', onSliderMove);
  window.removeEventListener('pointerup', onSliderUp);
  window.removeEventListener('pointercancel', onSliderUp);
  if (state.sliderPct >= 95) { state.sliderPct = 100; }
  else if (state.sliderPct < 5) { state.sliderPct = 0; }
  applySlider();
}
$('#slider-track').addEventListener('pointerdown', e => {
  const track = $('#slider-track');
  const thumb = $('#slider-thumb');
  const rect = track.getBoundingClientRect();
  const thumbRect = thumb.getBoundingClientRect();
  if (e.clientX >= thumbRect.left && e.clientX <= thumbRect.right) {
    onSliderDown(e);
  } else {
    const trackW = track.clientWidth;
    const thumbW = thumb.offsetWidth;
    const pad = 6;
    const maxLeft = trackW - thumbW - pad * 2;
    const relX = e.clientX - rect.left - pad - thumbW / 2;
    state.sliderPct = Math.max(0, Math.min(100, (relX / maxLeft) * 100));
    applySlider();
    onSliderDown(e);
  }
});

/* ================= HISTORY ================= */
function renderTx() {
  const wrap = $('#tx-list');
  const c = CARDS[state.activeCard];
  if (!c) return;
  const hist = $('#history-title');
  if (hist) hist.textContent = t('history') + ' · ' + cardName(c).toUpperCase();
  const last3 = (c.txs || []).slice(0, 3);
  if (!last3.length) {
    wrap.innerHTML = `<div class="empty">${t('no_txs')}</div>`;
    return;
  }
  wrap.innerHTML = last3.map(tx => txRowHTML(tx)).join('');
}
function txRowHTML(tx) {
  return `
    <div class="tx-row ${tx.type}">
      <div class="tx-avatar">${tx.type === 'in' ? '↓' : '↑'}</div>
      <div class="tx-info">
        <div class="tx-name">${tx.name}</div>
        <div class="tx-time">${tx.time}</div>
      </div>
      <div class="tx-amount ${tx.type}">${tx.type === 'in' ? '+' : '-'}${money(tx.amount)}</div>
    </div>`;
}
function renderFullHistory() {
  const wrap = $('#tx-list-full');
  const c = CARDS[state.activeCard];
  if (!c) return;
  $('#history-modal-title').textContent = t('history') + ' · ' + cardName(c).toUpperCase();

    wrap.innerHTML = `<div class="empty">${t('no_txs')}</div>`;
    return;
  }
  wrap.innerHTML = c.txs.map(tx => txRowHTML(tx)).join('');
}

  $$('#modal-deposit-anim .dep-anim-view').forEach(v => v.classList.remove('active'));
  $('#dep-anim-sending').classList.add('active');
  const rail = $('#dep-rail');
  rail.innerHTML = '';
  // spawn coins for each selected crypto (max 4)
  const coins = state.depSelected.slice(0, 4);
  coins.forEach((k, i) => {
    const meta = CRYPTO_META[k];
    const coin = document.createElement('div');
    coin.className = 'dep-coin';
    coin.style.background = `radial-gradient(circle at 30% 30%, ${meta.color}, ${meta.color}cc 60%, ${meta.color}88)`;
    coin.style.color = meta.color;
    coin.textContent = meta.icon;
    rail.appendChild(coin);
    setTimeout(() => {
      coin.classList.add('rolling');
      vibrate(6);
    }, 250 + i * 220);
  });
  openModal('modal-deposit-anim');
  // After animation done → show contract
  setTimeout(() => {
    const id = 'CT-2026-' + Math.floor(100000 + Math.random() * 900000);
    $('#contract-id').textContent = id;
    $$('#modal-deposit-anim .dep-anim-view').forEach(v => v.classList.remove('active'));
    $('#dep-anim-done').classList.add('active');
    vibrate([15,30,15]);
    // apply balance change
    const c = CARDS[state.activeCard];
    c.balance -= amount;
    c.txs.unshift({ name:'Deposit', time:new Date().toTimeString().slice(0,5), amount:amount, type:'out' });
    updateBalancePanel(); renderTx();
  }, 2500 + Math.max(0, coins.length - 1) * 220 + 400);
}
$('#dep-anim-close').onclick = () => closeModal('modal-deposit-anim');
$('#contract-dl').onclick = () => {

};

/* --- BLOCK --- */
function openBlockModal() {
  const c = CARDS[state.activeCard];
  const blocked = state.blocked.includes(c.id);
  $('#block-card-name').textContent = cardName(c) + ' · ' + c.num.slice(-4);
  $('#block-go').textContent = blocked ? t('unblock_btn') : t('block_btn');
  $('#block-pass').value = '';
  openModal('modal-block');
}
$('#block-cancel').onclick = () => closeModal('modal-block');

let recognition = null;
$('#block-mic').onclick = () => {
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SR) { $('#block-pass').focus(); return; }
  if (recognition) { try { recognition.stop(); } catch {} recognition = null; return; }
  recognition = new SR();
  recognition.lang = state.lang === 'ru' ? 'ru-RU' : 'en-US';
  recognition.interimResults = false;
  recognition.maxAlternatives = 1;
  recognition.onstart = () => toast('🎤 ' + t('listening'));
  recognition.onresult = ev => { $('#block-pass').value = ev.results[0][0].transcript.trim().toLowerCase(); recognition = null; };
  recognition.onerror = () => { recognition = null; };
  recognition.onend = () => { recognition = null; };
  try { recognition.start(); } catch { recognition = null; }
};

$('#block-go').onclick = () => {
  const val = $('#block-pass').value.trim().toLowerCase();
  const c = CARDS[state.activeCard];
  const blocked = state.blocked.includes(c.id);
  if (val !== state.reserve) { toast(t('wrong_reserve'), true); return; }
  if (blocked) { state.blocked = state.blocked.filter(x => x !== c.id); toast('✓ ' + t('unblock_ok')); }
  else { state.blocked.push(c.id); toast('🔒 ' + t('block_ok')); }
  localStorage.setItem('bank_blocked', JSON.stringify(state.blocked));
  closeModal('modal-block');
  renderCards();
};

/* --- SWAP --- */
function swapCalc() {
  const mode = state.swapMode;
  const crypto = state.swapCrypto;
  const price = CRYPTO_PRICES[crypto];
  const meta = CRYPTO_META[crypto];
  const fromAmt = parseFloat($('#swap-from-amt').value) || 0;
  if (mode === 'buy') {
    $('#swap-from-label').textContent = t('swap_from');
    $('#swap-to-label').textContent = t('swap_to');
    $('#swap-from-flag').textContent = '🇺🇸';
    $('#swap-from-sym').textContent = 'USD';
    $('#swap-to-flag').textContent = meta.icon;
    $('#swap-to-sym').textContent = crypto;
    $('#swap-to-amt').value = (fromAmt / price).toFixed(crypto === 'USDT' ? 2 : 6);
    $('#swap-rate-val').textContent = `1 USD = ${(1/price).toFixed(crypto === 'USDT' ? 2 : 6)} ${crypto}`;
  } else {
    $('#swap-from-label').textContent = t('swap_from');
    $('#swap-to-label').textContent = t('swap_to');
    $('#swap-from-flag').textContent = meta.icon;
    $('#swap-from-sym').textContent = crypto;
    $('#swap-to-flag').textContent = '🇺🇸';
    $('#swap-to-sym').textContent = 'USD';
    $('#swap-to-amt').value = (fromAmt * price).toFixed(2);
    $('#swap-rate-val').textContent = `1 ${crypto} = ${price.toLocaleString('en-US')} USD`;
  }
  const changes = { BTC: 1.15, ETH: -0.42, USDT: 0.01, TON: 2.31 };
  const ch = changes[crypto] || 0;
  const el = $('#swap-change');
  el.textContent = (ch >= 0 ? '▲ +' : '▼ ') + ch.toFixed(2) + '%';
  el.classList.toggle('dn', ch < 0);
  $('#swap-fee').textContent = money(0.5);
}
function buildSwapUI() {
  $('#swap-from-cur').onclick = e => { e.stopPropagation(); if (state.swapMode === 'buy') return; cycleCrypto(); };
  $('#swap-to-cur').onclick = e => { e.stopPropagation(); if (state.swapMode === 'sell') return; cycleCrypto(); };
  $('#swap-from-amt').addEventListener('input', swapCalc);
  $('#swap-flip').onclick = () => {
    state.swapMode = state.swapMode === 'buy' ? 'sell' : 'buy';
    $$('.swap-tab').forEach(tab => tab.classList.toggle('active', tab.dataset.tab === state.swapMode));
    $('#swap-from-amt').value = state.swapMode === 'buy' ? '100.00' : '0.001';
    swapCalc();
  };
  $$('.swap-tab').forEach(tab => {
    tab.onclick = () => {
      state.swapMode = tab.dataset.tab;
      $$('.swap-tab').forEach(x => x.classList.toggle('active', x === tab));
      $('#swap-from-amt').value = state.swapMode === 'buy' ? '100.00' : '0.001';
      swapCalc();
    };
  });
  $('#swap-cancel').onclick = () => closeModal('modal-crypto');
  $('#swap-go').onclick = () => {
    const fromAmt = parseFloat($('#swap-from-amt').value) || 0;
    if (fromAmt <= 0) { toast(t('invalid_amount'), true); return; }
    const crypto = state.swapCrypto;
    const price = CRYPTO_PRICES[crypto];
    const c = CARDS[state.activeCard];
    if (state.swapMode === 'buy') {
      if (fromAmt > Math.abs(c.balance)) { toast(t('insufficient'), true); return; }
      c.balance -= fromAmt;
      c.txs.unshift({ name:'Buy ' + crypto, time:new Date().toTimeString().slice(0,5), amount:fromAmt, type:'out' });
    } else {
      const usd = fromAmt * price;
      c.balance += usd;
      c.txs.unshift({ name:'Sell ' + crypto, time:new Date().toTimeString().slice(0,5), amount:usd, type:'in' });
    }
    updateBalancePanel(); renderTx();
    closeModal('modal-crypto');
    toast('✓ ' + t('swap_ok'));
  };
}
function cycleCrypto() {
  const keys = Object.keys(CRYPTO_PRICES);
  const idx = keys.indexOf(state.swapCrypto);
  state.swapCrypto = keys[(idx + 1) % keys.length];
  swapCalc();
}
$('#act-crypto').onclick = () => {
  state.swapMode = 'buy'; state.swapCrypto = 'BTC';
  $$('.swap-tab').forEach(tab => tab.classList.toggle('active', tab.dataset.tab === 'buy'));
  $('#swap-from-amt').value = '100.00';
  swapCalc();
  openModal('modal-crypto');
};

/* --- SUBS --- */
function renderSubs() {
  const wrap = $('#subs-list');
  wrap.innerHTML = SUBSCRIPTIONS.map(s => {
    const cancelled = state.subsCancelled.includes(s.id);
    return `
      <div class="sub-row ${cancelled ? 'cancelled' : ''}" data-id="${s.id}">
        <div class="sub-icon" style="color:${s.color}; background:${s.color}22;">${s.icon}</div>
        <div class="sub-info">
          <div class="sub-name">${s.name}</div>
          <div class="sub-price">${money(s.price)} ${t('per_month')}</div>
        </div>
        <button class="sub-btn ${cancelled ? 'restore' : ''}" data-act="${cancelled ? 'restore' : 'cancel'}">
          ${cancelled ? t('subs_restore') : t('subs_cancel')}
        </button>
      </div>`;
  }).join('');
  wrap.querySelectorAll('.sub-btn').forEach(btn => {
    btn.onclick = () => {
      const id = btn.closest('.sub-row').dataset.id;
      const act = btn.dataset.act;
      if (act === 'cancel') {
        if (!state.subsCancelled.includes(id)) state.subsCancelled.push(id);
        toast('✓ ' + t('subs_cancelled'));
      } else {
        state.subsCancelled = state.subsCancelled.filter(x => x !== id);
        toast('✓ ' + t('subs_restored'));
      }
      localStorage.setItem('bank_subs_cancelled', JSON.stringify(state.subsCancelled));
      renderSubs();
    };
  });
}
$('#act-subs').onclick = () => { closeModal('modal-more'); renderSubs(); openModal('modal-subs'); };
$('#subs-close').onclick = () => closeModal('modal-subs');

function aiIntent(s) {
  s = s.toLowerCase();
  if (/(hi|hello|привет|здравств|你好|hola|hallo|bonjour|merhaba|olá|مرحبا|こんにち)/.test(s)) return 'greet';
  if (/(balance|баланс|счет|счёт|остаток|余额|saldo|kontostand|solde|残高|bakiye|رصيد)/.test(s)) return 'balance';
  if (/(cashback|кэшбэк|кешбэк|返现|هدر)/.test(s)) return 'cashback';
  if (/(send|transfer|перевод|отправ|转账|enviar|überweis|virement|送金|gönder|تحويل)/.test(s)) return 'send';
  if (/(block|блок|заблок|冻结|bloque|sperr|bloquer|停止|bloke|حظر)/.test(s)) return 'block';
  if (/(credit|кредит|loan|贷款|crédito|kredit|crédit|ローン|kredi|قرض)/.test(s)) return 'credit';
  if (/(deposit|депозит|вклад|存款|depósito|einlage|dépôt|預金|mevduat|إيداع)/.test(s)) return 'deposit';
  if (/(crypto|крипт|btc|eth|usdt|ton|swap|обмен|兑换|cripto|krypto|暗号|kripto|عملات|صرف)/.test(s)) return 'crypto';
  if (/(sub|подписк|订阅|suscrip|abo|abonn|サブスク|abonelik|اشتراك)/.test(s)) return 'subs';
  if (/(card|карт|卡|tarjeta|karte|carte|カード|kart|بطاقة)/.test(s)) return 'card';
  if (/(security|безопас|安全|seguridad|sicher|sécurité|安全|güvenlik|أمان)/.test(s)) return 'security';
  if (/(thank|спасиб|谢谢|gracias|danke|merci|ありがと|obrigad|teşekkür|شكرا)/.test(s)) return 'thanks';
  return 'fallback';
}
function aiRespond(text) { const r = AI[aiIntent(text)]; return r[state.lang] || r.en || r.ru; }
function appendAiMsg(role, text) {
  const wrap = $('#ai-chat');
  const el = document.createElement('div');
  el.className = 'ai-msg ' + role;
  const b = document.createElement('div'); b.className = 'bubble'; b.textContent = text;
  el.appendChild(b); wrap.appendChild(el); wrap.scrollTop = wrap.scrollHeight;
}
function aiSend(text) {
  if (!text.trim()) return;
  appendAiMsg('user', text);
  $('#ai-input').value = '';
  $('#ai-send').disabled = true;
  const wrap = $('#ai-chat');
  const typing = document.createElement('div');
  typing.className = 'ai-msg bot';
  typing.innerHTML = '<div class="bubble"><span class="ai-typing"><span></span><span></span><span></span></span></div>';
  wrap.appendChild(typing); wrap.scrollTop = wrap.scrollHeight;
  setTimeout(() => {
    typing.remove();
    appendAiMsg('bot', aiRespond(text));
    $('#ai-send').disabled = false;
  }, 320 + Math.random() * 280);
}
