function customSelect(container, options, initialValue, onChange) {
  container.innerHTML = `
    <button type="button" class="cselect-btn">
      <span class="cselect-label"></span>
      <svg class="cselect-arrow" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" width="14" height="14"><polyline points="6 9 12 15 18 9"/></svg>
    </button>
    <div class="cselect-menu"></div>`;
  const menu = container.querySelector('.cselect-menu');
  const labelEl = container.querySelector('.cselect-label');
  const btn = container.querySelector('.cselect-btn');
  let value = initialValue;
  function render() {
    menu.innerHTML = options.map(o => `<div class="cselect-opt${String(o.value) === String(value) ? ' active' : ''}" data-v="${o.value}">${o.label}</div>`).join('');
    const cur = options.find(o => String(o.value) === String(value));
    labelEl.textContent = cur ? cur.label : '';
  }
  btn.addEventListener('click', e => {
    e.stopPropagation();
    document.querySelectorAll('.cselect.open').forEach(s => { if (s !== container) s.classList.remove('open'); });
    container.classList.toggle('open');
  });
  menu.addEventListener('click', e => {
    const opt = e.target.closest('.cselect-opt');
    if (!opt) return;
    value = opt.dataset.v;
    container.classList.remove('open');
    render();
    if (onChange) onChange(value);
  });
  render();
  return { get value() { return value; }, set(v) { value = v; render(); } };
}

function initPinKeypad(keypadEl, displayEl, max, onComplete) {
  displayEl.innerHTML = '';
  const slots = [];
  for (let i = 0; i < max; i++) { const s = document.createElement('div'); s.className = 'pin-slot'; displayEl.appendChild(s); slots.push(s); }
  let v = '';
  const render = (err = false) => slots.forEach((s, i) => { s.classList.toggle('filled', i < v.length); s.classList.toggle('err', err); });
  const press = k => {
    if (k === 'del') { v = v.slice(0, -1); render(); return; }
    if (k === 'ok') { if (v.length === max) onComplete(v); return; }
    if (v.length >= max) return;
    v += k; render(); vibrate(6);
    if (v.length === max) setTimeout(() => onComplete(v), 130);
  };
  keypadEl.querySelectorAll('button').forEach(b => { b.onclick = () => press(b.dataset.k); });
  return { reset() { v = ''; render(); }, setError() { render(true); setTimeout(() => { v = ''; render(false); }, 420); } };
}

let dragStartX = 0, dragStartLeft = 0, lastVibPct = -1;
function onSliderDown(e) {
  if (e.button !== undefined && e.button !== 0) return;
  state.sliderDragging = true;
  $('#slider-thumb').classList.add('dragging');
  dragStartX = e.clientX;
  const track = $('#slider-track'), thumb = $('#slider-thumb');
  const maxLeft = track.clientWidth - thumb.offsetWidth - 12;
  dragStartLeft = (state.sliderPct / 100) * maxLeft;
  e.preventDefault();
  window.addEventListener('pointermove', onSliderMove);
  window.addEventListener('pointerup', onSliderUp);
  window.addEventListener('pointercancel', onSliderUp);
  vibrate(10);
}
function onSliderMove(e) {
  if (!state.sliderDragging) return;
  const track = $('#slider-track'), thumb = $('#slider-thumb');
  const maxLeft = track.clientWidth - thumb.offsetWidth - 12;
  let dx = e.clientX - dragStartX;
  let newLeft = Math.max(0, Math.min(maxLeft, dragStartLeft + dx));
  state.sliderPct = (newLeft / maxLeft) * 100;
  applySlider();
  const p = Math.floor(state.sliderPct / 5) * 5;
  if (p !== lastVibPct) { lastVibPct = p; if (navigator.vibrate) navigator.vibrate(4); }
}
function onSliderUp() {
  if (!state.sliderDragging) return;
  state.sliderDragging = false;
  $('#slider-thumb').classList.remove('dragging');
  window.removeEventListener('pointermove', onSliderMove);
  window.removeEventListener('pointerup', onSliderUp);
  window.removeEventListener('pointercancel', onSliderUp);
  if (state.sliderPct >= 95) state.sliderPct = 100;
  else if (state.sliderPct < 5) state.sliderPct = 0;
  applySlider();
}
$('#slider-track').addEventListener('pointerdown', e => {
  const track = $('#slider-track'), thumb = $('#slider-thumb');
  const rect = track.getBoundingClientRect(), thumbRect = thumb.getBoundingClientRect();
  if (e.clientX >= thumbRect.left && e.clientX <= thumbRect.right) { onSliderDown(e); }
  else {
    const maxLeft = track.clientWidth - thumb.offsetWidth - 12;
    const relX = e.clientX - rect.left - 6 - thumb.offsetWidth / 2;
    state.sliderPct = Math.max(0, Math.min(100, (relX / maxLeft) * 100));
    applySlider(); onSliderDown(e);
  }
});

$('#edit-save').onclick = () => {
  if (!state.pendingEdit) return;
  const { cardIndex, field } = state.pendingEdit;
  const c = CARDS[cardIndex];
  const val = $('#edit-input').value.trim();
  if (field === 'num') {
    const digits = val.replace(/\D/g, '');
    if (digits.length < 16) { toast(t('invalid_card'), true); return; }
    const formatted = digits.slice(0, 16).replace(/(.{4})/g, '$1 ').trim();
    c.num = formatted; saveCardEdit(c.id, 'num', formatted);
  } else {
    const parts = val.split('·').map(s => s.trim());
    const exp = (parts[0] || '').replace(/\s/g, '');
    const cvv = (parts[1] || '').replace(/\D/g, '');
    if (!/^\d{2}\/\d{2}$/.test(exp)) { toast(t('invalid_card'), true); return; }
    if (cvv.length < 3) { toast(t('invalid_card'), true); return; }
    c.exp = exp; c.cvv = cvv;
    saveCardEdit(c.id, 'exp', exp);
    saveCardEdit(c.id, 'cvv', cvv);
  }
  closeModal('modal-edit');
  state.pendingEdit = null;
  renderCards();
  toast('✓ ' + t('save'));
};

function aiIntent(s) {
  s = s.toLowerCase();
  if (/(hi|hello|привет|здравств|你好|hola|hallo|bonjour|merhaba|olá|مرحبا|こんにち)/.test(s)) return 'greet';
  if (/(balance|баланс|счет|счёт|остаток|余额|saldo|kontostand|solde|残高|bakiye|رصيد)/.test(s)) return 'balance';
  if (/(cashback|кэшбэк|кешбэк|返现|هدر)/.test(s)) return 'cashback';
  if (/(send|transfer|перевод|отправ|转账|enviar|überweis|virement|送金|gönder|تحويل)/.test(s)) return 'send';
  if (/(google|play|app ?store|магазин|store|оплат|pay)/.test(s)) return 'store';
  if (/(crypto|крипт|btc|eth|usdt|ton|swap|обмен|兑换|cripto|krypto|暗号|kripto|عملات|صرف)/.test(s)) return 'crypto';
  if (/(card|карт|卡|tarjeta|karte|carte|カード|kart|بطاقة)/.test(s)) return 'card';
  if (/(security|безопас|安全|seguridad|sicher|sécurité|安全|güvenlik|أمان)/.test(s)) return 'security';
  if (/(thank|спасиб|谢谢|gracias|danke|merci|ありがと|obrigad|teşekkür|شكرا)/.test(s)) return 'thanks';
  return 'fallback';
}
