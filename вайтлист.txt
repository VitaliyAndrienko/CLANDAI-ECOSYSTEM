const TOTAL_SUPPLY = 100000000; // 100M CLND
const STORAGE_KEY = 'clandai_whitelist_v1';
const DELETE_TIMEOUT = 60000;

const state = {
  entries: [],
  sliderPct: 0,
  dragging: false,
  dragStartX: 0,
  dragStartPct: 0,
  pending: null,
  typingTimer: null,
  deleteTimers: new Map(),
};

const $ = s => document.querySelector(s);
const $$ = s => document.querySelectorAll(s);

/* ---------- UTIL ---------- */
function toast(msg, err=false){
  const el = $('#toast');
  el.textContent = msg;
  el.className = 'toast show' + (err ? ' err' : '');
  clearTimeout(el._h);
  el._h = setTimeout(() => el.className = 'toast', 2400);
}
function vibrate(ms=6){ if (navigator.vibrate) navigator.vibrate(ms); }
function fmtNumber(n){
  return Math.round(n).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ' ');
}
function shortAddr(a, n=6){
  if (!a) return '';
  if (a.length <= n*2+3) return a;
  return a.slice(0, n) + '…' + a.slice(-4);
}
function isValidAddr(a){
  const s = String(a || '').trim();
  if (!s) return false;
  if (/^0x[0-9a-fA-F]{40}$/.test(s)) return true;
  if (/^0x[0-9a-fA-F]{6,}$/.test(s)) return true;
  if (/^[a-zA-Z0-9_\-\:\.]{4,120}$/.test(s)) return true;
  return false;
}
function escapeHtml(s){
  return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
function pad2(n){ return String(n).padStart(2, '0'); }

/* ---------- STORAGE ---------- */
function load(){
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) { state.entries = []; return; }
    const arr = JSON.parse(raw);
    if (Array.isArray(arr)) state.entries = arr;
  } catch { state.entries = []; }
}
function save(){
  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state.entries)); } catch {}
}

function applySlider(){
  const trackW = slider.clientWidth;
  const thumbW = thumb.offsetWidth || 16;
  const pad = 6;
  const maxLeft = trackW - thumbW - pad*2;
  const left = pad + (state.sliderPct/100) * maxLeft;
  thumb.style.left = left + 'px';
  fill.style.width = (left + thumbW/2) + 'px';
  valueEl.textContent = state.sliderPct.toFixed(1) + '%';
  slider.classList.toggle('dragged', state.sliderPct > 0.5);

  const pct = state.sliderPct;
  const amount = Math.round(TOTAL_SUPPLY * pct / 100);
  $('#amount-display').textContent = fmtNumber(amount) + ' CLND';
  $('#pct-display').textContent = pct.toFixed(2) + '%';

  $$('#limit-presets button').forEach(b => {
    const v = parseFloat(b.dataset.p);
    b.classList.toggle('active', Math.abs(v - pct) < 0.15);
  });
}

function setPct(p){
  state.sliderPct = Math.max(0, Math.min(100, p));
  applySlider();
}

function sliderPointerDown(e){
  state.dragging = true;
  slider.classList.add('dragging');
  dragStart(e.clientX);
  e.preventDefault();
  window.addEventListener('pointermove', sliderPointerMove);
  window.addEventListener('pointerup', sliderPointerUp);
  window.addEventListener('pointercancel', sliderPointerUp);
  vibrate(8);
}
let _dragStartX = 0, _dragStartPct = 0;
function dragStart(x){
  _dragStartX = x;
  _dragStartPct = state.sliderPct;
}
function sliderPointerMove(e){
  if (!state.dragging) return;
  const trackW = slider.clientWidth;
  const thumbW = thumb.offsetWidth || 16;
  const pad = 6;
  const maxLeft = trackW - thumbW - pad*2;
  const dx = e.clientX - _dragStartX;
  const newLeft = Math.max(0, Math.min(maxLeft, (_dragStartPct/100)*maxLeft + dx));
  setPct((newLeft / maxLeft) * 100);
}
function sliderPointerUp(){
  if (!state.dragging) return;
  state.dragging = false;
  slider.classList.remove('dragging');
  window.removeEventListener('pointermove', sliderPointerMove);
  window.removeEventListener('pointerup', sliderPointerUp);
  window.removeEventListener('pointercancel', sliderPointerUp);
}

slider.addEventListener('pointerdown', e => {
  const rect = slider.getBoundingClientRect();
  const thumbRect = thumb.getBoundingClientRect();
  if (e.clientX >= thumbRect.left && e.clientX <= thumbRect.right){
    sliderPointerDown(e);
  } else {
    const trackW = slider.clientWidth;
    const thumbW = thumb.offsetWidth || 16;
    const pad = 6;
    const maxLeft = trackW - thumbW - pad*2;
    const relX = e.clientX - rect.left - pad - thumbW/2;
    setPct(Math.max(0, Math.min(100, (relX/maxLeft)*100)));
    sliderPointerDown(e);
  }
});

function toLocalInputValue(d){
  return `${d.getFullYear()}-${pad2(d.getMonth()+1)}-${pad2(d.getDate())}T${pad2(d.getHours())}:${pad2(d.getMinutes())}`;
}
function setDeadline(d){
  $('#deadline-input').value = toLocalInputValue(d);
}
function getDeadline(){
  const v = $('#deadline-input').value;
  if (!v) return null;
  const d = new Date(v);
  if (isNaN(d.getTime())) return null;
  return d;
}

$('#btn-confirm').onclick = () => {
  const addr = $('#addr-input').value.trim();
  const name = $('#name-input').value.trim();
  const desc = $('#desc-input').value.trim();
  const deadline = getDeadline();

  if (!isValidAddr(addr)){
    toast('Введите корректный адрес', true);
    $('#addr-input').focus();
    return;
  }
  if (state.sliderPct <= 0){
    toast('Установите лимит токенов', true);
    return;
  }
  if (!deadline){
    toast('Укажите дедлайн', true);
    return;
  }
  if (deadline.getTime() <= Date.now()){
    toast('Дедлайн должен быть в будущем', true);
    return;
  }
  const dup = state.entries.find(e => e.address.toLowerCase() === addr.toLowerCase() && !e.deleted);
  if (dup){
    toast('Этот адрес уже в вайтлисте', true);
    return;
  }

  const amount = Math.round(TOTAL_SUPPLY * state.sliderPct / 100);
  state.pending = {
    address: addr,
    name,
    desc,
    pct: state.sliderPct,
    amount,
    deadline: deadline.getTime(),
    createdAt: Date.now(),
  };

  runConfirmAnimation(state.pending);
};

function runConfirmAnimation(p){
  $('#form-panel').style.display = 'none';
  const panel = $('#confirm-panel');
  panel.style.display = 'block';
  window.scrollTo({top:0, behavior:'smooth'});

  const stamp = $('#stamp');
  const typingAddr = $('#typing-addr');
  const typingField = $('#typing-field');
  const title = $('#confirm-title');
  const details = $('#confirm-details');
  const doneBtn = $('#btn-done');

  stamp.classList.remove('show');
  typingAddr.textContent = '';
  typingAddr.className = 'typing-addr typing';
  typingField.className = 'typing-field typing';
  title.className = 'confirm-title';
  title.textContent = 'ПЕЧАТАЕМ АДРЕС...';
  details.classList.remove('show');
  doneBtn.disabled = true;
  doneBtn.textContent = 'ГОТОВО';

  $('#c-amount').textContent = fmtNumber(p.amount) + ' CLND';
  $('#c-pct').textContent = p.pct.toFixed(2) + '%';
  $('#c-deadline').textContent = new Date(p.deadline).toLocaleString('ru-RU', {
    day:'2-digit', month:'2-digit', year:'numeric', hour:'2-digit', minute:'2-digit'
  });

  const chars = p.address.split('');
  let i = 0;
  const speed = Math.max(18, Math.min(50, 1400 / chars.length));
  clearInterval(state.typingTimer);
  state.typingTimer = setInterval(() => {
    if (i >= chars.length){
      clearInterval(state.typingTimer);
      typingAddr.classList.remove('typing');
      typingAddr.classList.add('done');
      typingField.classList.remove('typing');
      typingField.classList.add('done');
      setTimeout(() => {
        stamp.classList.add('show');
        vibrate([20, 40, 20]);
        title.textContent = '✓ АДРЕС ДОБАВЛЕН В WHITELIST';
        title.classList.add('done');
        setTimeout(() => {
          details.classList.add('show');
          doneBtn.disabled = false;
          commitPending();
        }, 250);
      }, 180);
      return;
    }
    typingAddr.textContent += chars[i];
    if (i % 3 === 0) vibrate(3);
    i++;
  }, speed);
}

function commitPending(){
  if (!state.pending) return;
  const p = state.pending;
  const dup = state.entries.find(e => e.address.toLowerCase() === p.address.toLowerCase());
  if (dup){
    Object.assign(dup, { ...p, deleted:false, id: dup.id });
  } else {
    state.entries.unshift({
      id: 'wl_' + Date.now() + '_' + Math.random().toString(36).slice(2,7),
      ...p,
      deleted: false,
    });
  }
  save();
  renderList();
  state.pending = null;
}

function formatTimer(ms){
  if (ms <= 0) return 'EXPIRED';
  const totalMin = Math.floor(ms / 60000);
  const h = Math.floor(totalMin / 60);
  const m = totalMin % 60;
  if (h > 99) return '>99ч';
  return `${h}ч ${pad2(m)}м`;
}

function renderList(){
  const w = $('#wl-list');
  const active = state.entries.filter(e => !e.deleted && !e.removing).length;
  const total = state.entries.filter(e => !e.removing).length;
  $('#wl-count').textContent = total + (active !== total ? ' · ' + active : '');

  const visible = state.entries.filter(e => !e.removing);
  if (!visible.length){
    w.innerHTML = '<div class="wl-empty">Пока пусто. Добавьте первый адрес выше.</div>';
    return;
  }

  w.innerHTML = '';
  const fr = document.createDocumentFragment();
  visible.forEach(e => fr.appendChild(buildItem(e)));
  w.appendChild(fr);

  // restart/resume countdowns
  state.entries.forEach(e => {
    if (e.deleted && e.deletedAt) {
      const elapsed = Date.now() - e.deletedAt;
      const remaining = Math.max(0, DELETE_TIMEOUT - elapsed);
      if (remaining > 0) {
        startCountdown(e.id, remaining);
      } else {
        removeExpired(e.id);
      }
    }
  });

  updateTimers();
}

/* =========================================================
   DELETE / RESTORE
   ========================================================= */
function startDelete(id){
  const e = state.entries.find(x => x.id === id);
  if (!e || e.deleted) return;

  e.deleted = true;
  e.deletedAt = Date.now();
  save();
  renderList();
  vibrate([12, 30, 12]);
  toast('Удаление через 60 секунд · можно отменить');
}

function restoreItem(id){
  const e = state.entries.find(x => x.id === id);
  if (!e) return;
  if (state.deleteTimers.has(id)) {
    clearInterval(state.deleteTimers.get(id));
    state.deleteTimers.delete(id);
  }
  e.deleted = false;
  delete e.deletedAt;
  save();
  renderList();
  toast('✓ Удаление отменено');
}

function removeExpired(id){
  if (state.deleteTimers.has(id)) {
    clearInterval(state.deleteTimers.get(id));
    state.deleteTimers.delete(id);
  }
  state.entries = state.entries.filter(x => x.id !== id);
  save();
  const item = document.querySelector(`.wl-item[data-id="${id}"]`);
  if (item) {
    item.classList.add('removing');
    setTimeout(() => renderList(), 500);
  } else {
    renderList();
  }
}

function startCountdown(id, totalMs){
  if (state.deleteTimers.has(id)) {
    clearInterval(state.deleteTimers.get(id));
    state.deleteTimers.delete(id);
  }

  const item = document.querySelector(`.wl-item[data-id="${id}"]`);
  if (!item) return;
  const banner = item.querySelector('.delete-countdown');
  if (!banner) return;

  const fillEl = banner.querySelector('.delete-countdown-fill');
  const secondsEl = banner.querySelector('.cd-seconds');

  const startTime = Date.now();
  const tick = () => {
    const elapsed = Date.now() - startTime;
    const remaining = Math.max(0, totalMs - elapsed);
    const progress = remaining / totalMs;
    const seconds = Math.ceil(remaining / 1000);
    if (fillEl) fillEl.style.width = (progress * 100) + '%';
    if (secondsEl) secondsEl.textContent = seconds;
    if (remaining <= 0) {
      clearInterval(state.deleteTimers.get(id));
      state.deleteTimers.delete(id);
      removeExpired(id);
    }
  };
  tick();
  const intervalId = setInterval(tick, 100);
  state.deleteTimers.set(id, intervalId);
}

/* ---------- TIMERS (deadline) ---------- */
function updateTimers(){
  const now = Date.now();
  document.querySelectorAll('.wl-timer').forEach(el => {
    const dl = parseInt(el.dataset.deadline, 10);
    if (!dl) return;
    const textEl = el.querySelector('.wl-timer-text');
    const left = dl - now;
    if (left <= 0){
      el.classList.remove('warn');
      el.classList.add('danger');
      textEl.textContent = 'ИСТЁК';
    } else if (left < 3600 * 1000){
      el.classList.remove('danger');
      el.classList.add('warn');
      textEl.textContent = formatTimer(left);
    } else {
      el.classList.remove('warn', 'danger');
      textEl.textContent = formatTimer(left);
    }
  });
}
setInterval(updateTimers, 1000);
