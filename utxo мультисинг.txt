Ниже — консолидированный MVP. Всё, что обсуждалось в ревью, сведено в один исполняемый каркас: Bitcoin (PSBT) + Bare-legacy (Dogecoin/BCH/Zcash) + Modern-legacy (Dash/Qtum/RVN). Manifest с key separation, worst-case vsize с явной маркировкой, strict types.

Цель MVP: **Bitcoin signet — рабочий 4-of-5 от создания адреса до broadcast**, остальные сети — тот же код, но с явным отказом, если узел не проходит capability probe.

## `core/amounts.py`

```python
from decimal import Decimal, ROUND_DOWN

SAT = 100_000_000
_Q = Decimal("0.00000001")


class AmountError(ValueError):
    pass


def sats_to_btc(sats: int) -> Decimal:
    if isinstance(sats, bool) or not isinstance(sats, int):
        raise AmountError(f"sats must be int, got {type(sats).__name__}")
    if sats < 0:
        raise AmountError(f"sats must be >= 0, got {sats}")
    return (Decimal(sats) / Decimal(SAT)).quantize(_Q, rounding=ROUND_DOWN)


def btc_to_sats(btc) -> int:
    if isinstance(btc, float):
        raise AmountError("float not allowed; use Decimal or str")
    d = Decimal(str(btc))
    q = d.quantize(_Q, rounding=ROUND_DOWN)
    if q != d:
        raise AmountError(f"{d} has more than 8 decimals")
    return int(q * SAT)


def strict_sats(v) -> int:
    if isinstance(v, bool) or not isinstance(v, int):
        raise AmountError(f"amount_sats must be strict int, got {type(v).__name__}")
    if v <= 0:
        raise AmountError(f"amount_sats must be > 0, got {v}")
    return v
```

```python
import json
import os
import stat

CONF_PATH = os.path.expanduser(
    os.environ.get("CLANDAI_CONF", "~/.clandai/nodes.json")
)


def load_node_config(network: str) -> dict:
    st = os.stat(CONF_PATH)
    mode = stat.S_IMODE(st.st_mode)
    if mode & (stat.S_IRWXG | stat.S_IRWXO):
        raise PermissionError(
            f"{CONF_PATH} must be chmod 600 (got {oct(mode)}). "
            "RPC credentials must not be group/world readable."
        )
    with open(CONF_PATH) as f:
        data = json.load(f)
    if network not in data:
        raise KeyError(f"network {network!r} not in {CONF_PATH}")
    return data[network]
```

## `core/rpc.py`

```python
import json
import requests


class RPCError(RuntimeError):
    def __init__(self, method, error):
        super().__init__(f"RPC {method} failed: {error}")
        self.method = method
        self.error = error


class RPC:
    def __init__(self, url, user, password, wallet=None, timeout=30):
        self._base = url.rstrip("/")
        self._url = f"{self._base}/wallet/{wallet}" if wallet else self._base
        self._auth = (user, password)
        self._timeout = timeout
        self._id = 0

    def call(self, method, params=None):
        self._id += 1
        payload = {
            "jsonrpc": "1.0",
            "id": self._id,
            "method": method,
            "params": params if params is not None else [],
        }
        r = requests.post(
            self._url,
            data=json.dumps(payload),
            auth=self._auth,
            timeout=self._timeout,
            headers={"Content-Type": "application/json"},
        )
        r.raise_for_status()
        body = r.json()
        if body.get("error"):
            raise RPCError(method, body["error"])
        return body["result"]

    def with_wallet(self, wallet_name: str) -> "RPC":
        return RPC(
            self._base, self._auth[0], self._auth[1],
            wallet=wallet_name, timeout=self._timeout,
        )
```

```python
from .rpc import RPC

# Valid secp256k1 points, used only for capability probing.
_G = "0279be667ef9dcbbac55a06295ce870b07029bfcdb2dce28d959f2815b16f81798"
_2G = "02c6047f9441ed7d6d3045406e95c07cd85c778e4b8cef3ca7abac09b95c709ee5"


def _methods(rpc: RPC) -> set[str]:
    raw = rpc.call("help")
    out = set()
    for line in raw.splitlines():
        line = line.strip()
        if not line or line.startswith("=="):
            continue
        parts = line.split()
        if parts:
            out.add(parts[0])
    return out


def probe(rpc: RPC) -> dict:
    info = rpc.call("getnetworkinfo")
    methods = _methods(rpc)

    caps = {
        "version": info.get("version"),
        "subversion": info.get("subversion"),
        "protocol": info.get("protocolversion"),
        "chain": info.get("chain"),
        "relay_fee_btc_per_kvb": info.get("relayfee"),
        "methods": methods,
        "has_psbt": {
            "walletcreatefundedpsbt",
            "walletprocesspsbt",
            "combinepsbt",
            "finalizepsbt",
            "decodepsbt",
        } <= methods,
        "has_descriptors": {
            "getdescriptorinfo",
            "importdescriptors",
            "deriveaddresses",
        } <= methods,
        "has_testmempoolaccept": "testmempoolaccept" in methods,
        "has_sortedmulti": False,
    }

    if caps["has_descriptors"]:
        try:
            rpc.call(
                "getdescriptorinfo",
                [f"wsh(sortedmulti(1,{_G},{_2G}))"],
            )
            caps["has_sortedmulti"] = True
        except Exception:
            caps["has_sortedmulti"] = False

    return caps
```

## `core/policy.py`

```python
from __future__ import annotations
import hashlib
from dataclasses import dataclass

REQUIRED = 4
TOTAL = 5
VALID_KINDS = {"wsh", "sh", "sh-wsh"}


class PolicyError(Exception):
    pass


def _normalize_pubkey(pk: str) -> str:
    if not isinstance(pk, str):
        raise PolicyError(f"pubkey must be str, got {type(pk).__name__}")
    pk = pk.strip().lower()
    if len(pk) not in (66, 130):
        raise PolicyError(f"pubkey must be 33 or 65 bytes hex, got len {len(pk)}")
    try:
        bytes.fromhex(pk)
    except ValueError as e:
        raise PolicyError(f"pubkey not hex: {pk!r}") from e
    return pk


@dataclass(frozen=True)
class MultisigPolicy:
    pubkeys: tuple[str, ...]

    def __init__(self, pubkeys):
        normalized = tuple(sorted(_normalize_pubkey(pk) for pk in pubkeys))
        if len(normalized) != TOTAL:
            raise PolicyError(f"4-of-5 requires {TOTAL} pubkeys, got {len(normalized)}")
        if len(set(normalized)) != TOTAL:
            raise PolicyError("duplicate pubkeys in policy")
        object.__setattr__(self, "pubkeys", normalized)

    def descriptor(self, kind: str = "wsh") -> str:
        if kind not in VALID_KINDS:
            raise PolicyError(f"unsupported descriptor kind: {kind!r}")
        keys = ",".join(self.pubkeys)
        inner = f"sortedmulti({REQUIRED},{keys})"
        if kind == "wsh":
            return f"wsh({inner})"
        if kind == "sh":
            return f"sh({inner})"
        return f"sh(wsh({inner}))"

    def fingerprint(self) -> str:
        h = hashlib.sha256("|".join(self.pubkeys).encode()).hexdigest()
        return h[:16]
```

## `core/security.py`

Worst-case vsize. Явно документируем: это **верхняя граница**, а не оценка. Для реальной проверки после финализации нужен `decoderawtransaction`, но для pre-signature это правильный conservative подход.

```python
from __future__ import annotations
from dataclasses import dataclass

SAT = 100_000_000


class SecurityError(RuntimeError):
    pass


@dataclass
class ExpectedOutput:
    script_pubkey_hex: str
    amount_sats: int


def _varint_len(n: int) -> int:
    if n < 0xFD:
        return 1
    if n <= 0xFFFF:
        return 3
    if n <= 0xFFFFFFFF:
        return 5
    return 9


def _push_len(n: int) -> int:
    if n < 0x4C:
        return 1 + n
    if n <= 0xFF:
        return 2 + n
    if n <= 0xFFFF:
        return 3 + n
    return 5 + n


class TransactionSecurityEngine:
    """Runs on every signer before signing.

    Reconstructs everything from PSBT bytes; never trusts coordinator.
    Uses worst-case vsize for fee-rate checks — documented upper bound.
    """

    def __init__(self, adapter, *,
                 max_fee_rate_sats_per_vb: int = 200,
                 max_abs_fee_sats: int = 5_000_000):
        self.adapter = adapter
        self.policy_spk = adapter.policy_script_pubkey_hex().lower()
        self.max_fee_rate = max_fee_rate_sats_per_vb
        self.max_abs_fee = max_abs_fee_sats
        self.dust_threshold = self._resolve_dust_threshold(adapter)

    @staticmethod
    def _resolve_dust_threshold(adapter) -> int:
        fallback = getattr(adapter, "dust_threshold_sats", 546)
        try:
            relay = adapter.caps.get("relay_fee_btc_per_kvb")
            if not relay:
                return fallback
            # Bitcoin Core: dust = 3 * relay_fee_per_kvb * (output_size + 148)
            # Use P2WSH output size (8 + 1 + 34 = 43) as reference.
            dust = int(relay * SAT / 1000 * (43 + 148) * 3)
            return max(dust, fallback)
        except Exception:
            return fallback

    # ------------------------------------------------------------------
    def verify_psbt(self, psbt_b64: str,
                    expected: list[ExpectedOutput]) -> dict:
        decoded = self.adapter.rpc.call("decodepsbt", [psbt_b64])
        tx = decoded.get("tx") or {}
        vin = tx.get("vin", [])
        vout = tx.get("vout", [])
        inputs = decoded.get("inputs", [])

        if not inputs or not vin:
            raise SecurityError("PSBT has no inputs")

        # 1) Inputs spend our exact policy script.
        for i, (raw_in, dec_in) in enumerate(zip(vin, inputs)):
            spk = self._input_script_pubkey(raw_in, dec_in, i)
            if spk != self.policy_spk:
                raise SecurityError(
                    f"input[{i}] spends {spk}, expected policy {self.policy_spk}"
                )

        # 2) Outputs: exact match; anything else must be one change output.
        matched = [False] * len(expected)
        change_count = 0
        change_sats = 0
        dust = []

        for v in vout:
            spk = v["scriptPubKey"]["hex"].lower()
            value = int(round(v["value"] * SAT))
            hit = False
            for j, exp in enumerate(expected):
                if spk == exp.script_pubkey_hex.lower() and value == exp.amount_sats:
                    matched[j] = True
                    hit = True
                    break
            if hit:
                continue
            if spk == self.policy_spk:
                change_count += 1
                change_sats += value
                if value < self.dust_threshold:
                    dust.append((spk, value))
                continue
            raise SecurityError(
                f"unexpected output: script={spk} value_sats={value}"
            )

        if change_count > 1:
            raise SecurityError(
                f"change splitting: {change_count} outputs to policy (max 1)"
            )
        if dust:
            raise SecurityError(
                f"dust change below {self.dust_threshold} sats: {dust}"
            )
        for j, ok in enumerate(matched):
            if not ok:
                e = expected[j]
                raise SecurityError(
                    f"expected output missing: {e.script_pubkey_hex} "
                    f"{e.amount_sats}"
                )

        # 3) Fee — worst-case vsize, documented as upper bound.
        in_total = self._input_total_sats(inputs)
        out_total = sum(int(round(v["value"] * SAT)) for v in vout)
        fee = in_total - out_total
        if fee < 0:
            raise SecurityError(f"negative fee: {fee}")

        vsize_wc = self._worst_case_vsize(vin, vout, inputs)
        fee_rate_wc = fee / vsize_wc if vsize_wc else float("inf")

        if fee_rate_wc > self.max_fee_rate:
            raise SecurityError(
                f"worst-case fee rate {fee_rate_wc:.2f} sats/vB > "
                f"{self.max_fee_rate} (fee={fee}, vsize_wc={vsize_wc})"
            )
        if fee > self.max_abs_fee:
            raise SecurityError(f"fee {fee} > hard cap {self.max_abs_fee}")

        return {
            "inputs": len(inputs),
            "outputs": len(vout),
            "fee_sats": fee,
            "vsize_worst_case": vsize_wc,
            "fee_rate_worst_case_sats_per_vb": round(fee_rate_wc, 2),
            "change_count": change_count,
            "change_sats": change_sats,
        }

    # ------------------------------------------------------------------
    @staticmethod
    def _input_script_pubkey(raw_in, dec_in, idx) -> str:
        wu = dec_in.get("witness_utxo")
        if wu and "scriptPubKey" in wu:
            return wu["scriptPubKey"]["hex"].lower()
        nwu = dec_in.get("non_witness_utxo")
        if nwu:
            return nwu["vout"][int(raw_in["vout"])]["scriptPubKey"]["hex"].lower()
        raise SecurityError(f"input[{idx}] lacks UTXO data")

    @staticmethod
    def _input_total_sats(inputs) -> int:
        total = 0
        for inp in inputs:
            wu = inp.get("witness_utxo")
            if wu and "value" in wu:
                total += int(round(wu["value"] * SAT))
                continue
            nwu = inp.get("non_witness_utxo")
            if nwu:
                raise SecurityError(
                    "non-witness input without explicit prevtx"
                )
        return total

    @staticmethod
    def _worst_case_vsize(vin, vout, inputs) -> int:
        """Upper bound for vsize of the fully-signed 4-of-5 tx.

        Uses maximum DER sig length (73 + sighash) and exact push/varint
        lengths. Never underestimates — this is what fee-rate checks need.
        """
        is_segwit = any("witness_utxo" in i for i in inputs)

        outputs_bytes = 0
        for v in vout:
            n = len(v["scriptPubKey"]["hex"]) // 2
            outputs_bytes += 8 + _varint_len(n) + n

        n_in, n_out = len(vin), len(vout)

        if is_segwit:
            # P2WSH 4-of-5 witness: dummy(1) + 4 * push(73) + push(redeemScript 173)
            # Each stack item prefixed by its own varint length.
            redeem = 173
            witness_per_input = (
                1 + _varint_len(1)                    # dummy 0
                + 4 * (1 + _varint_len(73) + 73)      # sigs
                + 1 + _varint_len(redeem) + redeem    # witnessScript
            )
            base = 4 + 4 + _varint_len(n_in) + n_in * 41 + _varint_len(n_out) + outputs_bytes
            weight = base * 4 + 2 + n_in * witness_per_input
            return (weight + 3) // 4
        else:
            # P2SH 4-of-5: scriptSig = OP_0 + 4 * push(73) + push(173)
            script_sig = 1 + 4 * _push_len(73) + _push_len(173)
            input_bytes = 36 + _varint_len(script_sig) + script_sig + 4
            return (
                4 + 4 + _varint_len(n_in) + n_in * input_bytes
                + _varint_len(n_out) + outputs_bytes
            )
```

## `core/manifest.py`

```python
from __future__ import annotations
import json
from pathlib import Path

from ecdsa import VerifyingKey, SECP256k1
from ecdsa.errors import BadSignatureError, MalformedPointError
from ecdsa.util import sigdecode_der

from .security import ExpectedOutput, SecurityError
from .amounts import strict_sats

REQUIRED_FIELDS = {
    "network", "fingerprint", "nonce", "created_at",
    "outputs", "signature", "signer_pubkey",
}


class ManifestVerifier:
    """ECDSA-secp256k1 verifier for out-of-band transaction manifests.

    Enforces:
      - mandatory network + policy fingerprint binding (anti-replay),
      - mandatory nonce + created_at,
      - strict int amounts,
      - signer pubkey inside an explicit trusted set (NOT necessarily
        the same keys that sign the PSBT).
    """

    def __init__(self, trusted_pubkeys):
        self.trusted = {pk.lower() for pk in trusted_pubkeys}

    def verify_and_load(self, path: str, adapter) -> list[ExpectedOutput]:
        raw = Path(path).read_bytes()
        try:
            data = json.loads(raw.decode("utf-8"))
        except Exception as e:
            raise SecurityError(f"invalid manifest JSON: {e}") from e

        missing = REQUIRED_FIELDS - set(data.keys())
        if missing:
            raise SecurityError(f"manifest missing fields: {sorted(missing)}")

        # Binding: network + policy
        if data["network"] != adapter.name:
            raise SecurityError(
                f"cross-network replay rejected: manifest {data['network']!r} "
                f"vs adapter {adapter.name!r}"
            )
        if data["fingerprint"] != adapter.fingerprint():
            raise SecurityError(
                f"policy mismatch: manifest {data['fingerprint']!r} "
                f"vs adapter {adapter.fingerprint()!r}"
            )

        signer = str(data["signer_pubkey"]).lower()
        if signer not in self.trusted:
            raise SecurityError(
                f"signer {signer} not in authorized manifest key set"
            )

        canonical = self._canonical_core_bytes(data)

        try:
            pk_bytes = bytes.fromhex(signer)
            sig_bytes = bytes.fromhex(str(data["signature"]))
            vk = VerifyingKey.from_string(
                pk_bytes, curve=SECP256k1,
                valid_encodings=["compressed", "uncompressed", "hybrid"],
            )
            vk.verify(sig_bytes, canonical, sigdecode=sigdecode_der)
        except (BadSignatureError, MalformedPointError, ValueError) as e:
            raise SecurityError(f"manifest signature invalid: {e}") from e

        outputs: list[ExpectedOutput] = []
        for entry in data["outputs"]:
            addr = str(entry["address"])
            sats = strict_sats(entry["amount_sats"])
            info = adapter.rpc.call("validateaddress", [addr])
            if not info.get("isvalid"):
                raise SecurityError(
                    f"address invalid on {adapter.name}: {addr}"
                )
            outputs.append(ExpectedOutput(info["scriptPubKey"].lower(), sats))
        return outputs

    @staticmethod
    def _canonical_core_bytes(data: dict) -> bytes:
        core = {
            "created_at": int(data["created_at"]),
            "fingerprint": str(data["fingerprint"]),
            "network": str(data["network"]),
            "nonce": str(data["nonce"]),
            "outputs": [
                {
                    "address": str(o["address"]),
                    "amount_sats": strict_sats(o["amount_sats"]),
                }
                for o in data["outputs"]
            ],
        }
        return json.dumps(
            core, sort_keys=True, separators=(",", ":"), ensure_ascii=True,
        ).encode("ascii")


# ---- Coordinator-side helper (separate key from PSBT signers) ----

def sign_manifest(core: dict, signer_privkey_hex: str) -> dict:
    """Coordinator tool. NEVER ships in the signer's environment."""
    from ecdsa import SigningKey
    sk = SigningKey.from_string(bytes.fromhex(signer_privkey_hex),
                                curve=SECP256k1)
    payload = ManifestVerifier._canonical_core_bytes({**core, "signature": "x",
                                                      "signer_pubkey": "x"})
    sig = sk.sign(payload, sigencode_der := __import__("ecdsa").util.sigencode_der)
    vk = sk.get_verifying_key()
    pk_hex = (b"\x02" if vk.pubkey.point.y() % 2 == 0 else b"\x03") + \
             vk.pubkey.point.x().to_bytes(32, "big")
    return {**core, "signature": sig.hex(), "signer_pubkey": pk_hex.hex()}
```

## `adapters/base.py`

```python
from __future__ import annotations
from abc import ABC, abstractmethod

from ..core.capabilities import probe
from ..core.policy import MultisigPolicy
from ..core.rpc import RPC


class AdapterCapabilityError(RuntimeError):
    pass


class BaseAdapter(ABC):
    name = "base"
    required_caps: tuple[str, ...] = ()
    dust_threshold_sats = 546

    def __init__(self, rpc: RPC, policy: MultisigPolicy, *,
                 address_type: str = "bech32"):
        self.rpc = rpc
        self.policy = policy
        self.address_type = address_type
        self.caps = probe(rpc)
        self._check_caps()

    def _check_caps(self):
        missing = [c for c in self.required_caps if not self.caps.get(c)]
        if missing:
            raise AdapterCapabilityError(
                f"{self.name}: node lacks {missing} "
                f"(subversion={self.caps.get('subversion')})"
            )

    @abstractmethod
    def descriptor_kind(self) -> str: ...

    @abstractmethod
    def address(self) -> str: ...

    @abstractmethod
    def policy_script_pubkey_hex(self) -> str: ...

    def descriptor(self) -> str:
        raw = self.policy.descriptor(self.descriptor_kind())
        return self.rpc.call("getdescriptorinfo", [raw])["descriptor"]

    def fingerprint(self) -> str:
        return self.policy.fingerprint()

    def with_wallet(self, wallet_name: str) -> "BaseAdapter":
        return type(self)(
            self.rpc.with_wallet(wallet_name),
            self.policy,
            address_type=self.address_type,
        )
```

## `adapters/bitcoin.py`

```python
from __future__ import annotations
from ..core.amounts import sats_to_btc
from ..core.capabilities import CapabilityError
from .base import BaseAdapter


class BitcoinAdapter(BaseAdapter):
    name = "bitcoin"
    required_caps = ("has_descriptors", "has_sortedmulti", "has_psbt")
    dust_threshold_sats = 546

    def descriptor_kind(self) -> str:
        return "wsh" if self.address_type == "bech32" else "sh"

    def address(self) -> str:
        addrs = self.rpc.call("deriveaddresses", [self.descriptor()])
        if len(addrs) != 1:
            raise RuntimeError(f"expected 1 address, got {len(addrs)}")
        return addrs[0]

    def policy_script_pubkey_hex(self) -> str:
        info = self.rpc.call("validateaddress", [self.address()])
        if not info.get("isvalid") or "scriptPubKey" not in info:
            raise RuntimeError(f"node refused {self.address()!r}")
        return info["scriptPubKey"].lower()

    def ensure_watch_wallet(self, wallet_name: str) -> "BitcoinAdapter":
        try:
            self.rpc.call(
                "createwallet",
                [wallet_name, True, True, "", False, True],
            )
        except Exception as e:
            if "already exists" not in str(e):
                raise
        scoped = self.with_wallet(wallet_name)
        scoped.rpc.call("importdescriptors", [[{
            "desc": scoped.descriptor(),
            "active": True,
            "internal": False,
            "timestamp": "now",
            "range": [0, 0],
        }]])
        return scoped

    def create_psbt(self, outputs_sats: dict[str, int], *,
                    fee_rate: int | None = None) -> str:
        outputs = {
            addr: float(sats_to_btc(sats))
            for addr, sats in outputs_sats.items()
        }
        opts: dict = {"includeWatching": True, "add_inputs": True}
        if fee_rate:
            opts["fee_rate"] = fee_rate
        return self.rpc.call(
            "walletcreatefundedpsbt", [[], outputs, 0, opts],
        )["psbt"]

    def sign_psbt(self, psbt_b64: str) -> tuple[str, bool]:
        res = self.rpc.call("walletprocesspsbt", [psbt_b64])
        return res["psbt"], res["complete"]

    def combine_psbts(self, psbts: list[str]) -> str:
        return self.rpc.call("combinepsbt", [psbts])

    def finalize_psbt(self, psbt_b64: str) -> dict:
        return self.rpc.call("finalizepsbt", [psbt_b64])

    def send_raw(self, hex_tx: str) -> str:
```

```python
#!/usr/bin/env python3
import argparse
import json
import sys

from core.config import load_node_config
from core.policy import MultisigPolicy
from core.rpc import RPC
from core.security import TransactionSecurityEngine
from core.manifest import ManifestVerifier

from adapters.bitcoin import BitcoinAdapter
from adapters.legacy import (
    ModernLegacyAdapter, DogecoinAdapter, BitcoinCashAdapter,
)
from adapters.zcash_transparent import ZcashTransparentAdapter


ADAPTERS = {
    "bitcoin":     BitcoinAdapter,
    "litecoin":    BitcoinAdapter,
    "dash":        ModernLegacyAdapter,
    "qtum":        ModernLegacyAdapter,
    "ravencoin":   ModernLegacyAdapter,
    "dogecoin":    DogecoinAdapter,
    "bitcoincash": BitcoinCashAdapter,
    "zcash":       ZcashTransparentAdapter,
}


def _build(network: str, wallet: str | None, policy: MultisigPolicy):
    cfg = load_node_config(network)
    rpc = RPC(cfg["rpc_url"], cfg["rpc_user"], cfg["rpc_pass"], wallet=wallet)
    return ADAPTERS[network](
        rpc, policy, address_type=cfg.get("address_type", "bech32")
    )


def _manifest_keys(args, policy):
    """Manifest key set is separate from PSBT signer set."""
    if args.manifest_pubkeys:
        return tuple(args.manifest_pubkeys.split(","))
    # Fallback: policy keys. Log a warning — not recommended for mainnet.
    print(
        "[warn] manifest key separation not configured; using policy keys",
        file=sys.stderr,
    )
    return policy.pubkeys


def cmd_create(args, policy):
    a = _build(args.network, args.wallet, policy)
    print(json.dumps({
        "network": args.network,
        "fingerprint": a.fingerprint(),
        "descriptor": a.descriptor() if a.caps["has_descriptors"] else None,
        "address": a.address(),
        "script_pubkey": a.policy_script_pubkey_hex(),
        "node": {
            "version": a.caps["version"],
            "subversion": a.caps["subversion"],
            "has_psbt": a.caps["has_psbt"],
            "has_sortedmulti": a.caps["has_sortedmulti"],
        },
    }, indent=2))


def cmd_import_watch(args, policy):
    a = _build(args.network, args.wallet, policy)
    if not hasattr(a, "ensure_watch_wallet"):
        sys.exit(f"{args.network}: watch-only import not implemented")
    if not args.wallet:
        sys.exit("--wallet required")
    a.ensure_watch_wallet(args.wallet)
    print("OK")


def cmd_psbt_create(args, policy):
    a = _build(args.network, args.wallet, policy)
    outputs = {}
    for item in args.to:
        addr, sats = item.rsplit(":", 1)
        outputs[addr] = int(sats)
    print(a.create_psbt(outputs, fee_rate=args.fee_rate))


def cmd_psbt_sign_safe(args, policy):
    a = _build(args.network, args.wallet, policy)
    if not a.caps["has_psbt"]:
        sys.exit(f"{args.network}: node lacks PSBT support")

    verifier = ManifestVerifier(_manifest_keys(args, policy))
    expected = verifier.verify_and_load(args.manifest, a)

    engine = TransactionSecurityEngine(
        a, max_fee_rate_sats_per_vb=args.max_fee_rate,
    )
    report = engine.verify_psbt(args.psbt, expected)
    print("[verify] OK:", json.dumps(report, indent=2))

    signed, complete = a.sign_psbt(args.psbt)
    print(json.dumps({"psbt": signed, "complete": complete}))


def cmd_psbt_combine(args, policy):
    a = _build(args.network, args.wallet, policy)
    print(a.combine_psbts(args.psbt))


def cmd_psbt_finalize(args, policy):
    a = _build(args.network, args.wallet, policy)
    res = a.finalize_psbt(args.psbt)
    print(json.dumps(res, indent=2))
    if res.get("complete") and "hex" in res:
        # Report true vsize from the final tx.
        dec = a.rpc.call("decoderawtransaction", [res["hex"]])
        print(json.dumps({
            "vsize": dec["vsize"],
            "weight": dec["weight"],
        }, indent=2))


def cmd_send(args, policy):
    a = _build(args.network, args.wallet, policy)
    print(a.send_raw(args.hex))


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--network", required=True, choices=sorted(ADAPTERS))
    p.add_argument("--wallet", default=None)
    p.add_argument("--pubkeys", required=True,
                   help="5 pubkeys comma-separated")
    p.add_argument("--manifest-pubkeys", default=None,
                   help="trusted manifest signer keys (separate from policy)")
    sub = p.add_subparsers(dest="cmd", required=True)

    sub.add_parser("create")
    sub.add_parser("import-watch")

    c = sub.add_parser("psbt-create")
    c.add_argument("--to", action="append", required=True,
                   help="recipient_addr:amount_sats")
    c.add_argument("--fee-rate", type=int, default=None)

    s = sub.add_parser("psbt-sign-safe")
    s.add_argument("--psbt", required=True)
    s.add_argument("--manifest", required=True)
    s.add_argument("--max-fee-rate", type=int, default=200,
                   help="sats/vB worst-case cap")

    comb = sub.add_parser("psbt-combine")
    comb.add_argument("--psbt", action="append", required=True)

    fin = sub.add_parser("psbt-finalize")
    fin.add_argument("--psbt", required=True)

    sd = sub.add_parser("send")
    sd.add_argument("--hex", required=True)

    args = p.parse_args()
    policy = MultisigPolicy(args.pubkeys.split(","))

    {
        "create": cmd_create,
        "import-watch": cmd_import_watch,
        "psbt-create": cmd_psbt_create,
        "psbt-sign-safe": cmd_psbt_sign_safe,
        "psbt-combine": cmd_psbt_combine,
        "psbt-finalize": cmd_psbt_finalize,
        "send": cmd_send,
    }[args.cmd](args, policy)


if __name__ == "__main__":
    main()
```

## 1. `core/security.py` — добавляю `verify_raw`

Общая логика выходов вынесена в `_verify_outputs`, чтобы PSBT- и raw-пути не расходились. Raw-путь требует `prevtxs`: без них нельзя восстановить ни scriptPubKey, ни value входов, значит нельзя вычислить ни fee, ни fee rate.

```python
# core/security.py — добавления к существующему классу

from typing import Iterable


class TransactionSecurityEngine:
    # ... __init__ без изменений ...

    # =================================================================
    # PSBT path (как было)
    # =================================================================
    def verify_psbt(self, psbt_b64, expected):
        decoded = self.adapter.rpc.call("decodepsbt", [psbt_b64])
        tx = decoded.get("tx") or {}
        vin, vout, inputs = tx.get("vin", []), tx.get("vout", []), decoded.get("inputs", [])
        if not inputs or not vin:
            raise SecurityError("PSBT has no inputs")

        input_spks, input_total = [], 0
        for i, (raw_in, dec_in) in enumerate(zip(vin, inputs)):
            spk = self._psbt_input_spk(raw_in, dec_in, i)
            value = self._psbt_input_value(dec_in, i)
            input_spks.append(spk)
            input_total += value

        self._check_all_inputs_policy(input_spks)

        out_report = self._verify_outputs(vout, expected)
        fee = input_total - out_report["total_out_sats"]
        return self._finalize_report(
            vin, vout, input_spks, out_report, fee,
            segwit=any("witness_utxo" in x for x in inputs),
        )

    # =================================================================
    # Raw (legacy) path
    # =================================================================
    def verify_raw(self, hex_tx: str, expected: list[ExpectedOutput],
                   prevtxs: list[dict]) -> dict:
        """Verify a raw (non-PSBT) transaction before signing.

        `prevtxs` uses the same schema as `signrawtransactionwithwallet`:
            {"txid": <hex>, "vout": <int>,
             "scriptPubKey": <hex>, "amount": <BTC float>}

        Never trust the caller's hex alone: every input script and amount
        is pulled from prevtxs and re-checked against the policy.
        """
        if not prevtxs:
            raise SecurityError("raw verification requires prevtxs")

        decoded = self.adapter.rpc.call("decoderawtransaction", [hex_tx])
        vin, vout = decoded.get("vin", []), decoded.get("vout", [])
        if not vin:
            raise SecurityError("raw tx has no inputs")

        by_outpoint = {f"{p['txid']}:{int(p['vout'])}": p for p in prevtxs}

        input_spks, input_total = [], 0
        for i, raw_in in enumerate(vin):
            key = f"{raw_in['txid']}:{int(raw_in['vout'])}"
            prev = by_outpoint.get(key)
            if not prev:
                raise SecurityError(
                    f"input[{i}] {key} missing from prevtxs — cannot verify"
                )
            spk = str(prev["scriptPubKey"]).lower()
            value = int(round(float(prev["amount"]) * SAT))
            input_spks.append(spk)
            input_total += value

        self._check_all_inputs_policy(input_spks)

        out_report = self._verify_outputs(vout, expected)
        fee = input_total - out_report["total_out_sats"]
        # Raw = pre-segwit by definition for our legacy adapters.
        return self._finalize_report(
            vin, vout, input_spks, out_report, fee, segwit=False,
        )

    # =================================================================
    # Shared helpers
    # =================================================================
    def _check_all_inputs_policy(self, spks: list[str]) -> None:
        for i, spk in enumerate(spks):
            if spk != self.policy_spk:
                raise SecurityError(
                    f"input[{i}] spends {spk}, expected policy {self.policy_spk}"
                )

    def _verify_outputs(self, vout: list, expected: list[ExpectedOutput]) -> dict:
        matched = [False] * len(expected)
        change_count = 0
        change_sats = 0
        dust: list[tuple[str, int]] = []
        total_out = 0

        for v in vout:
            spk = v["scriptPubKey"]["hex"].lower()
            value = int(round(v["value"] * SAT))
            total_out += value

            hit = False
            for j, exp in enumerate(expected):
                if spk == exp.script_pubkey_hex.lower() and value == exp.amount_sats:
                    matched[j] = True
                    hit = True
                    break
            if hit:
                continue

            if spk == self.policy_spk:
                change_count += 1
                change_sats += value
                if value < self.dust_threshold:
                    dust.append((spk, value))
                continue

            raise SecurityError(
                f"unexpected output: script={spk} value_sats={value}"
            )

        if change_count > 1:
            raise SecurityError(
                f"change splitting: {change_count} outputs to policy (max 1)"
            )
        if dust:
            raise SecurityError(
                f"dust change below {self.dust_threshold} sats: {dust}"
            )

        for j, ok in enumerate(matched):
            if not ok:
                e = expected[j]
                raise SecurityError(
                    f"expected output missing: script={e.script_pubkey_hex} "
                    f"amount_sats={e.amount_sats}"
                )

        return {
            "total_out_sats": total_out,
            "change_count": change_count,
            "change_sats": change_sats,
        }

    def _finalize_report(self, vin, vout, input_spks, out_report,
                         fee, *, segwit: bool) -> dict:
        if fee < 0:
            raise SecurityError(f"negative fee: {fee}")

        vsize_wc = self._worst_case_vsize(vin, vout, segwit)
        rate_wc = fee / vsize_wc if vsize_wc else float("inf")

        if rate_wc > self.max_fee_rate:
            raise SecurityError(
                f"worst-case fee rate {rate_wc:.2f} sats/vB > "
                f"{self.max_fee_rate} (fee={fee}, vsize_wc={vsize_wc})"
            )
        if fee > self.max_abs_fee:
            raise SecurityError(
                f"fee {fee} > hard cap {self.max_abs_fee}"
            )

        return {
            "inputs": len(vin),
            "outputs": len(vout),
            "fee_sats": fee,
            "vsize_worst_case": vsize_wc,
            "fee_rate_worst_case_sats_per_vb": round(rate_wc, 2),
            "change_count": out_report["change_count"],
            "change_sats": out_report["change_sats"],
            "segwit": segwit,
        }

    @staticmethod
    def _psbt_input_spk(raw_in, dec_in, idx) -> str:
        wu = dec_in.get("witness_utxo")
        if wu and "scriptPubKey" in wu:
            return wu["scriptPubKey"]["hex"].lower()
        nwu = dec_in.get("non_witness_utxo")
        if nwu:
            return nwu["vout"][int(raw_in["vout"])]["scriptPubKey"]["hex"].lower()
        raise SecurityError(f"input[{idx}] lacks UTXO data")

    @staticmethod
    def _psbt_input_value(dec_in, idx) -> int:
        wu = dec_in.get("witness_utxo")
        if wu and "value" in wu:
            return int(round(wu["value"] * SAT))
        # non_witness_utxo case: caller must give us prevtx values.
        # In practice we require witness_utxo for P2WSH policies.
        raise SecurityError(
            f"input[{idx}] has no witness value; use verify_raw with prevtxs"
        )

    @staticmethod
    def _worst_case_vsize(vin, vout, segwit: bool) -> int:
        outputs_bytes = 0
        for v in vout:
            n = len(v["scriptPubKey"]["hex"]) // 2
            outputs_bytes += 8 + _varint_len(n) + n

        n_in, n_out = len(vin), len(vout)

        if segwit:
            redeem = 173
            witness_per_input = (
                1 + _varint_len(1)
                + 4 * (1 + _varint_len(73) + 73)
                + 1 + _varint_len(redeem) + redeem
            )
            base = 4 + 4 + _varint_len(n_in) + n_in * 41 + _varint_len(n_out) + outputs_bytes
            weight = base * 4 + 2 + n_in * witness_per_input
            return (weight + 3) // 4
        else:
            script_sig = 1 + 4 * _push_len(73) + _push_len(173)
            input_bytes = 36 + _varint_len(script_sig) + script_sig + 4
            return (
                4 + 4 + _varint_len(n_in) + n_in * input_bytes
                + _varint_len(n_out) + outputs_bytes
            )
```

> Замечание про `_psbt_input_value`: для P2WSH 4-of-5 у нас всегда `witness_utxo`. Для non-witness PSBT (что на Bitcoin-семействе сейчас норма) идём через `verify_raw` с `prevtxs`, который мы всё равно собираем для подписи.

## 2. `adapters/legacy.py` — цепная подпись без потери промежуточных состояний

`signrawtransactionwithwallet` возвращает новую версию hex. Каждый следующий подписант должен получить **предыдущую** версию. Добавляю явное хранение состояния в файл, чтобы это не осталось «в голове у оператора».

```python
from __future__ import annotations
import json
from pathlib import Path

from ..core.amounts import sats_to_btc
from .base import AdapterCapabilityError, BaseAdapter


class _RawSignerMixin:
    """Chain signing for bare and modern legacy adapters."""

    def decode_raw(self, hex_tx: str) -> dict:
        return self.rpc.call("decoderawtransaction", [hex_tx])

    def build_prevtxs(self, hex_tx: str) -> list[dict]:
        """Reconstruct signrawtransactionwithwallet prevtxs from the
        wallet's own view of the UTXOs this tx spends.

        Requires that the watch-only wallet holds all inputs. That is
        the operational invariant: any input not resolvable from the
        wallet is a hard error, not a warning.
        """
        dec = self.decode_raw(hex_tx)
        prevtxs: list[dict] = []
        for i, vin in enumerate(dec["vin"]):
            txid, vout = vin["txid"], int(vin["vout"])
            # gettxout works for unspent; for signing a tx that spends
            # already-seen outputs, the wallet knows them.
            try:
                utxo = self.rpc.call("gettxout", [txid, vout, True])
            except Exception as e:
                raise RuntimeError(
                    f"cannot resolve input[{i}] {txid}:{vout}: {e}"
                ) from e
            if not utxo:
                raise RuntimeError(
                    f"input[{i}] {txid}:{vout} not in UTXO set"
                )
            prevtxs.append({
                "txid": txid,
                "vout": vout,
                "scriptPubKey": utxo["scriptPubKey"]["hex"],
                "amount": utxo["value"],
            })
        return prevtxs

    def sign_step(self, hex_tx: str, prevtxs: list[dict]) -> dict:
        """One signing step. Returns {hex, complete, errors}.

        Caller MUST feed the returned hex to the next signer.
        """
        res = self.rpc.call(
            "signrawtransactionwithwallet", [hex_tx, prevtxs]
        )
        return {
            "hex": res["hex"],
            "complete": bool(res.get("complete")),
            "errors": res.get("errors", []),
        }


class ModernLegacyAdapter(_RawSignerMixin, BaseAdapter):
    name = "modern-legacy"
    required_caps = ("has_descriptors", "has_sortedmulti")
    dust_threshold_sats = 546

    def descriptor_kind(self) -> str: return "sh"

    def address(self) -> str:
        addrs = self.rpc.call("deriveaddresses", [self.descriptor()])
        if len(addrs) != 1:
            raise RuntimeError(f"expected 1 address, got {len(addrs)}")
        return addrs[0]

    def policy_script_pubkey_hex(self) -> str:
        info = self.rpc.call("validateaddress", [self.address()])
        return info["scriptPubKey"].lower()

    def create_unsigned(self, inputs, outputs_sats) -> str:
        outputs = {a: float(sats_to_btc(s)) for a, s in outputs_sats.items()}
        return self.rpc.call("createrawtransaction", [inputs, outputs])

    def fund(self, hex_tx: str) -> tuple[str, int]:
        res = self.rpc.call(
            "fundrawtransaction", [hex_tx, {"includeWatching": True}]
        )
        return res["hex"], res["changepos"]

    def send_raw(self, hex_tx: str) -> str:
        return self.rpc.call("sendrawtransaction", [hex_tx])


class BareMultisigAdapter(_RawSignerMixin, BaseAdapter):
    name = "bare-legacy"
    required_caps = ()
    dust_threshold_sats = 546

    def _require(self, name: str):
        if name not in self.caps["methods"]:
            raise AdapterCapabilityError(
                f"{self.name}: node lacks {name!r} "
                f"(subversion={self.caps.get('subversion')})"
            )

    def _multisig(self) -> dict:
        self._require("createmultisig")
        return self.rpc.call(
            "createmultisig", [4, list(self.policy.pubkeys), "legacy"]
        )

    def descriptor_kind(self) -> str: return "sh"

    def descriptor(self) -> str:
        keys = ",".join(self.policy.pubkeys)
        return f"sh(sortedmulti(4,{keys}))"

    def address(self) -> str:
        return self._multisig()["address"]

    def redeem_script_hex(self) -> str:
        return self._multisig()["redeemScript"]

    def policy_script_pubkey_hex(self) -> str:
        info = self.rpc.call("validateaddress", [self.address()])
        if not info.get("isvalid"):
            raise RuntimeError(f"node refused {self.address()!r}")
        return info["scriptPubKey"].lower()

    def create_unsigned(self, inputs, outputs_sats) -> str:
        outputs = {a: float(sats_to_btc(s)) for a, s in outputs_sats.items()}
        return self.rpc.call("createrawtransaction", [inputs, outputs])

    def fund(self, hex_tx: str) -> tuple[str, int]:
        self._require("fundrawtransaction")
        res = self.rpc.call(
            "fundrawtransaction", [hex_tx, {"includeWatching": True}]
        )
        return res["hex"], res["changepos"]

    def send_raw(self, hex_tx: str) -> str:
        return self.rpc.call("sendrawtransaction", [hex_tx])


# ---- per-network subclasses (dust thresholds) ----
class DogecoinAdapter(BareMultisigAdapter):
    name = "dogecoin"
    dust_threshold_sats = 100_000_000  # 1 DOGE relay minimum

class BitcoinCashAdapter(BareMultisigAdapter):
    name = "bitcoincash"
    dust_threshold_sats = 546

class ZcashTransparentAdapter(BareMultisigAdapter):
    name = "zcash-transparent"
    dust_threshold_sats = 1000
```

```python
# cli.py — добавляю в существующий файл

def cmd_sign_manifest(args, policy):
    """Coordinator-side. Private key comes from env or file, never argv."""
    import os
    from core.manifest import sign_manifest

    priv = os.environ.get("CLANDAI_COORD_PRIVKEY")
    if not priv:
        sys.exit("CLANDAI_COORD_PRIVKEY env not set")

    core = json.loads(Path(args.core).read_text())
    if "outputs" not in core:
        sys.exit("core manifest must contain 'outputs'")

    signed = sign_manifest(core, priv)
    Path(args.out).write_text(json.dumps(signed, indent=2, sort_keys=True))
    print(f"wrote {args.out}")


def cmd_raw_sign_safe(args, policy):
    """Legacy signer. One step in the signing chain.

    Reads the current hex from --hex-in, verifies against the signed
    manifest, signs, writes the new hex to --hex-out. Never signs if
    verification fails.
    """
    a = _build(args.network, args.wallet, policy)
    if not hasattr(a, "sign_step"):
        sys.exit(f"{args.network}: no raw signing support")

    hex_in = Path(args.hex_in).read_text().strip()

    # 1) Verify manifest signature and network/fingerprint binding.
    verifier = ManifestVerifier(_manifest_keys(args, policy))
    expected = verifier.verify_and_load(args.manifest, a)

    # 2) Reconstruct prevtxs from the wallet. Fails hard if any input
    #    is not resolvable — that means either a bug or a tampered tx.
    prevtxs = a.build_prevtxs(hex_in)

    # 3) Security engine checks fees, outputs, change, policy match.
    engine = TransactionSecurityEngine(
        a, max_fee_rate_sats_per_vb=args.max_fee_rate,
    )
    report = engine.verify_raw(hex_in, expected, prevtxs)
    print("[verify] OK:", json.dumps(report, indent=2))

    # 4) Sign one step. Previous signer's hex in, next signer's hex out.
    res = a.sign_step(hex_in, prevtxs)
    Path(args.hex_out).write_text(res["hex"])
    print(json.dumps({
        "hex_out": args.hex_out,
        "complete": res["complete"],
        "errors": res["errors"],
    }, indent=2))
```

## 1. Точная сверка vsize после финализации

Worst-case — это верхняя граница для pre-signature проверок. **После** финализации узел возвращает точный `vsize`. Если расхождение с оценкой больше 5% — это сигнал, что наша модель скрипта/подписи разошлась с реальностью сети. Логируем и не даём отправлять без явного `--force`.

```python
# core/security.py — добавляю метод в TransactionSecurityEngine

from dataclasses import dataclass


@dataclass
class VsizeCheck:
    actual_vsize: int
    worst_case_vsize: int
    delta_pct: float
    ok: bool


class TransactionSecurityEngine:
    # ... существующее ...

    def check_vsize_after_finalize(self, hex_tx: str,
                                   worst_case_vsize: int,
                                   *, tolerance_pct: float = 5.0) -> VsizeCheck:
        """Compare worst-case estimate against the finalized tx.

        Called after finalizepsbt/raw signing. If actual exceeds
        worst-case, our model is broken — refuse to broadcast.
        If actual is far below, our worst-case over-reserves; log it.
        """
        dec = self.adapter.rpc.call("decoderawtransaction", [hex_tx])
        actual = int(dec["vsize"])
        if actual > worst_case_vsize:
            # Underestimate — this is a bug. Do not broadcast.
            raise SecurityError(
                f"actual vsize {actual} exceeds worst-case {worst_case_vsize}; "
                "model is wrong — refusing to proceed"
            )
        delta = (worst_case_vsize - actual) / actual * 100 if actual else 0.0
        return VsizeCheck(
            actual_vsize=actual,
            worst_case_vsize=worst_case_vsize,
            delta_pct=round(delta, 2),
            ok=delta <= tolerance_pct,
        )
```

Подключаю в CLI. В `psbt-finalize` и `raw-sign-safe` в state-файл кладём `vsize_worst_case` из отчёта, чтобы проверка была воспроизводима:

```python
# cli.py — cmd_psbt_finalize

def cmd_psbt_finalize(args, policy):
    a = _build(args.network, args.wallet, policy)
    res = a.finalize_psbt(args.psbt)
    print(json.dumps(res, indent=2))
    if not (res.get("complete") and "hex" in res):
        return

    dec = a.rpc.call("decoderawtransaction", [res["hex"]])
    print(json.dumps({
        "actual_vsize": dec["vsize"],
        "actual_weight": dec["weight"],
    }, indent=2))

    # If caller stored the pre-signature report, compare.
    if args.worst_case_vsize:
        engine = TransactionSecurityEngine(a)
        chk = engine.check_vsize_after_finalize(
            res["hex"], args.worst_case_vsize,
            tolerance_pct=args.vsize_tolerance,
        )
        print(json.dumps({
            "vsize_check": chk.__dict__,
        }, indent=2))
        if not chk.ok:
            print(
                f"[warn] worst-case over-reserves by {chk.delta_pct}% "
                f"(actual={chk.actual_vsize}, wc={chk.worst_case_vsize})",
                file=sys.stderr,
            )
```

В `main()` добавляю флаги:

## 2. `tests/test_script_compat.py` — единый скрипт из двух путей

Ключевой тест. На каждой сети проверяет, что:
- `sortedmulti(4,...)` через `getdescriptorinfo` + `deriveaddresses` даёт **тот же** скрипт, что `createmultisig 4 [...]`;
- `redeemScript` — байт-в-байт одинаков;
- порядок pubkeys в policy не влияет на итог;
- ошибка, если сеть «почти Bitcoin», но скрипт другой.

```python
"""Verify that descriptor-based and RPC-based multisig agree byte-for-byte.

This is the test that catches forks that look like Bitcoin but build a
different script. Skip a network if the node lacks the necessary RPCs,
never silently pass.
"""
import json
import os
import pytest

from core.config import load_node_config
from core.policy import MultisigPolicy
from core.rpc import RPC

from adapters.bitcoin import BitcoinAdapter
from adapters.legacy import (
    ModernLegacyAdapter, BareMultisigAdapter,

    # (name, adapter_cls, descriptor_kind_expected)
    ("bitcoin",     BitcoinAdapter,       "wsh"),
    ("litecoin",    BitcoinAdapter,       "wsh"),
    ("dash",        ModernLegacyAdapter,  "sh"),
    ("qtum",        ModernLegacyAdapter,  "sh"),
    ("ravencoin",   ModernLegacyAdapter,  "sh"),
    ("dogecoin",    BareMultisigAdapter,  None),
    ("bitcoincash", BareMultisigAdapter,  None),
    ("zcash",       BareMultisigAdapter,  None),
]

    "0279be667ef9dcbbac55a06295ce870b07029bfcdb2dce28d959f2815b16f81798",
    "02c6047f9441ed7d6d3045406e95c07cd85c778e4b8cef3ca7abac09b95c709ee5",
    "02f9308a019258c31049344f85f89d5229b531c845836f99b08601f113bce036f9",
    "02e493dbf1c10d80f3581e4904930b1404cc6c13900ee0758474fa94abe8c4cd13",
    "022f8bde4d1a07209355b4a7250a5c5128e88b84bddc619ab7cba8d569b240efe4",

        return False
    try:
        cfg = load_node_config(network)
    except Exception:
        return False
    try:
        RPC(cfg["rpc_url"], cfg["rpc_user"], cfg["rpc_pass"]).call(
            "getnetworkinfo"
        )
        return True
    except Exception:
        return False


@pytest.mark.parametrize("network,cls,kind", NETWORKS)
def test_script_compat(network, cls, kind):
    if not _node_up(network):
        pytest.skip(f"{network} node not reachable")

    cfg = load_node_config(network)
    rpc = RPC(cfg["rpc_url"], cfg["rpc_user"], cfg["rpc_pass"])
    policy = MultisigPolicy(PUBKEYS)
    a = cls(rpc, policy, address_type=cfg.get("address_type", "bech32"))

    # 1) createmultisig — baseline. Available on all Bitcoin-derived Cores.
    if "createmultisig" in a.caps["methods"]:
        addr_type = "bech32" if a.address_type == "bech32" else "legacy"
        cm = rpc.call(
            "createmultisig", [4, list(policy.pubkeys), addr_type]
        )
    else:
        pytest.skip(f"{network}: no createmultisig")

    # 2) descriptor-based path, only where descriptors exist.
    if a.caps["has_descriptors"] and kind:
        desc_raw = policy.descriptor(kind)
        desc = rpc.call("getdescriptorinfo", [desc_raw])["descriptor"]
        addrs = rpc.call("deriveaddresses", [desc])
        assert len(addrs) == 1, f"{network}: deriveaddresses returned {addrs}"
        assert addrs[0] == cm["address"], (
            f"{network}: descriptor address {addrs[0]} != "
            f"createmultisig address {cm['address']}"
        )
    else:
        # Bare path: address comes from createmultisig directly.
        assert a.address() == cm["address"]

    # 3) redeemScript byte-for-byte (P2SH path).
    if "redeemScript" in cm:
        # Recover it via the descriptor path if possible.
        if a.caps["has_descriptors"] and kind == "sh":
            info = rpc.call("validateaddress", [cm["address"]])
            # Some nodes return scriptPubKey hex of the P2SH wrapper;
            # the *inner* redeemScript is what we compare.
            # We rebuild the expected inner script from pubkeys.
            expected_inner = _build_sortedmulti_script(policy.pubkeys)
            assert cm["redeemScript"].lower() == expected_inner.lower(), (
                f"{network}: redeemScript mismatch\n"
                f"  node: {cm['redeemScript']}\n"
                f"  ours: {expected_inner}"
            )


def _build_sortedmulti_script(pubkeys: tuple[str, ...]) -> str:
    """Construct the canonical 4-of-5 sortedmulti script from pubkeys.

    Format: OP_4 <push pk1> ... <push pk5> OP_5 OP_CHECKMULTISIG
    pubkeys MUST already be sorted (MultisigPolicy guarantees this).
    """
    parts = ["54"]  # OP_4
    for pk in pubkeys:
        raw = bytes.fromhex(pk)
        parts.append(f"{len(raw):02x}")  # push length (33 or 65)
        parts.append(pk)
    parts.append("55")  # OP_5
    parts.append("ae")  # OP_CHECKMULTISIG
    return "".join(parts)
```

```python
def test_addresses_differ_across_networks():
    """Same pubkeys, different networks → different addresses.

    If two networks produce the same address, either their address
    encodings collide (bug in our adapter) or we're pointed at the
    same node twice (config bug). Either way — fail.

    seen: dict[str, str] = {}
    for network in sorted(REGISTRY):
        try:
            a = _try_adapter(network)
        except pytest.skip.Exception:
            continue
        addr = a.address()
        for other, other_addr in seen.items():
            # Same address is possible across BTC/LTC testnets etc. in
            # theory but never for mainnet-prefix addresses. Treat as error.
            assert addr != other_addr, (
                f"{network} and {other} produced identical address {addr}"
            )
        seen[network] = addr
    if len(seen) < 2:
        pytest.skip("fewer than 2 networks reachable; nothing to compare")
```

## 5. `core/drift.py` — мониторинг relayfee и dust

При инициализации адаптера снимаем `relayfee` и `mempoolminfee`. Сохраняем в файл `~/.clandai/drift/<network>.json` (chmod 600). Если при следующем запуске значения расходятся > X% — предупреждение, а не молчаливое изменение поведения.

```python
"""Detect silent drift in node policy: relayfee, mempoolminfee, dust.

A node operator can lower or raise these without warning. Our worst-case
fee model assumes a specific dust threshold. If the node's policy has
drifted, we want a warning *before* signing, not a rejected tx.
"""
from __future__ import annotations
import json
import os
import stat
from dataclasses import asdict, dataclass
from pathlib import Path


DRIFT_DIR = Path(os.path.expanduser("~/.clandai/drift"))


@dataclass
class NodePolicySnapshot:
    relay_fee_btc_per_kvb: float
    mempool_min_fee_btc_per_kvb: float
    dust_threshold_sats: int


def _secure_write(path: Path, data: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(data, indent=2, sort_keys=True))
    os.chmod(tmp, stat.S_IRUSR | stat.S_IWUSR)
    tmp.replace(path)


def _read(path: Path) -> dict | None:
    if not path.exists():
        return None
    try:
        return json.loads(path.read_text())
    except Exception:
        return None


def snapshot(adapter) -> NodePolicySnapshot:
    info = adapter.rpc.call("getnetworkinfo")
    mempool = adapter.rpc.call("getmempoolinfo")
    return NodePolicySnapshot(
        relay_fee_btc_per_kvb=float(info.get("relayfee", 0.0)),
        mempool_min_fee_btc_per_kvb=float(mempool.get("mempoolminfee", 0.0)),
        dust_threshold_sats=int(
            getattr(adapter, "dust_threshold_sats", 546)
        ),
    )


def check_and_update(adapter, *, tolerance_pct: float = 20.0) -> list[str]:
    """Return list of human-readable drift warnings (empty if none)."""
    snap = snapshot(adapter)
    path = DRIFT_DIR / f"{adapter.name}.json"
    prev = _read(path)
    warnings: list[str] = []

    if prev is not None:
        for field in ("relay_fee_btc_per_kvb",
                      "mempool_min_fee_btc_per_kvb",
                      "dust_threshold_sats"):
            old = float(prev.get(field, 0))
            new = float(getattr(snap, field))
            if old <= 0:
                continue
            delta = abs(new - old) / old * 100
            if delta > tolerance_pct:
                warnings.append(
                    f"{adapter.name}.{field}: {old} → {new} "
                    f"({delta:.1f}% change)"
                )

    _secure_write(path, asdict(snap))
    return warnings
```

## 1. `core/security.py` — поддержка `non_witness_utxo`

Методы `_psbt_input_spk` и `_psbt_input_value` теперь принимают `raw_in` и читают из `non_witness_utxo`, если `witness_utxo` отсутствует. Call-site в `verify_psbt` обновлён.

```python
# core/security.py — заменяем два метода и call-site в verify_psbt

def verify_psbt(self, psbt_b64, expected):
    # ... до цикла по входам без изменений ...
    input_spks, input_total = [], 0
    for i, (raw_in, dec_in) in enumerate(zip(vin, inputs)):
        spk = self._psbt_input_spk(raw_in, dec_in, i)
        value = self._psbt_input_value(raw_in, dec_in, i)
        input_spks.append(spk)
        input_total += value
    # ... дальше без изменений ...


@staticmethod
def _psbt_input_spk(raw_in, dec_in, idx) -> str:
    wu = dec_in.get("witness_utxo")
    if wu and "scriptPubKey" in wu:
        return wu["scriptPubKey"]["hex"].lower()
    nwu = dec_in.get("non_witness_utxo")
    if nwu:
        prev_vout = int(raw_in["vout"])
        try:
            return nwu["vout"][prev_vout]["scriptPubKey"]["hex"].lower()
        except (KeyError, IndexError) as e:
            raise SecurityError(
                f"input[{idx}] non_witness_utxo missing vout {prev_vout}"
            ) from e
    raise SecurityError(f"input[{idx}] lacks UTXO data")


@staticmethod
def _psbt_input_value(raw_in, dec_in, idx) -> int:
    wu = dec_in.get("witness_utxo")
    if wu and "value" in wu:
        return int(round(wu["value"] * SAT))
    nwu = dec_in.get("non_witness_utxo")
    if nwu:
        prev_vout = int(raw_in["vout"])
        try:
            v = nwu["vout"][prev_vout]
        except (KeyError, IndexError) as e:
            raise SecurityError(
                f"input[{idx}] non_witness_utxo missing vout {prev_vout}"
            ) from e
        if "value" not in v:
            raise SecurityError(
                f"input[{idx}] prev vout has no value"
            )
        return int(round(v["value"] * SAT))
    raise SecurityError(f"input[{idx}] lacks UTXO value")
```

Плюс добавь `segwit=...` детектор, который не полагается на `witness_utxo` только:

```python
# adapters/legacy.py — добавляем в BareMultisigAdapter

class BareMultisigAdapter(_RawSignerMixin, BaseAdapter):
    # ... существующее ...

    def ensure_watch_wallet(self, wallet_name: str) -> "BareMultisigAdapter":
        """P2SH watch-only via importmulti / importaddress.

        No descriptors on the node (Dogecoin 1.14, some BCH builds), so
        we import the redeemScript and the P2SH address by hand.
        """
        self._ensure_wallet_exists(wallet_name)
        scoped = self.with_wallet(wallet_name)

        addr = self.address()
        redeem = self.redeem_script_hex()

        # Preferred: importmulti — one call, both entries, atomic.
        if "importmulti" in self.caps["methods"]:
            scoped.rpc.call("importmulti", [[
                {
                    "scriptPubKey": {"address": addr},
                    "timestamp": "now",
                    "watchonly": True,
                    "label": "clandai-4of5",
                    "redeemscript": redeem,
                },
                {
                    "scriptPubKey": {"hex": redeem},
                    "timestamp": "now",
                    "watchonly": True,
                    "label": "clandai-4of5-redeem",
                },
            ]])
        else:
            # Fallback: importaddress pair. Works on every Bitcoin-derived Core.
            scoped.rpc.call("importaddress", [addr, "clandai-4of5", False])
            scoped.rpc.call("importaddress", [redeem, "", False])

        return scoped
```

```bash
  WIF=$(json_get "json.load(open('$KEYS_FILE'))['signers'][$i]['wif']")
  btc createwallet "$WALLET" >/dev/null 2>&1 || true
  # importdescriptors is the non-deprecated path.
  IMPORT='[{"desc":"combo('"$WIF"')","timestamp":"now","active":false,"internal":false}]'
  btc -rpcwallet="$WALLET" importdescriptors "$IMPORT" >/dev/null
done
log "signer wallets ready"

# --- 3. watch-only wallet ---
"${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
    --wallet clandai_e2e_watch import-watch >/dev/null
log "watch wallet ready"

MS_ADDR=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" create \
          | python3 -c "import sys,json; print(json.load(sys.stdin)['address'])")
FINGERPRINT=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" create \
          | python3 -c "import sys,json; print(json.load(sys.stdin)['fingerprint'])")
log "4-of-5 address: $MS_ADDR (fp=$FINGERPRINT)"

# --- 4. fund if needed ---
COUNT=$(btc -rpcwallet=clandai_e2e_watch listunspent 0 9999999 \
        | python3 -c "import sys,json; print(len(json.load(sys.stdin)))")
if [[ "$COUNT" == "0" ]]; then
  log "funding from $CLANDAI_FAUCET_WALLET"
  btc -rpcwallet="$CLANDAI_FAUCET_WALLET" sendtoaddress "$MS_ADDR" 0.001 >/dev/null
  # Wait for the watch wallet to see the incoming UTXO (mempool is enough).
  for _ in $(seq 1 30); do
    COUNT=$(btc -rpcwallet=clandai_e2e_watch listunspent 0 9999999 \
            | python3 -c "import sys,json; print(len(json.load(sys.stdin)))")
    [[ "$COUNT" != "0" ]] && break
    sleep 2
  done
  [[ "$COUNT" == "0" ]] && { log "funding timeout"; exit 1; }
fi
log "UTXO visible"

# --- 5. build PSBT ---
DEST=$(btc -rpcwallet="$CLANDAI_FAUCET_WALLET" getnewaddress "" bech32)
log "destination: $DEST"

PSBT=$("${CLI[@]}" --network "$NETWORK" --wallet clandai_e2e_watch \
        --pubkeys "$PUBKEYS" \
        psbt-create --to "$DEST:$SEND_SATS" --fee-rate "$FEE_RATE")
log "PSBT: ${PSBT:0:40}..."

# --- 6. sign manifest ---
NONCE=$(python3 -c "import uuid; print(str(uuid.uuid4()))")
NOW=$(python3 -c "import time; print(int(time.time()))")

cat > "$WORKDIR/manifest_core.json" <<EOF
{
  "network": "$NETWORK",
  "fingerprint": "$FINGERPRINT",
  "nonce": "$NONCE",
  "created_at": $NOW,
  "outputs": [{"address": "$DEST", "amount_sats": $SEND_SATS}]
}
EOF

CLANDAI_COORD_PRIVKEY="$COORD_PRIV" \
  "${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
              --manifest-pubkeys "$COORD_PUB" \
              sign-manifest \
              --core "$WORKDIR/manifest_core.json" \
              --out  "$WORKDIR/manifest.json" >/dev/null
log "manifest signed"

# --- 7. sign with 4 signers ---
sign_with() {
  local n="$1" psbt="$2"
  "${CLI[@]}" --network "$NETWORK" --wallet "clandai_e2e_signer$n" \
              --pubkeys "$PUBKEYS" --manifest-pubkeys "$COORD_PUB" \
              psbt-sign-safe --psbt "$psbt" \
                             --manifest "$WORKDIR/manifest.json" \
                             --max-fee-rate 50 \
                             --state-out "$WORKDIR/state.json" \
    | python3 -c "import sys,json; print(json.load(sys.stdin)['psbt'])"
}

PSBT=$(sign_with 1 "$PSBT"); log "signer 1 done"
PSBT=$(sign_with 2 "$PSBT"); log "signer 2 done"
PSBT=$(sign_with 3 "$PSBT"); log "signer 3 done"

# --- 8. NEGATIVE: 3-of-5 must NOT finalize ---
NEG=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
       psbt-finalize --psbt "$PSBT")
NEG_COMPLETE=$(echo "$NEG" | python3 -c "import sys,json; print(json.load(sys.stdin)['complete'])")
if [[ "$NEG_COMPLETE" != "False" ]]; then
  log "FAIL: 3 signatures finalized — negative test broken"
  exit 1
fi
log "negative test OK: 3 signatures insufficient"

# --- 9. signer 4, finalize ---
PSBT=$(sign_with 4 "$PSBT"); log "signer 4 done"

WC_VSIZE=$(python3 -c "import json; print(json.load(open('$WORKDIR/state.json'))['worst_case_vsize'])")

FINAL=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
        psbt-finalize --psbt "$PSBT" \
                      --worst-case-vsize "$WC_VSIZE" \
                      --vsize-tolerance 5)
echo "$FINAL" > "$WORKDIR/final.json"

HEX=$(echo "$FINAL" | python3 -c "import sys,json; d=json.load(sys.stdin); print(d.get('hex',''))")
[[ -z "$HEX" ]] && { log "FAIL: no hex in finalize output"; cat "$WORKDIR/final.json" >&2; exit 1; }
log "finalized, vsize: $(echo "$FINAL" | python3 -c "import sys,json; print(json.load(sys.stdin)['actual_vsize'])")"

# --- 10. mempool accept then broadcast ---
"${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
            testmempoolaccept --hex "$HEX" > "$WORKDIR/mempool.json"
ALLOWED=$(python3 -c "import json; print(json.load(open('$WORKDIR/mempool.json'))[0]['allowed'])")
[[ "$ALLOWED" != "True" ]] && { log "FAIL: rejected by mempool"; cat "$WORKDIR/mempool.json" >&2; exit 1; }
log "mempool accepts"

TXID=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" send --hex "$HEX")
log "broadcast: $TXID"
echo "$TXID" > "$WORKDIR/last_txid.txt"

log "OK — full cycle complete"
```

## 1. `tests/e2e_signet.sh` — параллельная подпись

Заменяю шаги 7–9. Четыре подписанта получают **исходный** PSBT независимо, потом `psbt-combine` собирает все четыре подписи. Это не только быстрее — это правильная модель: подписанты не видят чужих подписей до сборки, у координатора нет промежуточного состояния.

```bash
# --- 7. parallel signing: each signer gets the ORIGINAL PSBT ---
sign_one() {
  local n="$1" input_psbt="$2" out_file="$3"
  "${CLI[@]}" --network "$NETWORK" --wallet "clandai_e2e_signer$n" \
              --pubkeys "$PUBKEYS" --manifest-pubkeys "$COORD_PUB" \
              psbt-sign-safe --psbt "$input_psbt" \
                             --manifest "$WORKDIR/manifest.json" \
                             --max-fee-rate 50 \
                             --state-out "$WORKDIR/state_signer$n.json" \
    > "$out_file"
}

ORIGINAL_PSBT="$PSBT"
for i in 1 2 3 4; do
  sign_one "$i" "$ORIGINAL_PSBT" "$WORKDIR/signed_$i.json" &
done
wait
log "all 4 signers finished (parallel)"

for i in 1 2 3 4; do
  COMPLETE=$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_$i.json'))['complete'])")
  # Each individual signature must NOT complete the tx — that's the point.
  if [[ "$COMPLETE" != "False" ]]; then
    log "FAIL: signer $i alone produced a complete PSBT"
    exit 1
  fi
done
log "each individual signature is incomplete, as expected"

# --- 8. NEGATIVE: any 3-of-4 combined must NOT finalize ---
COMBINE3=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
  psbt-combine \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_1.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_2.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_3.json'))['psbt'])")")

NEG=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
       psbt-finalize --psbt "$COMBINE3")
NEG_COMPLETE=$(echo "$NEG" | python3 -c "import sys,json; print(json.load(sys.stdin)['complete'])")
if [[ "$NEG_COMPLETE" != "False" ]]; then
  log "FAIL: 3 signatures finalized — negative test broken"
  exit 1
fi
log "negative test OK: 3-of-4 insufficient"

# --- 9. combine all 4 and finalize ---
COMBINED=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
  psbt-combine \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_1.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_2.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_3.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_4.json'))['psbt'])")")
log "4 signatures combined"

WC_VSIZE=$(python3 -c "import json; print(json.load(open('$WORKDIR/state_signer1.json'))['worst_case_vsize'])")

FINAL=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
        psbt-finalize --psbt "$COMBINED" \
                      --worst-case-vsize "$WC_VSIZE" \
                      --vsize-tolerance 5)
echo "$FINAL" > "$WORKDIR/final.json"
```

Плюс отдельный кейс — **дублирующиеся подписи одного ключа**:

```bash
# --- 8b. NEGATIVE: same signer twice + two others must NOT finalize ---
COMBINE_DUP=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
  psbt-combine \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_1.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_1.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_2.json'))['psbt'])")" \
    --psbt "$(python3 -c "import json; print(json.load(open('$WORKDIR/signed_3.json'))['psbt'])")")

DUP=$("${CLI[@]}" --network "$NETWORK" --pubkeys "$PUBKEYS" \
       psbt-finalize --psbt "$COMBINE_DUP")
DUP_COMPLETE=$(echo "$DUP" | python3 -c "import sys,json; print(json.load(sys.stdin)['complete'])")
if [[ "$DUP_COMPLETE" != "False" ]]; then
  log "FAIL: duplicate signer counted twice — combinepsbt should dedupe"
  exit 1
fi
log "negative test OK: duplicate signer does not count twice"
```

> Замечание: `combinepsbt` в Bitcoin Core хранит одну подпись на ключ. Дубликат того же PSBT не даёт второй подписи. Тест фиксирует это поведение как контракт — если будущая версия ноды начнёт считать дубликаты, e2e сломается до mainnet.

## 2. Второй негативный сценарий — подмена манифеста после подписи

Три уровня проверки. Каждый ломает подпись или binding.

### 2a. Подмена получателя

```bash
# --- 11. NEGATIVE: recipient swapped after manifest was signed ---
TAMPERED="$WORKDIR/manifest_tampered_recipient.json"
python3 - "$WORKDIR/manifest.json" "$TAMPERED" <<'PY'
import json, sys
m = json.load(open(sys.argv[1]))
# Change only the recipient address; signature bytes untouched.
m["outputs"][0]["address"] = "tb1qw508d6qejxtdg4y5r3zarvary0c5xw7kxpjzsx"
json.dump(m, open(sys.argv[2], "w"), indent=2, sort_keys=True)
PY

set +e
OUT=$("${CLI[@]}" --network "$NETWORK" --wallet clandai_e2e_signer1 \
        --pubkeys "$PUBKEYS" --manifest-pubkeys "$COORD_PUB" \
        psbt-sign-safe --psbt "$ORIGINAL_PSBT" \
                       --manifest "$TAMPERED" \
                       --max-fee-rate 50 2>&1)
RC=$?
set -e
if [[ $RC -eq 0 ]]; then
  log "FAIL: tampered recipient accepted"
  exit 1
fi
echo "$OUT" | grep -q "signature invalid" \
  || { log "FAIL: wrong rejection reason: $OUT"; exit 1; }
log "negative test OK: tampered recipient rejected"
```

### 2b. Подмена fingerprint (другая политика 4-of-5)

```bash
# --- 12. NEGATIVE: fingerprint swap — manifest bound to a different policy ---
TAMPERED_FP="$WORKDIR/manifest_tampered_fp.json"
python3 - "$WORKDIR/manifest.json" "$TAMPERED_FP" <<'PY'

m["fingerprint"] = "0" * 16
json.dump(m, open(sys.argv[2], "w"), indent=2, sort_keys=True)
PY

set +e
OUT=$("${CLI[@]}" --network "$NETWORK" --wallet clandai_e2e_signer1 \
        --pubkeys "$PUBKEYS" --manifest-pubkeys "$COORD_PUB" \
        psbt-sign-safe --psbt "$ORIGINAL_PSBT" \
                       --manifest "$TAMPERED_FP" \
                       --max-fee-rate 50 2>&1)
RC=$?
set -e
[[ $RC -ne 0 ]] || { log "FAIL: fingerprint swap accepted"; exit 1; }
echo "$OUT" | grep -qE "policy mismatch|signature invalid" \
  || { log "FAIL: wrong rejection reason: $OUT"; exit 1; }
log "negative test OK: fingerprint swap rejected"
```

> Проверка `policy mismatch` идёт **до** проверки подписи в `ManifestVerifier.verify_and_load`. Если `fingerprint` не совпал с `adapter.fingerprint()`, движок падает раньше криптографии. Это правильно: не тратим CPU на ECDSA для заведомо чужого манифеста.

### 2c. Cross-network replay (Bitcoin-манифест на Litecoin-адаптере)

Здесь нужен второй узел. Если он недоступен — тест skip, но не молча:

### 2d. Replay того же nonce (защита от повторного использования)

Это не техническая проверка криптографии, это **политика координатора**. `ManifestVerifier` не хранит историю nonce — он stateless. Правильная защита: координатор ведёт журнал nonce и никогда не подписывает два манифеста с одинаковым `nonce` при том же `fingerprint`.

Добавляю в CLI отдельную команду `manifest-log` и жёсткую проверку в `sign-manifest`:

```python
# core/manifest_log.py
"""Coordinator-side nonce ledger.

Single source of truth: a JSONL file where each line is
  {"nonce": ..., "fingerprint": ..., "created_at": ..., "sha256": ...}

sign-manifest refuses to sign a nonce that already appears for the
same fingerprint. This is what stops an accidental double-spend from
the coordinator side.
"""
from __future__ import annotations
import hashlib
import json
import os
import stat
from pathlib import Path


LOG_DIR = Path(os.path.expanduser("~/.clandai/manifests"))


def _path(network: str) -> Path:
    return LOG_DIR / f"{network}.jsonl"


def _canonical_hash(core: dict) -> str:
    from core.manifest import ManifestVerifier
    return hashlib.sha256(
        ManifestVerifier._canonical_core_bytes(
            {**core, "signature": "x", "signer_pubkey": "x"}
        )
    ).hexdigest()


def seen_nonce(network: str, fingerprint: str, nonce: str) -> bool:
    p = _path(network)
    if not p.exists():
        return False
    with p.open() as f:
        for line in f:
            try:
                row = json.loads(line)
            except Exception:
                continue
            if row.get("fingerprint") == fingerprint \
               and row.get("nonce") == nonce:
                return True
    return False


def append(network: str, fingerprint: str, nonce: str, core: dict) -> None:
    LOG_DIR.mkdir(parents=True, exist_ok=True)
    os.chmod(LOG_DIR, stat.S_IRWXU)
    p = _path(network)
    row = {
        "nonce": nonce,
        "fingerprint": fingerprint,
        "created_at": int(core["created_at"]),
        "sha256": _canonical_hash(core),
    }
    with p.open("a") as f:
        f.write(json.dumps(row, sort_keys=True) + "\n")
    os.chmod(p, stat.S_IRUSR | stat.S_IWUSR)
```

Использование в `cmd_sign_manifest`:

```python
# cli.py — заменяем тело cmd_sign_manifest

def cmd_sign_manifest(args, policy):
    import os
    from core.manifest import sign_manifest
    from core.manifest_log import seen_nonce, append

    priv = os.environ.get("CLANDAI_COORD_PRIVKEY")
    if not priv:
        sys.exit("CLANDAI_COORD_PRIVKEY env not set")

    core = json.loads(Path(args.core).read_text())
    if "outputs" not in core:
        sys.exit("core manifest must contain 'outputs'")

    for field in ("network", "fingerprint", "nonce", "created_at"):
        if field not in core:
            sys.exit(f"manifest core missing field {field!r}")

    if seen_nonce(core["network"], core["fingerprint"], core["nonce"]):
        sys.exit(
            f"refusing to sign: nonce {core['nonce']!r} already used "
            f"for fingerprint {core['fingerprint']!r}"
        )

    signed = sign_manifest(core, priv)
    Path(args.out).write_text(json.dumps(signed, indent=2, sort_keys=True))
    append(core["network"], core["fingerprint"], core["nonce"], core)

    print(json.dumps({
        "out": args.out,
        "signer_pubkey": signed["signer_pubkey"],
        "nonce": core["nonce"],
    }))
```

## 3. Обновлённая последовательность e2e

```
 1. generate keys (idempotent, cached)
 2. create signer wallets + import WIFs
 3. create watch-only wallet via import-watch
 4. fund 4-of-5 address from faucet
 5. build PSBT (original)
 6. sign manifest (logged in nonce ledger)
 7. sign with 4 wallets IN PARALLEL
 7b. assert each individual signature is incomplete
 8. NEGATIVE: any 3-of-4 combined → not finalizable
 8b. NEGATIVE: same signer twice + 2 others → not finalizable
 9. combine 4 signed PSBTs → finalize → check_vsize
10. testmempoolaccept → send
11. NEGATIVE: tampered recipient → signature invalid
12. NEGATIVE: tampered fingerprint → policy mismatch
13. NEGATIVE: cross-network replay → cross-network (skip if no LTC node)
14. NEGATIVE: duplicate nonce → coordinator refuses
```

Что делает e2e теперь: 6 позитивных проверок, 5 негативных, всё в одном файле, идемпотентно.

## 4. Итоговая матрица покрытия

| Угроза | Механизм защиты | Где проверено |
|---|---|---|
| 3 подписи вместо 4 | `OP_CHECKMULTISIG` + `finalizepsbt` incomplete | e2e шаг 8 |
| Дублирующийся подписант | `combinepsbt` дедуп по ключу | e2e шаг 8b |
| Подмена получателя после подписи манифеста | ECDSA over canonical core | e2e шаг 11 |
| Подмена `fingerprint` (другая политика) | `policy mismatch` до крипто | e2e шаг 12 |
| Cross-network replay | `network` binding в манифесте | e2e шаг 13 |
| Повторное использование nonce | `manifest_log` + отказ координатора | e2e шаг 14 |
| Подмена сдачи | `_verify_outputs` → change_spk == policy_spk | unit `test_security_raw` + signet |
| Change splitting | `change_count > 1` → reject | unit |
| Dust change | `dust_threshold` per network | unit |

## 1. `core/hwi.py` — обёртка над HWI

HWI (`bitcoin-core/HWI`) — стандартный мост к Ledger/Trezor/Coldcard/bitbox. Работает только с PSBT, поэтому для bare-legacy сетей остаётся только software-подпись (`walletprocesspsbt` / `signrawtransactionwithwallet`). Это ограничение фиксируем явно.

```python
"""Subprocess wrapper around `hwi.py`.

Why subprocess, not python-hid: HWI is the reference implementation,
ships with every hardware vendor's quirks, and is updated independently.
Pin the version and hash in production; verify GPG on install.
"""
from __future__ import annotations

import json
import shutil
import subprocess
from dataclasses import dataclass, field


class HWIError(RuntimeError):
    pass


@dataclass
class HWIDevice:
    model: str
    fingerprint: str
    path: str
    needs_pin: bool = False
    needs_passphrase: bool = False
    raw: dict = field(default_factory=dict)


def _hwi_bin() -> str:
    p = shutil.which("hwi.py") or shutil.which("hwi")
    if not p:
        raise HWIError(
            "hwi.py not found in PATH. "
            "Install: pip install hwi"
        )
    return p


def _run(args: list[str], *, timeout: int = 180) -> str:
    """Run HWI, return stdout. stderr becomes part of the error."""
    try:
        r = subprocess.run(
            [_hwi_bin(), *args],
            capture_output=True, text=True, timeout=timeout, check=False,
        )
    except subprocess.TimeoutExpired as e:
        raise HWIError(f"hwi timeout after {timeout}s: {args}") from e
    if r.returncode != 0:
        raise HWIError(
            f"hwi {' '.join(args)} failed rc={r.returncode}\n"
            f"stdout: {r.stdout.strip()}\n"
            f"stderr: {r.stderr.strip()}"
        )
    return r.stdout.strip()


def enumerate_devices() -> list[HWIDevice]:
    """List connected hardware wallets. Raises if HWI binary missing."""
    out = _run(["enumerate"])
    if not out:
        return []
    try:
        data = json.loads(out)
    except json.JSONDecodeError as e:
        raise HWIError(f"cannot parse hwi enumerate output: {out!r}") from e

    devices: list[HWIDevice] = []
    for d in data:
        devices.append(HWIDevice(
            model=d.get("model", "unknown"),
            fingerprint=d.get("fingerprint", ""),
            path=d.get("path", ""),
            needs_pin=d.get("needs_pin_sent", False),
            needs_passphrase=d.get("needs_passphrase_sent", False),
            raw=d,
        ))
    return devices


def require_device(fingerprint: str) -> HWIDevice:
    """Fail closed: exactly one device with this fingerprint must be present."""
    devs = [d for d in enumerate_devices() if d.fingerprint == fingerprint]
    if not devs:
        raise HWIError(
            f"no HWI device with fingerprint {fingerprint!r} connected"
        )
    if len(devs) > 1:
        raise HWIError(
            f"multiple HWI devices match fingerprint {fingerprint!r}: "
            f"{[d.path for d in devs]}"
        )
    return devs[0]


def sign_psbt(psbt_b64: str, fingerprint: str) -> str:
    """Sign a PSBT on the device. Returns signed PSBT.

    The device is identified by fingerprint — never by index, never by
    path alone. Index shifts between calls; fingerprint does not.
    """
    # Verify the device is actually present before invoking signtx, so
    # the error message is precise ("device missing") rather than a raw
    # HWI stack trace.
    require_device(fingerprint)
    out = _run([
        "-d", fingerprint,
        "signtx", psbt_b64,
    ])
    # HWI signtx prints the signed PSBT on stdout.
    if not out.startswith("cHNidP"):
        raise HWIError(
            f"hwi signtx returned unexpected output: {out[:80]!r}"
        )
    return out
```

## 7. Что дальше — два варианта

### A. Immutable manifest log

Сейчас `manifest_log` — локальный JSONL с `chmod 600`. Для координатора, работающего с реальными деньгами, журнал должен быть append-only и с внешней проверкой целостности. Минимальная реализация:

- каждый ряд — `{nonce, fingerprint, created_at, sha256, prev_hash}`;
- `prev_hash` — хеш предыдущей строки; формируется цепочка, как в блокчейне;
- журнал реплицируется в S3 с object lock (compliance mode), или в IPFS с pinning;
- `manifest-log-verify` — команда, которая проверяет цепочку от начала и падает при первом разрыве.

Это не блокер для mainnet, но это то, что аудитор спросит первым.

### B. HD-деривация через ranged descriptors

Текущая политика — 5 фиксированных pubkeys. Это правильная модель для cold-storage кошелька, но не для «новых адресов на каждый депозит». Ranged descriptor `wsh(sortedmulti(4, [fp/84'/0'/0']xpub1/0/*, ...))` даёт:

- один корневой набор xpub-ов;
- деривация адресов по индексу без новой политики;
- fingerprint политики = хеш xpub-ов, а не 5 конкретных pubkeys.

Правки: `MultisigPolicy` расширяется до `PolicyKind.FIXED` / `PolicyKind.XPUB`, `AddressDerivation` получает `derive_range(start, end)`, `verify_psbt` сверяет входы по префиксу деривации, а не по одному scriptPubKey.


Закрываю четыре пункта из чек-листа. Иду от дешёвых к сложным: flock → Zcash branchid → bare-legacy fee → HD-деривация.

## 1. `core/manifest_log.py` — атомарная блокировка

`seen_nonce` + `append` были двумя отдельными вызовами. Между ними другое соединение могло вклиниться. Заменяю на единый `sign_guarded` с `fcntl.flock(LOCK_EX)` и одновременно проверкой+записью под одной блокировкой.

```python
"""Coordinator-side nonce ledger. Process-safe, thread-safe.

Every mutating operation runs under an exclusive fcntl.flock on the
ledger file. Read+write are atomic with respect to other coordinators
that also use this module.

Why fcntl and not a DB: the ledger must be a file that can be backed
up, hashed, replicated with rsync, and audited by tools that do not
know about our schema. A JSONL with a hash chain gives that for free.
"""
from __future__ import annotations

import fcntl
import hashlib
import json
import os
import stat
from contextlib import contextmanager
from pathlib import Path
from typing import Iterator


LOG_DIR = Path(os.path.expanduser("~/.clandai/manifests"))


class LedgerError(RuntimeError):
    pass


def _path(network: str) -> Path:
    return LOG_DIR / f"{network}.jsonl"


def _canonical_hash(core: dict) -> str:
    from core.manifest import ManifestVerifier
    return hashlib.sha256(
        ManifestVerifier._canonical_core_bytes(
            {**core, "signature": "x", "signer_pubkey": "x"}
        )
    ).hexdigest()


@contextmanager
def _locked(network: str) -> Iterator[Path]:
    """Hold an exclusive flock on the ledger for the whole block."""
    LOG_DIR.mkdir(parents=True, exist_ok=True)
    os.chmod(LOG_DIR, stat.S_IRWXU)
    p = _path(network)
    # Open a+ so the file exists for flock even on first use.
    fd = os.open(p, os.O_RDWR | os.O_CREAT, 0o600)
    try:
        fcntl.flock(fd, fcntl.LOCK_EX)
        yield p
    finally:
        fcntl.flock(fd, fcntl.LOCK_UN)
        os.close(fd)


def _read_chain(p: Path) -> list[dict]:
    if not p.exists():
        return []
    rows: list[dict] = []
    with p.open() as f:
        for lineno, line in enumerate(f, 1):
            line = line.strip()
            if not line:
                continue
            try:
                rows.append(json.loads(line))
            except json.JSONDecodeError as e:
                raise LedgerError(
                    f"{p}: line {lineno} is not valid JSON: {e}"
                ) from e
    return rows


def _last_hash(rows: list[dict]) -> str:
    return rows[-1]["row_hash"] if rows else "0" * 64


def _row_hash(prev_hash: str, body: dict) -> str:
    blob = prev_hash.encode() + b"|" + json.dumps(
        body, sort_keys=True, separators=(",", ":"),
    ).encode()
    return hashlib.sha256(blob).hexdigest()


def reserve_nonce(network: str, fingerprint: str, nonce: str,
                  core: dict) -> dict:
    """Atomically check + reserve a nonce.

    Raises LedgerError if the nonce already exists for this fingerprint.
    Returns the newly appended row on success.

    The flock is held across read+write, so two concurrent coordinators
    cannot both pass the check.
    """
    with _locked(network) as p:
        rows = _read_chain(p)
        for row in rows:
            if (row.get("fingerprint") == fingerprint
                    and row.get("nonce") == nonce):
                raise LedgerError(
                    f"nonce {nonce!r} already used for "
                    f"fingerprint {fingerprint!r}"
                )
        body = {
            "nonce": nonce,
            "fingerprint": fingerprint,
            "created_at": int(core["created_at"]),
            "sha256": _canonical_hash(core),
        }
        prev = _last_hash(rows)
        row = {**body, "prev_hash": prev,
               "row_hash": _row_hash(prev, body)}
        with p.open("a") as f:
            f.write(json.dumps(row, sort_keys=True) + "\n")
            f.flush()
            os.fsync(f.fileno())
        os.chmod(p, stat.S_IRUSR | stat.S_IWUSR)
        return row


def verify_chain(network: str) -> tuple[int, str]:
    """Verify hash chain integrity. Returns (rows, last_hash).

    Any break in the chain raises LedgerError. Safe to call from
    audit tooling without holding the lock — the file is append-only.
    """
    with _locked(network) as p:
        rows = _read_chain(p)
        prev = "0" * 64
        for i, row in enumerate(rows, 1):
            body = {k: row[k] for k in
                    ("nonce", "fingerprint", "created_at", "sha256")}
            expected = _row_hash(prev, body)
            if row.get("prev_hash") != prev:
                raise LedgerError(
                    f"{p}: row {i} prev_hash mismatch "
                    f"(expected {prev}, got {row.get('prev_hash')})"
                )
            if row.get("row_hash") != expected:
                raise LedgerError(
                    f"{p}: row {i} row_hash mismatch"
                )
            prev = row["row_hash"]
        return len(rows), prev
```

`cmd_sign_manifest` теперь вызывает `reserve_nonce` вместо `seen_nonce` + `append`:

```python
# cli.py — заменяем тело cmd_sign_manifest

def cmd_sign_manifest(args, policy):
    import os
    from core.manifest import sign_manifest
    from core.manifest_log import reserve_nonce, LedgerError

    priv = os.environ.get("CLANDAI_COORD_PRIVKEY")
    if not priv:
        sys.exit("CLANDAI_COORD_PRIVKEY env not set")

    core = json.loads(Path(args.core).read_text())
    if "outputs" not in core:
        sys.exit("core manifest must contain 'outputs'")

    for field in ("network", "fingerprint", "nonce", "created_at"):
        if field not in core:
            sys.exit(f"manifest core missing field {field!r}")

    # Reserve BEFORE signing. If reservation fails, no manifest bytes
    # were produced, so an attacker cannot reuse the nonce even if the
    # signing key is online.
    try:
        row = reserve_nonce(
            core["network"], core["fingerprint"], core["nonce"], core,
        )
    except LedgerError as e:
        sys.exit(f"refusing to sign: {e}")

    signed = sign_manifest(core, priv)
    Path(args.out).write_text(json.dumps(signed, indent=2, sort_keys=True))
    print(json.dumps({
        "out": args.out,
        "signer_pubkey": signed["signer_pubkey"],
        "nonce": core["nonce"],
        "row_hash": row["row_hash"],
    }))


def cmd_manifest_verify(args, policy):
```

## 2. Zcash — consensus branch ID

Zcash требует `consensusbranchid` в `signrawtransactionwithwallet` и `sendrawtransaction` при Overwinter+. Без него узел отвергает или подписывает не то. Добавляю в `ZcashTransparentAdapter` получение текущего branch ID и прокидывание в RPC.

```python
# adapters/zcash_transparent.py — полная замена

from __future__ import annotations

from ..core.amounts import sats_to_btc
from .legacy import BareMultisigAdapter


class ZcashTransparentAdapter(BareMultisigAdapter):
    """Zcash transparent (t-address) only.

    Overwinter+ requires consensusbranchid on every raw signing and
    broadcast call. We fetch it from the node at construction and pin
    it for the lifetime of the adapter. If the network upgrades mid-flight,
    a new adapter must be constructed — signing with a stale branchid
    will be rejected by the node.
    """
    name = "zcash-transparent"
    dust_threshold_sats = 1000

    def __init__(self, rpc, policy, *, address_type="legacy"):
        super().__init__(rpc, policy, address_type=address_type)
        self._branchid: int | None = None

    @property
    def branchid(self) -> int:
        if self._branchid is None:
            self._branchid = self._fetch_branchid()
        return self._branchid

    def _fetch_branchid(self) -> int:
        info = self.rpc.call("getblockchaininfo")
        # Upstream zcashd exposes 'consensus' → 'nextblock' or 'chaintip'.
        cons = info.get("consensus") or {}
        for key in ("nextblock", "chaintip"):
            hexid = cons.get(key)
            if hexid:
                return int(hexid, 16) if isinstance(hexid, str) else int(hexid)
        # Pre-NU5 nodes may not have 'consensus'. Fall back to the
        # upgrade field. This path is only for old testnets.
        upgrades = info.get("upgrades") or {}
        for name in ("nu5", "canopy", "heartwood", "blossom", "sapling",
                     "overwinter"):
            up = upgrades.get(name)
            if up and up.get("status") in ("active", "pending"):
                hexid = up.get("branchid")
                if hexid:
                    return int(hexid, 16)
        raise RuntimeError(
            "cannot determine consensus branch id from getblockchaininfo; "
            f"node returned: {info}"
        )

    # ---- override every raw RPC that needs branchid ----

    def sign_step(self, hex_tx: str, prevtxs: list[dict]) -> dict:
        self._require("signrawtransactionwithwallet")
        res = self.rpc.call(
            "signrawtransactionwithwallet",
            [hex_tx, prevtxs, None, self.branchid],
        )
        return {
            "hex": res["hex"],
            "complete": bool(res.get("complete")),
            "errors": res.get("errors", []),
        }

    def send_raw(self, hex_tx: str) -> str:
        return self.rpc.call(
            "sendrawtransaction", [hex_tx, 0, self.branchid],
        )

    def build_prevtxs(self, hex_tx: str) -> list[dict]:
        """Same as parent, but decoderawtransaction on Zcash accepts
        branchid too — we pin it for consistency."""
        dec = self.rpc.call("decoderawtransaction", [hex_tx, self.branchid])
        # Parent logic from here on. Kept inline to make branchid explicit.
        prevtxs: list[dict] = []
        for i, vin in enumerate(dec["vin"]):
            txid, vout = vin["txid"], int(vin["vout"])
            utxo = self.rpc.call("gettxout", [txid, vout, True])
            if not utxo:
                raise RuntimeError(
                    f"input[{i}] {txid}:{vout} not in UTXO set"
                )
            prevtxs.append({
                "txid": txid,
                "vout": vout,
                "scriptPubKey": utxo["scriptPubKey"]["hex"],
                "amount": utxo["value"],
            })
        return prevtxs

    def create_unsigned(self, inputs, outputs_sats) -> str:
        outputs = {a: float(sats_to_btc(s)) for a, s in outputs_sats.items()}
        return self.rpc.call(
            "createrawtransaction", [inputs, outputs, 0, self.branchid],
        )
```

## 3. `BareMultisigAdapter.fund()` — явный fee для 4-of-5

Dogecoin 1.14.x `fundrawtransaction` может занизить комиссию, потому что до подписи не знает размер redeemScript. Решение: считать комиссию сами, используя `_worst_case_vsize`, и передавать `feeRate` в BTC/kvB явно. Дополнительно можно сравнивать результат с `fundrawtransaction` и поднимать ошибку при расхождении.

```python
# adapters/legacy.py — дополняем _RawSignerMixin и BareMultisigAdapter

from ..core.security import _varint_len, _push_len  # переиспользуем


class _RawSignerMixin:
    # ... существующие методы ...

    def estimate_script_sig_bytes(self) -> int:
        """Exact scriptSig length for a fully-signed 4-of-5 P2SH input.

        OP_0 + 4 * push(73) + push(redeemScript)
        redeemScript = OP_4 <pk1>..<pk5> OP_5 OP_CHECKMULTISIG
        """
        keys = self.policy.pubkeys
        # OP_4 (1) + per-key push + OP_5 (1) + OP_CHECKMULTISIG (1)
        redeem = 1 + sum(1 + 33 for _ in keys) + 1 + 1
        return 1 + 4 * _push_len(73) + _push_len(redeem)


class BareMultisigAdapter(_RawSignerMixin, BaseAdapter):
    # ... существующее ...

    def estimate_vsize(self, n_in: int, n_out: int,
                       n_change: int = 1) -> int:
        """Worst-case vsize for a fully-signed 4-of-5 P2SH tx.

        Used to override the node's fee estimator on chains where
        fundrawtransaction mis-sizes multisig inputs.
        """
        script_sig = self.estimate_script_sig_bytes()
        input_bytes = 36 + _varint_len(script_sig) + script_sig + 4
        # Conservative output size: P2PKH/P2SH is 8 + 1 + 25 = 34.
        out_bytes = 34
        base = (
            4 + 4
            + _varint_len(n_in) + n_in * input_bytes
            + _varint_len(n_out + n_change)
            + (n_out + n_change) * out_bytes
        )
        return base

    def fund(self, hex_tx: str, *,
             fee_rate_sats_per_vb: int | None = None) -> tuple[str, int]:
        """Fund with explicit fee, overriding the node's estimator.

        On Dogecoin 1.14.x the node does not know the redeemScript size
        before signing, so its estimate is off. We compute vsize from
        the actual policy and pass an exact feeRate in BTC/kvB.
        """
        self._require("fundrawtransaction")

        opts: dict = {"includeWatching": True}

        if fee_rate_sats_per_vb is not None:
            dec = self.decode_raw(hex_tx)
            n_in = len(dec["vin"])
            n_out = len(dec["vout"])
            # +1 reserved for change; node may or may not add it.
            vsize = self.estimate_vsize(n_in, n_out, n_change=1)
            fee_sats = vsize * fee_rate_sats_per_vb
            fee_btc = float(Decimal(fee_sats) / Decimal(100_000_000))
            opts["feeRate"] = fee_btc  # BTC/kvB — Core convention

        res = self.rpc.call("fundrawtransaction", [hex_tx, opts])
        return res["hex"], res["changepos"]
```

Чтобы не тянуть приватные `_varint_len`/`_push_len` из `security`, дублирую их локально в `legacy.py` (они крошечные, и связность модулей важнее экономии строк):

```python
# adapters/legacy.py — в начале файла

def _varint_len(n: int) -> int:
    if n < 0xFD: return 1
    if n <= 0xFFFF: return 3
    if n <= 0xFFFFFFFF: return 5
    return 9


def _push_len(n: int) -> int:
    if n < 0x4C: return 1 + n
    if n <= 0xFF: return 2 + n
    if n <= 0xFFFF: return 3 + n
    return 5 + n
```

## 4. HD-деривация через ranged descriptors

Самая большая правка. `MultisigPolicy` расширяется до двух режимов, `AddressDerivation` вынесен в отдельный модуль, `verify_psbt` учится проверять входы по префиксу деривации. Совместимость с существующей фиксированной политикой сохранена — это отдельный класс, а не замена.

```python
# core/policy_hd.py

"""HD (xpub-based) 4-of-5 policy.

A ranged descriptor wsh(sortedmulti(4, [fp/84h/0h/0h]xpub1/0/*, ...))
gives us a deterministic family of addresses. The policy fingerprint is
a hash of the 5 xpubs plus the derivation template, NOT of any single
address.

Backward compatible with MultisigPolicy: callers can keep using the
fixed-key policy. This module is a separate class, mixed in only when
the operator provides xpubs.
"""
from __future__ import annotations

import hashlib
import re
from dataclasses import dataclass


XPUB_RE = re.compile(
    r"^(xpub|tpub|ypub|zpub|upub|vpub)[1-9A-HJ-NP-Za-km-z]{100,120}$"
)


class HDPolicyError(Exception):
    pass


@dataclass(frozen=True)
class HDMultisigPolicy:
    """5 xpubs + a derivation template, 4-of-5.

    template is one of:
      - "bip84"  → /0/* (external), m/84'/0'/0'
      - "bip49"  → /0/*, m/49'/0'/0'
      - "bip44"  → /0/*, m/44'/0'/0'
      - custom string, inserted verbatim between xpub and /0/*
    """
    xpubs: tuple[str, ...]
    template: str = "bip84"
    network: str = "bitcoin"

    def __init__(self, xpubs, template="bip84", network="bitcoin"):
        if len(xpubs) != 5:
            raise HDPolicyError(f"need 5 xpubs, got {len(xpubs)}")
        norm = tuple(sorted(self._validate(x) for x in xpubs))
        if len(set(norm)) != 5:
            raise HDPolicyError("duplicate xpubs in policy")
        object.__setattr__(self, "xpubs", norm)
        object.__setattr__(self, "template", template)
        object.__setattr__(self, "network", network)

    @staticmethod
    def _validate(x: str) -> str:
        x = x.strip()
        if not XPUB_RE.match(x):
            raise HDPolicyError(f"not an xpub/tpub: {x[:20]}...")
        return x

    def _derivation_suffix(self) -> str:
        if self.template == "bip84":
            return "/84h/0h/0h"
        if self.template == "bip49":
            return "/49h/0h/0h"
        if self.template == "bip44":
            return "/44h/0h/0h"
        # Custom template must already be a fully-qualified suffix.
        if not self.template.startswith("/"):
            raise HDPolicyError(
                f"custom template must start with '/', got {self.template!r}"
            )
        return self.template

    def descriptor(self, kind: str = "wsh") -> str:
        if kind not in ("wsh", "sh", "sh-wsh"):
            raise HDPolicyError(f"unsupported kind {kind!r}")
        suffix = self._derivation_suffix()
        # NB: no fingerprint prefix — the node resolves xpub to its own
        # fingerprint at import time. We keep the descriptor portable.
        keys = ",".join(f"{x}{suffix}/0/*" for x in self.xpubs)
        inner = f"sortedmulti(4,{keys})"
        if kind == "wsh":
            return f"wsh({inner})"
        if kind == "sh":
            return f"sh({inner})"
        return f"sh(wsh({inner}))"

    def fingerprint(self) -> str:
        h = hashlib.sha256(
            ("|".join(self.xpubs) + "#" + self.template + "#" + self.network)
            .encode()
        ).hexdigest()
        return h[:16]

    @property
    def pubkeys(self):
        """For code that still calls policy.pubkeys — deliberately absent.

        Fixed-key callers must not silently accept an HD policy.
        """
        raise HDPolicyError(
            "HDMultisigPolicy has no fixed pubkeys; "
            "use adapter.address_at(index) instead"
        )
```

```python
# core/derivation.py

"""Address derivation for HD policies. Separate from the adapter so it
can be unit-tested without a node.

Address content is derived locally; the node is only asked to confirm
via deriveaddresses / validateaddress. If the two disagree, we fail.
"""
from __future__ import annotations


class DerivationError(RuntimeError):
    pass


class AddressDerivation:
    def __init__(self, adapter):
        self.adapter = adapter
        self.policy = adapter.policy

    def address_at(self, index: int) -> str:
        if index < 0:
            raise DerivationError(f"index must be >= 0, got {index}")
        desc = self.adapter.descriptor()
        addrs = self.adapter.rpc.call(
            "deriveaddresses", [desc, [index, index]],
        )
        if not addrs:
            raise DerivationError(
                f"node returned no address for index {index}"
            )
        return addrs[0]

    def addresses(self, start: int, end: int) -> list[str]:
        if start < 0 or end < start:
            raise DerivationError(
                f"invalid range [{start}, {end}]"
            )
        if end - start > 1000:
            raise DerivationError(
                "range too large; derive in batches of ≤ 1000"
            )
        desc = self.adapter.descriptor()
        return self.adapter.rpc.call(
            "deriveaddresses", [desc, [start, end]],
        )

    def script_pubkey_at(self, index: int) -> str:
        addr = self.address_at(index)
        info = self.adapter.rpc.call("validateaddress", [addr])
        if not info.get("isvalid"):
            raise DerivationError(f"node rejects derived address {addr}")
        return info["scriptPubKey"].lower()

    def script_pubkey_set(self, start: int, end: int) -> set[str]:
        """All scriptPubKeys in [start, end]. Used by the security engine
        to validate input scripts against the policy family."""
        return {self.script_pubkey_at(i) for i in range(start, end + 1)}
```

`TransactionSecurityEngine` — учится принимать family скриптов вместо одного:

```python
# core/security.py — изменяем конструктор и проверку входов

class TransactionSecurityEngine:
    def __init__(self, adapter, *,
                 max_fee_rate_sats_per_vb: int = 200,
                 max_abs_fee_sats: int = 5_000_000,
                 policy_script_pubkeys: set[str] | None = None):
        self.adapter = adapter
        # Fixed policy: single script. HD policy: caller supplies the set
        # of scripts derived within the wallet's gap limit.
        if policy_script_pubkeys is not None:
            self.policy_spks = {s.lower() for s in policy_script_pubkeys}
        else:
            self.policy_spks = {adapter.policy_script_pubkey_hex().lower()}
        self.policy_spk = next(iter(self.policy_spks))  # for diagnostics
        self.max_fee_rate = max_fee_rate_sats_per_vb
        self.max_abs_fee = max_abs_fee_sats
        self.dust_threshold = self._resolve_dust_threshold(adapter)

    def _check_all_inputs_policy(self, spks):
        for i, spk in enumerate(spks):
            if spk not in self.policy_spks:
                raise SecurityError(
                    f"input[{i}] spends {spk}, not in policy family "
                    f"({len(self.policy_spks)} scripts)"
                )
```

## 1. `core/ledger.py` — абстракция + Postgres

`manifest_log.py` остаётся как `FileLedger`. Добавляется `PostgresLedger` с `UNIQUE(network, fingerprint, nonce)` — это атомарность, которую `flock` дать не может за пределами одной ФС.

```python
"""Nonce ledger abstraction.

Two backends:
  - FileLedger: fcntl.flock + JSONL, single-host, single-FS.
  - PostgresLedger: UNIQUE constraint, multi-host, multi-region.

The interface is deliberately tiny: reserve_nonce() is the only
mutating call. verify() is read-only and safe to call from audit tools.
"""
from __future__ import annotations

import fcntl
import hashlib
import json
import os
import stat
from abc import ABC, abstractmethod
from contextlib import contextmanager
from pathlib import Path
from typing import Iterator


class LedgerError(RuntimeError):
    pass


class Ledger(ABC):
    @abstractmethod
    def reserve_nonce(self, network: str, fingerprint: str,
                      nonce: str, core: dict) -> dict: ...

    @abstractmethod
    def verify(self, network: str) -> tuple[int, str]: ...


# --------------------------------------------------------------- file

LOG_DIR = Path(os.path.expanduser("~/.clandai/manifests"))


def _canonical_hash(core: dict) -> str:
    from core.manifest import ManifestVerifier
    return hashlib.sha256(
        ManifestVerifier._canonical_core_bytes(
            {**core, "signature": "x", "signer_pubkey": "x"}
        )
    ).hexdigest()


def _row_hash(prev: str, body: dict) -> str:
    return hashlib.sha256(
        prev.encode() + b"|" +
        json.dumps(body, sort_keys=True, separators=(",", ":")).encode()
    ).hexdigest()


class FileLedger(Ledger):
    """Single-host append-only JSONL with fcntl.flock and a hash chain."""

    def __init__(self, root: Path | None = None):
        self.root = root or LOG_DIR

    def _path(self, network: str) -> Path:
        return self.root / f"{network}.jsonl"

    @contextmanager
    def _locked(self, network: str) -> Iterator[Path]:
        self.root.mkdir(parents=True, exist_ok=True)
        os.chmod(self.root, stat.S_IRWXU)
        p = self._path(network)
        fd = os.open(p, os.O_RDWR | os.O_CREAT, 0o600)
        try:
            fcntl.flock(fd, fcntl.LOCK_EX)
            yield p
        finally:
            fcntl.flock(fd, fcntl.LOCK_UN)
            os.close(fd)

    def _read(self, p: Path) -> list[dict]:
        if not p.exists():
            return []
        rows = []
        with p.open() as f:
            for i, line in enumerate(f, 1):
                line = line.strip()
                if not line:
                    continue
                try:
                    rows.append(json.loads(line))
                except json.JSONDecodeError as e:
                    raise LedgerError(f"{p}:{i}: {e}") from e
        return rows

    def reserve_nonce(self, network, fingerprint, nonce, core) -> dict:
        with self._locked(network) as p:
            rows = self._read(p)
            for r in rows:
                if (r.get("fingerprint") == fingerprint
                        and r.get("nonce") == nonce):
                    raise LedgerError(
                        f"nonce {nonce!r} already used for {fingerprint!r}"
                    )
            body = {
                "nonce": nonce,
                "fingerprint": fingerprint,
                "created_at": int(core["created_at"]),
                "sha256": _canonical_hash(core),
            }
            prev = rows[-1]["row_hash"] if rows else "0" * 64
            row = {**body, "prev_hash": prev,
                   "row_hash": _row_hash(prev, body)}
            with p.open("a") as f:
                f.write(json.dumps(row, sort_keys=True) + "\n")
                f.flush()
                os.fsync(f.fileno())
            os.chmod(p, stat.S_IRUSR | stat.S_IWUSR)
            return row

    def verify(self, network) -> tuple[int, str]:
        with self._locked(network) as p:
            rows = self._read(p)
            prev = "0" * 64
            for i, row in enumerate(rows, 1):
                body = {k: row[k] for k in
                        ("nonce", "fingerprint", "created_at", "sha256")}
                if row.get("prev_hash") != prev:
                    raise LedgerError(f"{p}: row {i} prev_hash mismatch")
                if row.get("row_hash") != _row_hash(prev, body):
                    raise LedgerError(f"{p}: row {i} row_hash mismatch")
                prev = row["row_hash"]
            return len(rows), prev


# ----------------------------------------------------------- postgres

class PostgresLedger(Ledger):
    """Multi-host coordinator ledger.

    Schema (created on first use, idempotent):

        CREATE TABLE IF NOT EXISTS nonce_ledger (
            network       text NOT NULL,
            fingerprint   text NOT NULL,
            nonce         text NOT NULL,
            created_at    bigint NOT NULL,
            sha256        text NOT NULL,
            row_hash      text NOT NULL,
            inserted_at   timestamptz NOT NULL DEFAULT now(),
            PRIMARY KEY (network, fingerprint, nonce)
        );

    reserve_nonce() uses INSERT ... ON CONFLICT DO NOTHING RETURNING.
    Two concurrent coordinators cannot both pass — the DB serializes.
    """

    SCHEMA = """
    CREATE TABLE IF NOT EXISTS nonce_ledger (
        network       text NOT NULL,
        fingerprint   text NOT NULL,
        nonce         text NOT NULL,
        created_at    bigint NOT NULL,
        sha256        text NOT NULL,
        row_hash      text NOT NULL,
        inserted_at   timestamptz NOT NULL DEFAULT now(),
        PRIMARY KEY (network, fingerprint, nonce)
    );
    """

    def __init__(self, dsn: str):
        # psycopg is imported lazily so non-DB deployments need no driver.
        import psycopg
        self._psycopg = psycopg
        self._dsn = dsn
        with psycopg.connect(dsn, autocommit=True) as conn:
            conn.execute(self.SCHEMA)

    def _last_hash(self, conn, network: str) -> str:
        row = conn.execute(
            "SELECT row_hash FROM nonce_ledger "
            "WHERE network = %s ORDER BY inserted_at DESC LIMIT 1",
            (network,),
        ).fetchone()
        return row[0] if row else "0" * 64

    def reserve_nonce(self, network, fingerprint, nonce, core) -> dict:
        body = {
            "nonce": nonce,
            "fingerprint": fingerprint,
            "created_at": int(core["created_at"]),
            "sha256": _canonical_hash(core),
        }
        with self._psycopg.connect(self._dsn) as conn:
            # SERIALIZABLE ensures two concurrent inserts can't both
            # read the same "last hash" and produce divergent chains.
            conn.execute("SET TRANSACTION ISOLATION LEVEL SERIALIZABLE")
            prev = self._last_hash(conn, network)
            body["row_hash"] = _row_hash(prev, body)
            inserted = conn.execute(
                "INSERT INTO nonce_ledger "
                "(network, fingerprint, nonce, created_at, sha256, row_hash) "
                "VALUES (%s, %s, %s, %s, %s, %s) "
                "ON CONFLICT (network, fingerprint, nonce) DO NOTHING "
                "RETURNING row_hash",
                (network, fingerprint, nonce,
                 body["created_at"], body["sha256"], body["row_hash"]),
            ).fetchone()
            if inserted is None:
                raise LedgerError(
                    f"nonce {nonce!r} already used for {fingerprint!r}"
                )
            conn.commit()
        return body

    def verify(self, network) -> tuple[int, str]:
        with self._psycopg.connect(self._dsn) as conn:
            rows = conn.execute(
                "SELECT nonce, fingerprint, created_at, sha256, row_hash "
                "FROM nonce_ledger WHERE network = %s "
                "ORDER BY inserted_at ASC",
                (network,),
            ).fetchall()
        prev = "0" * 64
        for i, (n, fp, ca, sha, rh) in enumerate(rows, 1):
            body = {"nonce": n, "fingerprint": fp,
                    "created_at": ca, "sha256": sha}
            if rh != _row_hash(prev, body):
                raise LedgerError(f"row {i} hash mismatch")
            prev = rh
        return len(rows), prev


# -------------------------------------------------------------- factory

def get_ledger(backend: str = "file", **kw) -> Ledger:
    if backend == "file":
        return FileLedger(root=kw.get("root"))
    if backend == "postgres":
        dsn = kw.get("dsn") or os.environ.get("CLANDAI_PG_DSN")
        if not dsn:
            raise LedgerError(
                "postgres backend requires --ledger-dsn or CLANDAI_PG_DSN"
            )
        return PostgresLedger(dsn)
    raise LedgerError(f"unknown ledger backend: {backend!r}")
```

## 2. `core/derivation_audit.py` — независимая HD-деривация

Ключевой принцип: адрес, полученный через ноду (`deriveaddresses`), сверяется с адресом, собранным полностью в Python — BIP32 CKDpub, witness script, bech32. Это ловит баги как в адаптере, так и в узле. Использую `ecdsa` только для арифметики точек secp256k1; BIP32, сборку скрипта и bech32 пишу сам.

```python
"""Independent HD address derivation for audit purposes.

If the node and this module disagree on an address, one of them is
wrong, and we refuse to proceed. This is the only way to catch a fork
that "looks like" BIP84 but tweaks one byte.

Only ecdsa is used, and only for secp256k1 point arithmetic.
BIP32, script building, and address encoding are implemented here
so they can be audited line by line.
"""
from __future__ import annotations

import hashlib
import hmac
from dataclasses import dataclass

from ecdsa.curves import SECP256k1
from ecdsa.ellipticcurve import Point

_CURVE = SECP256k1
_G = _CURVE.generator
_N = _CURVE.order
_P = _CURVE.curve.p()

_XPUB_VERSIONS = {b"\x04\x88\xb2\x1e", b"\x04\x35\x87\xcf"}


class AuditError(RuntimeError):
    pass


# --------------------------------------------------------- base58

_B58 = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"


def _b58decode(s: str) -> bytes:
    n = 0
    for c in s:
        try:
            n = n * 58 + _B58.index(c)
        except ValueError as e:
            raise AuditError(f"invalid base58 char {c!r}") from e
    pad = len(s) - len(s.lstrip("1"))
    body = n.to_bytes((n.bit_length() + 7) // 8, "big") if n else b""
    return b"\x00" * pad + body


def _b58check(payload: bytes) -> str:
    checksum = hashlib.sha256(hashlib.sha256(payload).digest()).digest()[:4]
    data = payload + checksum
    n = int.from_bytes(data, "big")
    out = ""
    while n > 0:
        n, r = divmod(n, 58)
        out = _B58[r] + out
    pad = len(data) - len(data.lstrip(b"\x00"))
    return "1" * pad + out


# ------------------------------------------------------- secp256k1

def _ser_p(P: Point) -> bytes:
    return bytes([0x02 if P.y() % 2 == 0 else 0x03]) + \
           P.x().to_bytes(32, "big")


def _parse_p(b: bytes) -> Point:
    if len(b) != 33 or b[0] not in (0x02, 0x03):
        raise AuditError(f"compressed pubkey expected, got {b.hex()[:8]}…")
    x = int.from_bytes(b[1:], "big")
    y_sq = (pow(x, 3, _P) + 7) % _P
    y = pow(y_sq, (_P + 1) // 4, _P)
    if (y & 1) != (b[0] & 1):
        y = _P - y
    if pow(y, 2, _P) != y_sq:
        raise AuditError("invalid pubkey: not on curve")
    return Point(_CURVE.curve, x, y, _N)


# ----------------------------------------------------------- BIP32

def _ckd_pub(parent_pub: bytes, parent_chain: bytes,
             index: int) -> tuple[bytes, bytes]:
    if index >= 0x80000000:
        raise AuditError("cannot derive hardened from xpub")
    data = parent_pub + index.to_bytes(4, "big")
    I = hmac.new(parent_chain, data, hashlib.sha512).digest()
    IL, IR = I[:32], I[32:]
    il_int = int.from_bytes(IL, "big")
    if il_int >= _N:
        raise AuditError("invalid child: IL >= n")
    child = _parse_p(parent_pub) + il_int * _G
    if child == _CURVE.curve.infinity:
        raise AuditError("invalid child: point at infinity")
    return _ser_p(child), IR


def _parse_xpub(xpub: str) -> tuple[bytes, bytes]:
    data = _b58decode(xpub)
    if len(data) != 82:
        raise AuditError(f"xpub length {len(data)} != 82")
    body, checksum = data[:-4], data[-4:]
    if hashlib.sha256(hashlib.sha256(body).digest()).digest()[:4] != checksum:
        raise AuditError("xpub checksum mismatch")
    version = body[:4]
    if version not in _XPUB_VERSIONS:
        raise AuditError(f"unsupported xpub version {version.hex()}")
    chain = body[13:45]
    pubkey = body[45:78]
    return pubkey, chain


def derive_pubkey_at(xpub: str, path: list[int]) -> bytes:
    pub, chain = _parse_xpub(xpub)
    for idx in path:
        pub, chain = _ckd_pub(pub, chain, idx)
    return pub


# ---------------------------------------------------------- bech32

_CHARSET = "qpzry9x8gf2tvdw0s3jn54khce6mua7l"
_GEN = [0x3B6A57B2, 0x26508E6D, 0x1EA119FA, 0x3D4233DD, 0x2A1462B3]


def _polymod(values):
    chk = 1
    for v in values:
        b = chk >> 25
        chk = ((chk & 0x1FFFFFF) << 5) ^ v
        for i in range(5):
            chk ^= _GEN[i] if ((b >> i) & 1) else 0
    return chk


def _hrp_expand(hrp):
    return [ord(c) >> 5 for c in hrp] + [0] + [ord(c) & 31 for c in hrp]


def _create_checksum(hrp, data, spec):
    const = 1 if spec == "bech32" else 0x2BC830A3
    values = _hrp_expand(hrp) + data
    pm = _polymod(values + [0] * 6) ^ const
    return [(pm >> 5 * (5 - i)) & 31 for i in range(6)]


def _convertbits(data, frombits, tobits):
    acc = 0
    bits = 0
    ret = []
    maxv = (1 << tobits) - 1
    for v in data:
        acc = (acc << frombits) | v
        bits += frombits
        while bits >= tobits:
            bits -= tobits
            ret.append((acc >> bits) & maxv)
    if bits:
        ret.append((acc << (tobits - bits)) & maxv)
    return ret


def _bech32_encode(hrp: str, witver: int, program: bytes) -> str:
    spec = "bech32" if witver == 0 else "bech32m"
    data = [witver] + _convertbits(program, 8, 5)
    checksum = _create_checksum(hrp, data, spec)
    return hrp + "1" + "".join(_CHARSET[d] for d in data + checksum)


# --------------------------------------------------------- assembly

_BIP_BASE = {
    "bip84": [84 + 0x80000000, 0x80000000, 0x80000000, 0],
    "bip49": [49 + 0x80000000, 0x80000000, 0x80000000, 0],
    "bip44": [44 + 0x80000000, 0x80000000, 0x80000000, 0],
}

_HRP = {
    "bitcoin": "bc",
    "testnet": "tb",
    "signet":  "tb",
    "regtest": "bcrt",
}

_P2SH_PREFIX = {
    "bitcoin": 0x05,
    "testnet": 0xC4,
    "signet":  0xC4,
    "regtest": 0xC4,
}


def _witness_script(pubkeys_hex: list[str]) -> bytes:
    keys = sorted(bytes.fromhex(pk) for pk in pubkeys_hex)
    if len(keys) != 5:
        raise AuditError(f"need 5 pubkeys, got {len(keys)}")
    parts = [b"\x54"]  # OP_4
    for k in keys:
        if len(k) == 33:
            parts.append(bytes([len(k)]) + k)
        else:
            parts.append(b"\x4c" + bytes([len(k)]) + k)
    parts.append(b"\x55")  # OP_5
    parts.append(b"\xae")  # OP_CHECKMULTISIG
    return b"".join(parts)


@dataclass
class AuditResult:
    address: str
    script_pubkey_hex: str
    witness_script_hex: str


def independent_address(xpubs: list[str], template: str, index: int,
                        kind: str = "wsh",
                        network: str = "bitcoin") -> AuditResult:
    if template not in _BIP_BASE:
        raise AuditError(f"unsupported template {template!r}")
    path = _BIP_BASE[template] + [index]
    pubkeys = [derive_pubkey_at(x, path).hex() for x in xpubs]
    ws = _witness_script(pubkeys)

    if kind == "wsh":
        program = hashlib.sha256(ws).digest()
        addr = _bech32_encode(_HRP[network], 0, program)
        spk = b"\x00\x20" + program
        return AuditResult(addr, spk.hex(), ws.hex())

    if kind == "sh":
        try:
            h160 = hashlib.new("ripemd160",
                               hashlib.sha256(ws).digest()).digest()
        except ValueError as e:
            raise AuditError(
                "ripemd160 unavailable; install pycryptodome or "
                "build Python against OpenSSL with ripemd160"
            ) from e
        addr = _b58check(bytes([_P2SH_PREFIX[network]]) + h160)
        spk = b"\xa9\x14" + h160 + b"\x87"
        return AuditResult(addr, spk.hex(), ws.hex())

    raise AuditError(f"unsupported kind {kind!r}")
```

## 3. `tests/test_derivation_audit.py` — сверка node vs audit

```python
"""Independent derivation must match the node byte-for-byte.
```

## 4. `core/gap_limit.py` — мониторинг и безопасное расширение

Важно: Bitcoin Core **не позволяет** расширить `range` у уже импортированного descriptor — `importdescriptors` вернёт ошибку «descriptor already exists». Правильная стратегия:

- импортировать сразу с запасом (`range: [0, 1000]` по умолчанию);
- мониторить `next_index` из `listdescriptors`;
- при приближении к границе — **алерт**, а не тихое расширение;
- расширение — только через миграцию кошелька (отдельная операция).

```python
"""Gap-limit monitoring.

Bitcoin Core refuses to widen a descriptor's range after import. So
the correct operational model is: import with a large range upfront,
monitor next_index, and alert before the limit is reached. Widening
requires a wallet migration — an operator action, not an automatic one.
"""
from __future__ import annotations

from dataclasses import dataclass


DEFAULT_GAP = 1000
ALERT_THRESHOLD = 0.8


@dataclass
class GapStatus:
    descriptor_range: tuple[int, int]
    next_index: int
    used_fraction: float
    needs_alert: bool


class GapLimitError(RuntimeError):
    pass


def status(wallet_adapter, descriptor: str) -> GapStatus:
    """Read the wallet's current view of our descriptor's range.

    wallet_adapter.rpc must be scoped to /wallet/<name>.
    """
    info = wallet_adapter.rpc.call("listdescriptors", [True])
    for d in info.get("descriptors", []):
        if d.get("desc") == descriptor:
            rng = d.get("range") or [0, 0]
            nxt = int(d.get("next_index", 0))
            size = rng[1] - rng[0] + 1
            used = max(nxt, 1)
            frac = used / size if size else 1.0
            return GapStatus(
                descriptor_range=(rng[0], rng[1]),
                next_index=nxt,
                used_fraction=round(frac, 4),
                needs_alert=frac >= ALERT_THRESHOLD,
            )
    raise GapLimitError(
        f"descriptor not present in wallet: {descriptor[:40]}…"
    )


def require_headroom(wallet_adapter, descriptor: str,
                     *, threshold: float = ALERT_THRESHOLD) -> None:
    """Raise if the wallet is too close to its gap limit.

    Called before creating a new receiving address. If this fires,
    the operator must migrate the wallet to a wider range before new
    deposits are accepted.
    """
    s = status(wallet_adapter, descriptor)
    if s.used_fraction >= threshold:
        raise GapLimitError(
            f"gap limit near exhaustion: {s.next_index} / "
            f"{s.descriptor_range[1]} used "
            f"({s.used_fraction * 100:.1f}%). "
            f"Migrate the wallet to a wider range before accepting "
            f"new deposits."
        )
```

## 2. `core/hsm.py` — низкоуровневая обёртка PKCS#11

Отвечает только за: открыть сессию, найти ключ, подписать 32 байта, вернуть DER-подпись с low-S. Ничего про Bitcoin не знает.

```python
"""PKCS#11 wrapper for secp256k1 ECDSA signing.

Does not know about Bitcoin, PSBT, sighashes. Signs 32-byte digests.

Why low-S normalization here and not in the caller: every Bitcoin
verifier rejects high-S signatures (BIP62). If we forget to normalize
in one caller, the tx is invalid. Doing it once, at the only place
that can produce a signature, removes the class of bug entirely.
"""
from __future__ import annotations

import os
from contextlib import contextmanager
from dataclasses import dataclass
from typing import Iterator

import pkcs11
from pkcs11 import Attribute, KeyType, Mechanism, ObjectClass
from pkcs11.util.ec import encode_named_curve_parameters

# secp256k1 OID from SEC 2: 1.3.132.0.10
_SECP256K1_OID = bytes([0x06, 0x05, 0x2B, 0x81, 0x04, 0x00, 0x0A])

# Order of secp256k1, half-point for low-S normalization.
_N = 0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141
_HALF_N = _N // 2


class HSMError(RuntimeError):
    pass


@dataclass
class HSMKeyRef:
    slot_id: int
    key_label: str
    key_id: bytes | None
    pubkey_compressed: bytes   # 33 bytes


def _load_lib(lib_path: str | None):
    path = lib_path or os.environ.get("CLANDAI_PKCS11_LIB")
    if not path:
        raise HSMError(
            "PKCS#11 library path not set: pass --hsm-lib or "
            "CLANDAI_PKCS11_LIB"
        )
    if not os.path.exists(path):
        raise HSMError(f"PKCS#11 library not found: {path}")
    try:
        return pkcs11.lib(path)
    except pkcs11.exceptions.PKCS11Error as e:
        raise HSMError(f"cannot load {path}: {e}") from e


@contextmanager
def _session(lib, slot_id: int, pin: str) -> Iterator:
    slot = None
    for s in lib.get_slots(token_present=True):
        if s.slot_id == slot_id:
            slot = s
            break
    if slot is None:
        raise HSMError(f"slot {slot_id} not found or no token present")
    token = slot.get_token()
    try:
        session = token.open(user_pin=pin)
    except pkcs11.exceptions.PinIncorrect as e:
        raise HSMError("PIN incorrect") from e
    except pkcs11.exceptions.PinLocked as e:
        raise HSMError("token locked after too many wrong PINs") from e
    try:
        yield session
    finally:
        session.close()


def _read_pin(path: str) -> str:
    """PIN is read from a file with 600 perms, never from argv."""
    st = os.stat(path)
    if st.st_mode & 0o077:
        raise HSMError(
            f"PIN file {path} must be chmod 600 (got {oct(st.st_mode)})"
        )
    with open(path) as f:
        pin = f.read().strip()
    if not pin:
        raise HSMError(f"PIN file {path} is empty")
    return pin


def _compressed_pubkey_from_ec_point(ec_point_der: bytes) -> bytes:
    """Extract 33-byte compressed pubkey from CKA_EC_POINT.

    Per PKCS#11, CKA_EC_POINT is DER OCTET STRING wrapping the SEC1
    encoded point: 04 <x> <y> for uncompressed. Some HSMs return raw
    SEC1 without the OCTET STRING wrapper. Handle both.
    """
    # Strip DER OCTET STRING header if present.
    if ec_point_der[:1] == b"\x04" and ec_point_der[1] == len(ec_point_der) - 2:
        body = ec_point_der[2:]
    else:
        body = ec_point_der
    if len(body) == 65 and body[0] == 0x04:
        x, y = body[1:33], body[33:65]
        prefix = 0x02 if (y[-1] & 1) == 0 else 0x03
        return bytes([prefix]) + x
    if len(body) == 33 and body[0] in (0x02, 0x03):
        return body
    raise HSMError(f"unrecognized EC point encoding: {body[:8].hex()}…")


def enumerate_keys(lib_path: str | None = None,
                   pin_path: str | None = None) -> list[dict]:
    """List slots, tokens, and EC keys. Read-only, no signing."""
    lib = _load_lib(lib_path)
    out: list[dict] = []
    for slot in lib.get_slots(token_present=True):
        token = slot.get_token()
        try:
            info = {
                "slot_id": slot.slot_id,
                "label": token.label,
                "model": token.model,
                "manufacturer": token.manufacturer_id,
                "flags": str(token.flags),
                "keys": [],
            }
        except pkcs11.exceptions.TokenNotPresent:
            continue
        if pin_path:
            pin = _read_pin(pin_path)
            with _session(lib, slot.slot_id, pin) as s:
                for obj in s.get_objects({
                    Attribute.CLASS: ObjectClass.PRIVATE_KEY,
                    Attribute.KEY_TYPE: KeyType.EC,
                }):
                    try:
                        label = obj[Attribute.LABEL]
                    except pkcs11.exceptions.AttributeTypeInvalid:
                        label = ""
                    info["keys"].append({"label": label})
        out.append(info)
    return out


def find_key(lib_path: str | None, slot_id: int, pin_path: str,
             key_label: str) -> HSMKeyRef:
    """Locate a key by label; return its reference and derived pubkey."""
    lib = _load_lib(lib_path)
    pin = _read_pin(pin_path)
    with _session(lib, slot_id, pin) as session:
        candidates = list(session.get_objects({
            Attribute.CLASS: ObjectClass.PRIVATE_KEY,
            Attribute.KEY_TYPE: KeyType.EC,
            Attribute.LABEL: key_label,
        }))
        if not candidates:
            raise HSMError(
                f"no EC private key with label {key_label!r} in slot {slot_id}"
            )
        if len(candidates) > 1:
            raise HSMError(
                f"{len(candidates)} keys with label {key_label!r}; "
                "labels must be unique per slot"
            )
        priv = candidates[0]
        try:
            ec_point = priv[Attribute.EC_POINT]
        except pkcs11.exceptions.AttributeTypeInvalid:
            # Some HSMs hide pubkey on the private object; look up the
            # matching public key by CKA_ID.
            key_id = priv[Attribute.ID]
            pubs = list(session.get_objects({
                Attribute.CLASS: ObjectClass.PUBLIC_KEY,
                Attribute.KEY_TYPE: KeyType.EC,
                Attribute.ID: key_id,
            }))
            if not pubs:
                raise HSMError(
                    f"no public key matches private key {key_label!r}"
                )
            ec_point = pubs[0][Attribute.EC_POINT]
            key_id = priv[Attribute.ID]
        else:
            try:
                key_id = priv[Attribute.ID]
            except pkcs11.exceptions.AttributeTypeInvalid:
                key_id = None

        pub = _compressed_pubkey_from_ec_point(bytes(ec_point))

    return HSMKeyRef(
        slot_id=slot_id,
        key_label=key_label,
        key_id=key_id,
        pubkey_compressed=pub,
    )


def _der_encode(r: int, s: int) -> bytes:
    def _int_bytes(x: int) -> bytes:
        b = x.to_bytes(32, "big").lstrip(b"\x00")
        if not b:
            b = b"\x00"
        if b[0] & 0x80:
            b = b"\x00" + b
        return b
    rb = _int_bytes(r)
    sb = _int_bytes(s)
    body = b"\x02" + bytes([len(rb)]) + rb + b"\x02" + bytes([len(sb)]) + sb
    return b"\x30" + bytes([len(body)]) + body


def sign_sighash(lib_path: str | None, key: HSMKeyRef, pin_path: str,
                 sighash: bytes) -> bytes:
    """Sign a 32-byte sighash. Returns DER-encoded, low-S signature.

    CKM_ECDSA signs a raw digest. Output is 64 bytes R||S big-endian,
    which we DER-encode. Low-S normalization is mandatory for Bitcoin.
    """
    if len(sighash) != 32:
        raise HSMError(f"sighash must be 32 bytes, got {len(sighash)}")

    lib = _load_lib(lib_path)
    pin = _read_pin(pin_path)
    with _session(lib, key.slot_id, pin) as session:
        priv = None
        for obj in session.get_objects({
            Attribute.CLASS: ObjectClass.PRIVATE_KEY,
            Attribute.LABEL: key.key_label,
        }):
            if key.key_id is None or obj[Attribute.ID] == key.key_id:
                priv = obj
                break
        if priv is None:
            raise HSMError(f"key {key.key_label!r} disappeared mid-session")

        # Sign. CKM_ECDSA takes the raw 32-byte digest.
        sig_raw = priv.sign(sighash, mechanism=Mechanism.ECDSA)
        if len(sig_raw) != 64:
            raise HSMError(
                f"HSM returned {len(sig_raw)}-byte signature, expected 64"
            )
        r = int.from_bytes(sig_raw[:32], "big")
        s = int.from_bytes(sig_raw[32:], "big")

        if not (0 < r < _N) or not (0 < s < _N):
            raise HSMError("HSM produced out-of-range signature")
        if s > _HALF_N:
            s = _N - s
        return _der_encode(r, s)
```

## 3. `core/psbt_sighash.py` — BIP143 и legacy sighash

Считает sighash из данных `decodepsbt` плюс witnessScript. Никаких зависимостей кроме `hashlib`.

```python
"""Sighash computation for 4-of-5 multisig PSBTs.

Implements BIP143 (segwit v0) and legacy sighash for SIGHASH_ALL.
Assumes no ANYONECANPAY, no SINGLE — our policies forbid them; if the
PSBT asks for those, we raise, not silently compute wrong bytes.
"""
from __future__ import annotations

import hashlib
import struct

SIGHASH_ALL = 0x01


class SighashError(RuntimeError):
    pass


def _dsha(b: bytes) -> bytes:
    return hashlib.sha256(hashlib.sha256(b).digest()).digest()


def _varint(n: int) -> bytes:
    if n < 0xFD:
        return bytes([n])
    if n <= 0xFFFF:
        return b"\xfd" + struct.pack("<H", n)
    if n <= 0xFFFFFFFF:
        return b"\xfe" + struct.pack("<I", n)
    return b"\xff" + struct.pack("<Q", n)


def _push_data(b: bytes) -> bytes:
    n = len(b)
    if n < 0x4C:
        return bytes([n]) + b
    if n <= 0xFF:
        return b"\x4c" + bytes([n]) + b
    if n <= 0xFFFF:
        return b"\x4d" + struct.pack("<H", n) + b
    return b"\x4e" + struct.pack("<I", n) + b


def _outpoint(txid_hex: str, vout: int) -> bytes:
    # decodepsbt gives txid in display order; wire order is reversed.
    return bytes.fromhex(txid_hex)[::-1] + struct.pack("<I", vout)


def _serialize_output(spk_hex: str, value_sats: int) -> bytes:
    spk = bytes.fromhex(spk_hex)
    return struct.pack("<q", value_sats) + _varint(len(spk)) + spk


def _check_sighash_type(decoded_input: dict) -> None:
    st = decoded_input.get("sighash_type")
    if st is None:
        return  # default SIGHASH_ALL
    # decodepsbt returns sighash_type as int
    if isinstance(st, int) and st != SIGHASH_ALL:
        raise SighashError(
            f"SIGHASH_ALL required by CLANDAI policy; PSBT asks for 0x{st:02x}"
        )


def legacy_sighash(decoded: dict, input_index: int,
                   script_code_hex: str, sighash_type: int = SIGHASH_ALL) -> bytes:
    """Legacy sighash for P2SH inputs (no witness)."""
    _check_sighash_type(decoded["inputs"][input_index])
    tx = decoded["tx"]
    vin = tx["vin"]
    vout = tx["vout"]

    script_code = bytes.fromhex(script_code_hex)

    # Serialize all inputs: for the signing input, scriptCode replaces
    # scriptSig; for others, empty scriptSig.
    parts = [struct.pack("<i", tx["version"])]
    parts.append(_varint(len(vin)))
    for i, v in enumerate(vin):
        parts.append(_outpoint(v["txid"], int(v["vout"])))
        if i == input_index:
            parts.append(_varint(len(script_code)) + script_code)
        else:
            parts.append(b"\x00")  # empty scriptSig
        parts.append(struct.pack("<I", v["sequence"]))
    parts.append(_varint(len(vout)))
    for v in vout:
        parts.append(_serialize_output(
            v["scriptPubKey"]["hex"], int(round(v["value"] * 100_000_000))
        ))
    parts.append(struct.pack("<I", tx["locktime"]))
    parts.append(struct.pack("<I", sighash_type))
    return _dsha(b"".join(parts))


def bip143_sighash(decoded: dict, input_index: int,
                   witness_script_hex: str,
                   amount_sats: int,
                   sighash_type: int = SIGHASH_ALL) -> bytes:
    """BIP143 sighash for P2WSH inputs."""
    _check_sighash_type(decoded["inputs"][input_index])
    tx = decoded["tx"]
    vin = tx["vin"]
    vout = tx["vout"]

    hash_prevouts = _dsha(b"".join(
        _outpoint(v["txid"], int(v["vout"])) for v in vin
    ))
    hash_sequence = _dsha(b"".join(
        struct.pack("<I", v["sequence"]) for v in vin
    ))
    hash_outputs = _dsha(b"".join(
        _serialize_output(v["scriptPubKey"]["hex"],
                          int(round(v["value"] * 100_000_000)))
        for v in vout
    ))

    this = vin[input_index]
    preimage = b"".join([
        struct.pack("<i", tx["version"]),
        hash_prevouts,
        hash_sequence,
        _outpoint(this["txid"], int(this["vout"])),
        bytes.fromhex(witness_script_hex),
        struct.pack("<q", amount_sats),
        struct.pack("<I", this["sequence"]),
        hash_outputs,
        struct.pack("<I", tx["locktime"]),
        struct.pack("<I", sighash_type),
    ])
    return _dsha(preimage)
```

## 4. `core/hsm_signer.py` — glue: HSM + PSBT

Собирает всё: читает PSBT, для каждого входа с нашим ключом считает sighash, подписывает, вставляет partial sig, возвращает новый PSBT.

```python
"""Sign a 4-of-5 PSBT with one HSM key.

Produces a PSBT with exactly one additional partial signature. The
caller is responsible for combining signatures from the four signers
via `combinepsbt` — same as with HWI or software wallets.

Requires `embit` for PSBT serialization. We refuse to hand-roll PSBT
bytes because a single field-length bug silently corrupts the tx.
"""
from __future__ import annotations

from dataclasses import dataclass

from embit.psbt import PSBT
from embit.script import Script

from .hsm import HSMKeyRef, sign_sighash
from .psbt_sighash import bip143_sighash, legacy_sighash, SighashError


class HSMSignerError(RuntimeError):
    pass


@dataclass
class HSMSigningResult:
    psbt_b64: str
    inputs_signed: int
    inputs_total: int


def _lookup_witness_script(decoded_input: dict,
                           witness_script_hex: str) -> bool:
    """Confirm the PSBT's witnessScript field, if present, matches ours.

    If the field is missing, the signer is expected to know it from the
    policy — that is our case, since CLANDAI policy is authoritative.
    """
    ws = decoded_input.get("witness_script")
    if ws is None:
        return True
    return ws.lower() == witness_script_hex.lower()


def sign_psbt_with_hsm(
    adapter,
    psbt_b64: str,
    hsm_key: HSMKeyRef,
    *,
    lib_path: str | None,
    pin_path: str,
    witness_script_hex: str,
    redeem_script_hex: str | None,
) -> HSMSigningResult:
    """Add one partial signature per input owned by hsm_key.

    For P2WSH: sign with BIP143, insert into partial_sigs under the
    input's witnessScript context.
    For P2SH: sign legacy, insert into partial_sigs under redeemScript.

    Inputs already signed by this key are skipped (idempotent).
    """
    # Validate the key is actually part of the policy — refuse to sign
    # with an HSM key that does not belong to this 4-of-5.
    policy_pubkeys = set(adapter.policy.pubkeys)
    pub_hex = hsm_key.pubkey_compressed.hex()
    if pub_hex not in policy_pubkeys:
        raise HSMSignerError(
            f"HSM key {hsm_key.key_label!r} pubkey {pub_hex[:16]}… "
            f"is not in the policy"
        )

    decoded = adapter.rpc.call("decodepsbt", [psbt_b64])
    psbt = PSBT.from_string(psbt_b64)

    inputs_signed = 0
    inputs_total = len(decoded["inputs"])
    segwit = any("witness_utxo" in i for i in decoded["inputs"])

    for i, dec_in in enumerate(decoded["inputs"]):
        # Already signed by this key?
        existing = dec_in.get("partial_signatures") or {}
        if pub_hex in {k.lower() for k in existing.keys()}:
            continue

        if segwit:
            if not _lookup_witness_script(dec_in, witness_script_hex):
                raise HSMSignerError(
                    f"input[{i}] witnessScript mismatch"
                )
            wu = dec_in.get("witness_utxo")
            if not wu:
                raise HSMSignerError(
                    f"input[{i}] P2WSH but no witness_utxo; cannot sign"
                )
            amount_sats = int(round(wu["value"] * 100_000_000))
            sighash = bip143_sighash(
                decoded, i, witness_script_hex, amount_sats,
            )
        else:
            if redeem_script_hex is None:
                raise HSMSignerError(
                    f"input[{i}] legacy P2SH but no redeemScript provided"
                )
            sighash = legacy_sighash(decoded, i, redeem_script_hex)

        sig_der = sign_sighash(lib_path, hsm_key, pin_path, sighash)
        # SIGHASH_ALL byte appended per BIP143 / legacy convention.
        sig_with_type = sig_der + b"\x01"

        # Insert into the PSBT. embit uses pubkey as key for partial sigs.
        pub_bytes = bytes.fromhex(pub_hex)
        psbt.inputs[i].partial_sigs[pub_bytes] = sig_with_type
        inputs_signed += 1

    return HSMSigningResult(
        psbt_b64=psbt.to_string(),
        inputs_signed=inputs_signed,
        inputs_total=inputs_total,
    )
```

## 5. `adapters/bitcoin.py` — метод `sign_psbt_with_hsm`

```python
# adapters/bitcoin.py — добавляем метод

from ..core.hsm_signer import sign_psbt_with_hsm, HSMSignerError


class BitcoinAdapter(BaseAdapter):
    # ... существующее ...

    def sign_psbt_with_hsm(self, psbt_b64: str, hsm_key,
                           *, lib_path, pin_path) -> tuple[str, bool]:
        """Sign via PKCS#11 HSM. Returns (psbt, complete).

        `complete` is checked locally via finalizepsbt dry-run — the
        HSM does not know the multisig threshold.
        """
        witness_script_hex = self._witness_script_hex()
        redeem_script_hex = (
            None if self.descriptor_kind() == "wsh"
            else self._redeem_script_hex()
        )

        try:
            res = sign_psbt_with_hsm(
                self, psbt_b64, hsm_key,
                lib_path=lib_path, pin_path=pin_path,
                witness_script_hex=witness_script_hex,
                redeem_script_hex=redeem_script_hex,
            )
        except HSMSignerError:
            raise

        fin = self.rpc.call("finalizepsbt", [res.psbt_b64, False])
        return res.psbt_b64, bool(fin.get("complete", False))

    # -- helpers --------------------------------------------------------
    def _witness_script_hex(self) -> str:
        """Build the canonical 4-of-5 witnessScript.

        OP_4 <push pk1>…<push pk5> OP_5 OP_CHECKMULTISIG, pubkeys sorted.
        Must match the descriptor the node imported — if it doesn't,
        `validateaddress` on the derived address will disagree.
        """
        parts = ["54"]
        for pk in self.policy.pubkeys:
            raw = bytes.fromhex(pk)
            parts.append(f"{len(raw):02x}")
            parts.append(pk)
        parts += ["55", "ae"]
        return "".join(parts)

    def _redeem_script_hex(self) -> str:
        """Same script; the difference is only the P2SH wrapper."""
        return self._witness_script_hex()
```

```python
def _sign_psbt_dispatch(a, psbt: str, args) -> tuple[str, bool]:
    # Priority: HSM > HWI > software. Mutually exclusive — enforce it.
    modes = sum(bool(x) for x in (
        args.hsm_slot is not None,
        args.hwi or args.hwi_fingerprint,
    ))
    if modes > 1:
        sys.exit("choose exactly one of: HSM, HWI, software")

    if args.hsm_slot is not None:
        if not hasattr(a, "sign_psbt_with_hsm"):
            sys.exit(f"{a.name}: HSM signing not supported")
        key = _resolve_hsm_key(args)
        print(
            f"[hsm] using key {key.key_label!r}, "
            f"pubkey {key.pubkey_compressed.hex()[:16]}…",
            file=sys.stderr,

                lib_path=args.hsm_lib,
                pin_path=args.hsm_pin_file,
            )
        except HSMError as e:
            sys.exit(f"HSM signing failed: {e}")

    if args.hwi or args.hwi_fingerprint:
        # ... существующая ветка без изменений ...

    return a.sign_psbt(psbt)
```

## 7. `tests/test_hsm.py`

Mock PKCS#11 + verify sighash correctness against a known vector.

```python
"""HSM wrapper tests. No hardware required.

Covers:
  - low-S normalization (BIP62)
  - DER encoding (round-trip)
  - pubkey extraction from CKA_EC_POINT
  - refusal to sign with a key not in the policy
  - BIP143 sighash against a hand-computed vector
"""
import hashlib
import pytest

from core.hsm import (
    _der_encode, _compressed_pubkey_from_ec_point,
    _N, _HALF_N, HSMError,
)
from core.psbt_sighash import bip143_sighash, SighashError


# --- DER / low-S ---

def test_der_encode_roundtrip():
    r = 0x01
    s = 0x7F
    der = _der_encode(r, s)
    assert der[0] == 0x30
    # Simple parser, enough for the test.
    body = der[2:]
    assert body[0] == 0x02
    rlen = body[1]
    r_bytes = body[2:2+rlen]
    s_off = 2 + rlen
    assert body[s_off] == 0x02
    slen = body[s_off+1]
    s_bytes = body[s_off+2:s_off+2+slen]
    assert int.from_bytes(r_bytes, "big") == r
    assert int.from_bytes(s_bytes, "big") == s


def test_der_encode_high_bit_padding():
    """Integers with high bit set must be padded with 0x00."""
    der = _der_encode(0x80, 0xFF)
    # r = 0x80 encodes as 02 02 00 80
    assert der[2:6] == b"\x02\x02\x00\x80"


def test_low_s_normalization_boundary():
    """s = half_N + 1 must become N - (half_N + 1) = half_N - 1 (for odd N)."""
    s = _HALF_N + 1
    assert s > _HALF_N
    normalized = _N - s
    assert normalized <= _HALF_N
    assert 0 < normalized < _N


# --- pubkey extraction ---

def test_extract_pubkey_uncompressed():
    # SEC1 uncompressed secp256k1 generator point.
    G = bytes.fromhex(
        "04"
        "79BE667EF9DCBBAC55A06295CE870B07029BFCDB2DCE28D959F2815B16F81798"
        "483ADA7726A3C4655DA4FBFC0E1108A8FD17B448A68554199C47D08FFB10D4B8"
    )
    # Wrapped in DER OCTET STRING.
    der = b"\x04" + bytes([len(G)]) + G
    pub = _compressed_pubkey_from_ec_point(der)
    assert pub.hex() == (
        "0279be667ef9dcbbac55a06295ce870b07029bfcdb2dce28d959f2815b16f81798"
    )


def test_extract_pubkey_already_compressed():
    comp = bytes.fromhex(
        "0279be667ef9dcbbac55a06295ce870b07029bfcdb2dce28d959f2815b16f81798"
    )
    assert _compressed_pubkey_from_ec_point(comp) == comp


def test_extract_pubkey_garbage():
    with pytest.raises(HSMError):
        _compressed_pubkey_from_ec_point(b"\x04\x00\x01")


# --- BIP143 sighash sanity ---

def test_bip143_rejects_non_default_sighash():
    decoded = {
        "tx": {"version": 2, "locktime": 0,
               "vin": [{"txid": "aa" * 32, "vout": 0, "sequence": 0xFFFFFFFF}],
               "vout": [{"scriptPubKey": {"hex": "0014" + "00" * 20},
                         "value": 1.0}]},
        "inputs": [{"sighash_type": 0x83}],  # SINGLE|ANYONECANPAY
    }
    with pytest.raises(SighashError, match="SIGHASH_ALL required"):
        bip143_sighash(decoded, 0, "51", 100_000)


def test_bip143_deterministic():
    """Same inputs → same sighash; change amount → different sighash."""
    ws = "54" + ("21" + "02" + "11" * 32) * 1 + "55ae"
    decoded = {
        "tx": {"version": 2, "locktime": 0,
               "vin": [{"txid": "aa" * 32, "vout": 0, "sequence": 0xFFFFFFFE}],
               "vout": [{"scriptPubKey": {"hex": "0014" + "00" * 20},
                         "value": 0.001}]},
        "inputs": [{}],
    }
    h1 = bip143_sighash(decoded, 0, ws, 100_000)
    h2 = bip143_sighash(decoded, 0, ws, 100_000)
    h3 = bip143_sighash(decoded, 0, ws, 100_001)
    assert h1 == h2
    assert h1 != h3
    assert len(h1) == 32
```

## 9. Что закрыто пунктом 3

| Свойство | Как обеспечено |
|---|---|
| Приватный ключ не покидает HSM | `CKA_EXTRACTABLE=false` (задаётся при генерации); `sign_sighash` только подписывает |
| Low-S normalization | `sign_sighash` принудительно заменяет `s` на `N - s` при `s > N/2` |
| PIN не в argv и не в env | `_read_pin` читает файл, требует `chmod 600` |
| Ключ из policy | `sign_psbt_with_hsm` проверяет `pubkey in policy.pubkeys` до подписи |
| Единственный режим подписи | CLI запрещает одновременные `--hsm-*` и `--hwi-*` |
| SIGHASH_ALL mandatory | `psbt_sighash` отвергает `SIGHASH_SINGLE/NONE/ANYONECANPAY` |
| Идемпотентность | Уже подписанный вход пропускается, не переподписывается |
| Хэш-цепочка nonce ledger | независима от HSM, работает как раньше |
