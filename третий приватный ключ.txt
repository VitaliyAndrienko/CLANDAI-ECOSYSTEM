/* ============ PATTERN LOCK ============ */
function initPatternLock(wrap, onChange, onLive) {
  const svg = wrap.querySelector('.pattern-svg');
  const dots = [...wrap.querySelectorAll('.pattern-dot')];
  const seq = [];
  let drawing = false;

  function drawLines() {
    svg.innerHTML = '';
    if (seq.length < 2) return;
    const accent = getComputedStyle(document.documentElement).getPropertyValue('--neon').trim() || '#00ffaa';
    const wr = wrap.getBoundingClientRect();
    const cs = seq.map(i => {
      const r = dots[i].getBoundingClientRect();
      return { x: r.left + r.width / 2 - wr.left, y: r.top + r.height / 2 - wr.top };
    });
    for (let i = 1; i < cs.length; i++) {
      const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      line.setAttribute('x1', cs[i - 1].x);
      line.setAttribute('y1', cs[i - 1].y);
      line.setAttribute('x2', cs[i].x);
      line.setAttribute('y2', cs[i].y);
      line.setAttribute('stroke', accent);
      line.setAttribute('stroke-width', '3');
      line.setAttribute('stroke-linecap', 'round');
      svg.appendChild(line);
    }
  }
  function findDot(x, y) {
    for (const d of dots) {
      const r = d.getBoundingClientRect();
      const cx = r.left + r.width / 2, cy = r.top + r.height / 2;
      if (Math.hypot(x - cx, y - cy) < r.width * 0.7) return d;
    }
    return null;
  }
  function addDot(d) {
    const i = +d.dataset.i;
    if (seq.includes(i)) return;
    seq.push(i);
    d.classList.add('active');
    drawLines();
    if (onLive) onLive([...seq]);
    if (navigator.vibrate) navigator.vibrate(6);
  }
  function reset() {
    seq.length = 0;
    dots.forEach(d => d.classList.remove('active'));
    svg.innerHTML = '';
    if (onLive) onLive([]);
  }
  wrap.addEventListener('pointerdown', e => {
    if (!wrap.contains(e.target)) return;
    const d = findDot(e.clientX, e.clientY);
    if (d) { drawing = true; addDot(d); wrap.setPointerCapture(e.pointerId); }
  });
  wrap.addEventListener('pointermove', e => {
    if (!drawing) return;
    const d = findDot(e.clientX, e.clientY);
    if (d) addDot(d);
  });
  wrap.addEventListener('pointerup', () => {
    if (!drawing) return;
    drawing = false;
    if (seq.length >= 4 && onChange) onChange([...seq]);
  });
  wrap.addEventListener('pointercancel', () => { drawing = false; });
  return { reset };
}

  let v = '';
  function render(err = false) {
    slots.forEach((s, i) => {
      s.classList.toggle('filled', i < v.length);
      s.classList.toggle('err', err);
    });
  }
  function press(k) {
    if (k === 'del') { v = v.slice(0, -1); render(); return; }
    if (k === 'ok') { if (v.length === length) onComplete(v); return; }
    if (v.length >= length) return;
    v += k; render(); vibrate(5);
    if (v.length === length) setTimeout(() => onComplete(v), 140);
  }
  keypadEl.querySelectorAll('button').forEach(b => {
    b.onclick = () => press(b.dataset.k);
  });
  return {
    reset() { v = ''; render(); },
    setError() { render(true); setTimeout(() => { v = ''; render(false); }, 420); },
    get value() { return v; }
  };
}

function startOpDelete(id) {
  const op = state.ops.find(o => o.id === id);
  if (!op || op.deleting) return;
  op.deleting = true;
  op.deletedAt = Date.now();
  vibrate([10, 30, 10]);
  renderOps();
  toast('Удаление через 60 секунд · можно отменить');
}

function cancelOpDelete(id) {
  const op = state.ops.find(o => o.id === id);
  if (!op) return;
  if (state.opTimers.has(id)) {
    clearInterval(state.opTimers.get(id));
    state.opTimers.delete(id);
  }
  op.deleting = false;
  delete op.deletedAt;
  renderOps();
  toast('✓ Удаление отменено');
}

function startOpCountdown(id, totalMs) {
  if (state.opTimers.has(id)) {
    clearInterval(state.opTimers.get(id));
    state.opTimers.delete(id);
  }
  const item = document.querySelector(`.op-item[data-id="${id}"]`);
  if (!item) return;
  const fillEl = item.querySelector('.op-countdown-fill');
  const secEl = item.querySelector('.op-seconds');
  if (!fillEl || !secEl) return;

  const startTime = Date.now();
  const tick = () => {
    const elapsed = Date.now() - startTime;
    const remaining = Math.max(0, totalMs - elapsed);
    const pct = remaining / totalMs;
    fillEl.style.width = (pct * 100) + '%';
    secEl.textContent = Math.ceil(remaining / 1000);
    if (remaining <= 0) {
      const handle = state.opTimers.get(id);
      if (handle) { clearInterval(handle); state.opTimers.delete(id); }
      finalizeOpDelete(id);
    }
  };
  tick();
  const handle = setInterval(tick, 100);
  state.opTimers.set(id, handle);
}

function finalizeOpDelete(id) {
  const item = document.querySelector(`.op-item[data-id="${id}"]`);
  if (item) item.classList.add('removed');
  setTimeout(() => {
    state.ops = state.ops.filter(o => o.id !== id);
    renderOps();
    if (!state.ops.length) toast('Операции не осталось', true);
  }, 450);
}

function startRemoteTimer() {
  stopRemoteTimer();
  state.remoteSeconds = 300;
  const valueEl = $('#remote-timer-value');
  const barEl = $('#remote-timer-bar');
  const total = 300;

  function update() {
    valueEl.textContent = fmtTime(state.remoteSeconds);

    valueEl.classList.remove('warn', 'danger');
    barEl.classList.remove('warn', 'danger');
    if (state.remoteSeconds <= 60) { valueEl.classList.add('warn'); barEl.classList.add('warn'); }

      stopRemoteTimer();

      $('#btn-confirm-sign').disabled = true;
      $('#btn-confirm-sign').style.opacity = '.4';
      $('#btn-confirm-sign').style.animation = 'none';
    }
  }
  update();
  state.remoteTimerHandle = setInterval(() => {
    state.remoteSeconds--;
    update();
  }, 1000);
}

  if (state.remoteTimerHandle) {
    clearInterval(state.remoteTimerHandle);
    state.remoteTimerHandle = null;
  }
}

/* ============================================================
   MATRIX CANVAS  —  FIXED SIZE
   Используем clientWidth/clientHeight (не зависят от transform),
   чтобы canvas всегда заполнял модалку полностью.
   ============================================================ */
let matrixCtx = null;
let matrixDrops = [];
const matrixChars = '0123456789ABCDEF';
const matrixFontSize = 14;
let matrixCanvas = null;
let matrixResizeHandler = null;

function startCanvasMatrix() {
  matrixCanvas = document.getElementById('matrix-canvas');
  matrixCtx = matrixCanvas.getContext('2d');

  function resize() {
    // clientWidth / clientHeight игнорируют CSS-трансформы (scale)
    // и всегда возвращают реальный layout-размер элемента.
    const w = matrixCanvas.clientWidth || matrixCanvas.parentElement.clientWidth;
    const h = matrixCanvas.clientHeight || matrixCanvas.parentElement.clientHeight;
    if (!w || !h) return false;

    const dpr = window.devicePixelRatio || 1;
    matrixCanvas.width  = Math.floor(w * dpr);
    matrixCanvas.height = Math.floor(h * dpr);

    // Не трогаем inline style.width / style.height —
    // CSS уже задаёт width:100% / height:100% через #matrix-canvas.

    matrixCtx.setTransform(dpr, 0, 0, dpr, 0, 0);

    const cols = Math.ceil(w / matrixFontSize);
    if (matrixDrops.length !== cols) {
      matrixDrops = new Array(cols).fill(1);
    }
    return true;
  }

  // Первый ресайз — с ретраем, если layout ещё не готов
  requestAnimationFrame(() => {
    if (!resize()) requestAnimationFrame(() => { resize(); });
  });

  // Повторный ресайз после завершения входной анимации модалки
  setTimeout(() => { if (state.matrixRunning) resize(); }, 420);

  // Реакция на изменение окна
  if (matrixResizeHandler) window.removeEventListener('resize', matrixResizeHandler);
  matrixResizeHandler = () => { if (state.matrixRunning) resize(); };
  window.addEventListener('resize', matrixResizeHandler);

  const chars = matrixChars.split('');
  const ctx = matrixCtx;

  function draw() {
    if (!state.matrixRunning) return;

    // Используем реальный layout-размер, а не внутренний размер буфера
    const w = matrixCanvas.clientWidth || matrixCanvas.width;
    const h = matrixCanvas.clientHeight || matrixCanvas.height;

    ctx.fillStyle = 'rgba(5,6,8,0.09)';
    ctx.fillRect(0, 0, w, h);
    ctx.font = matrixFontSize + 'px "JetBrains Mono", monospace';
    ctx.textBaseline = 'top';

    for (let i = 0; i < matrixDrops.length; i++) {
      const ch = chars[Math.floor(Math.random() * chars.length)];
      const x = i * matrixFontSize;
      const y = matrixDrops[i] * matrixFontSize;

      const isPink = Math.random() > 0.55;
      if (isPink) {
        ctx.fillStyle = 'rgba(255,43,214,' + (0.5 + Math.random() * 0.5) + ')';
        ctx.shadowColor = '#ff2bd6';
      } else {
        ctx.fillStyle = 'rgba(0,255,170,' + (0.6 + Math.random() * 0.4) + ')';
        ctx.shadowColor = '#00ffaa';
      }
      ctx.shadowBlur = 7;
      ctx.fillText(ch, x, y);
      ctx.shadowBlur = 0;

      if (y > h && Math.random() > 0.975) {
        matrixDrops[i] = 0;
      }
      matrixDrops[i]++;
    }
    state.matrixRAF = requestAnimationFrame(draw);
  }
  draw();
}
