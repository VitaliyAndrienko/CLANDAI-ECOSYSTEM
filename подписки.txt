function startCountdownVisual(restoreBtn, remainingMs, id, type){
  if (!restoreBtn) return;
  const bar = restoreBtn.querySelector('.progress-bar');
  if (!bar) return;

  const startTime = Date.now();
  const totalMs = remainingMs;

  if (state.deleteTimers.has(id)) {
    clearInterval(state.deleteTimers.get(id));
  }

  const tick = () => {
    const elapsed = Date.now() - startTime;
    const remaining = Math.max(0, totalMs - elapsed);
    const progress = remaining / totalMs;
    bar.style.transform = `scaleX(${progress})`;
    if (remaining <= 0) {
      clearInterval(state.deleteTimers.get(id));
      state.deleteTimers.delete(id);
      if (type === 'crypto') {
        state.crypto = state.crypto.filter(x => x.id !== id);
      } else if (type === 'wallet') {
        state.wallet = state.wallet.filter(x => x.id !== id);
      }
      save();
      const card = document.querySelector(`.sub-card[data-id="${id}"]`);
      if (card) {
        card.classList.add('removing');
        setTimeout(() => { renderAll(); }, 500);
      } else {
        renderAll();
      }
    }
  };
  tick();
  const intervalId = setInterval(tick, 100);
  state.deleteTimers.set(id, intervalId);
}

function restoreCrypto(id){

function openDateTimePicker(displayEl, currentValue, onConfirm){
  state.dtTarget = displayEl;
  state.dtValue = currentValue ? new Date(currentValue) : new Date();
  state.dtViewDate = new Date(state.dtValue.getFullYear(), state.dtValue.getMonth(), 1);
  renderDateTimePicker();
  openModal('modal-datetime');
  state._dtConfirm = onConfirm;
  document.querySelectorAll('.datetime-display').forEach(el => el.classList.remove('active'));
  displayEl.classList.add('active');
}

function renderDateTimePicker(){
  const view = state.dtViewDate;
  const selected = state.dtValue;
  const title = view.toLocaleString('ru-RU', { month: 'long', year: 'numeric' });
  $('#dt-title').textContent = title.charAt(0).toUpperCase() + title.slice(1);

  const daysContainer = $('#dt-days');
  daysContainer.innerHTML = '';

  const firstDay = new Date(view.getFullYear(), view.getMonth(), 1);
  const startDayOfWeek = (firstDay.getDay() + 6) % 7;
  const daysInMonth = new Date(view.getFullYear(), view.getMonth() + 1, 0).getDate();
  const prevMonthDays = new Date(view.getFullYear(), view.getMonth(), 0).getDate();

  const today = new Date();
  const todayStr = `${today.getFullYear()}-${today.getMonth()}-${today.getDate()}`;
  const selectedStr = selected ? `${selected.getFullYear()}-${selected.getMonth()}-${selected.getDate()}` : '';

  for (let i = startDayOfWeek - 1; i >= 0; i--) {
    const d = prevMonthDays - i;
    const btn = document.createElement('button');
    btn.className = 'dt-day other';
    btn.textContent = d;
    btn.onclick = () => {
      state.dtViewDate = new Date(view.getFullYear(), view.getMonth() - 1, d);
      state.dtValue.setFullYear(view.getFullYear(), view.getMonth() - 1, d);
      renderDateTimePicker();
    };
    daysContainer.appendChild(btn);
  }

  for (let d = 1; d <= daysInMonth; d++) {
    const btn = document.createElement('button');
    btn.className = 'dt-day';
    const dStr = `${view.getFullYear()}-${view.getMonth()}-${d}`;
    if (dStr === todayStr) btn.classList.add('today');
    if (dStr === selectedStr) btn.classList.add('selected');
    btn.textContent = d;
    btn.onclick = () => {
      state.dtValue.setFullYear(view.getFullYear(), view.getMonth(), d);
      renderDateTimePicker();
    };
    daysContainer.appendChild(btn);
  }

  const totalCells = 42;
  const remaining = totalCells - (startDayOfWeek + daysInMonth);
  for (let d = 1; d <= remaining; d++) {
    const btn = document.createElement('button');
    btn.className = 'dt-day other';
    btn.textContent = d;
    btn.onclick = () => {
      state.dtViewDate = new Date(view.getFullYear(), view.getMonth() + 1, d);
      state.dtValue.setFullYear(view.getFullYear(), view.getMonth() + 1, d);
      renderDateTimePicker();
    };
    daysContainer.appendChild(btn);
  }

  $('#dt-hour').value = pad2(selected.getHours());
  $('#dt-minute').value = pad2(selected.getMinutes());
}

function initSlider(sliderId, fillId, thumbId, valueElId, inputId, min, max, formatter, onInput){
  const slider = $('#' + sliderId);
  const fill = $('#' + fillId);
  const thumb = $('#' + thumbId);
  const valueEl = $('#' + valueElId);
  const input = $('#' + inputId);
  let value = min;
  let dragging = false;
  let dragStartX = 0, dragStartValue = 0;

  function updateDisplay() {
    const trackW = slider.clientWidth;
    const thumbW = thumb.offsetWidth || 16;
    const pad = 6;
    const maxLeft = trackW - thumbW - pad * 2;
    const pct = max === min ? 0 : (value - min) / (max - min);
    const left = pad + pct * maxLeft;
    thumb.style.left = left + 'px';
    fill.style.width = (left + thumbW / 2) + 'px';
    valueEl.textContent = formatter(value);
    slider.classList.toggle('dragged', value > min + (max - min) * 0.005);
    input.value = value.toFixed(2);
  }

  function setValue(v) {
    value = Math.max(min, Math.min(max, v));
    updateDisplay();
    if (onInput) onInput(value);
  }

  slider.addEventListener('pointerdown', e => {
    const rect = slider.getBoundingClientRect();
    const thumbRect = thumb.getBoundingClientRect();
    if (e.clientX >= thumbRect.left && e.clientX <= thumbRect.right) {
      dragging = true;
      dragStartX = e.clientX;
      dragStartValue = value;
    } else {
      const trackW = slider.clientWidth;
      const thumbW = thumb.offsetWidth || 16;
      const pad = 6;
      const maxLeft = trackW - thumbW - pad * 2;
      const relX = e.clientX - rect.left - pad - thumbW / 2;
      const pct = Math.max(0, Math.min(1, relX / maxLeft));
      setValue(min + pct * (max - min));
      dragging = true;
      dragStartX = e.clientX;
      dragStartValue = value;
    }
    slider.classList.add('dragging');
    slider.setPointerCapture(e.pointerId);
    e.preventDefault();
  });
  slider.addEventListener('pointermove', e => {
    if (!dragging) return;
    const trackW = slider.clientWidth;
    const thumbW = thumb.offsetWidth || 16;
    const pad = 6;
    const maxLeft = trackW - thumbW - pad * 2;
    const dx = e.clientX - dragStartX;
    const newLeft = Math.max(0, Math.min(maxLeft, dragStartValue * maxLeft / (max - min) + dx));
    const pct = newLeft / maxLeft;
    setValue(min + pct * (max - min));
  });
  slider.addEventListener('pointerup', () => {
    if (!dragging) return;
    dragging = false;
    slider.classList.remove('dragging');
  });
  slider.addEventListener('pointercancel', () => {
    dragging = false;
    slider.classList.remove('dragging');
  });

  input.addEventListener('input', e => {
    const v = parseFloat(e.target.value);
    if (!isNaN(v)) setValue(v);
  });

  window.addEventListener('resize', updateDisplay);
  requestAnimationFrame(updateDisplay);

  return { setValue, getValue: () => value, update: updateDisplay };
}

   ========================================================= */
function initCustomSelect(selector, onChange) {
  const el = $(selector);
  const btn = el.querySelector('.custom-select-btn');
  const label = el.querySelector('.selected-label');
  const opts = el.querySelectorAll('.custom-select-opt');
  btn.onclick = (e) => { e.stopPropagation(); el.classList.toggle('open'); };
  opts.forEach(opt => {
    opt.onclick = (e) => {
      e.stopPropagation();
      const val = opt.dataset.v;
      label.textContent = opt.textContent;
      opts.forEach(o => o.classList.remove('active'));
      opt.classList.add('active');
      el.classList.remove('open');
      if (onChange) onChange(val, opt.textContent);
    };
  });
  document.addEventListener('click', () => el.classList.remove('open'));
}

/* =========================================================
   TOKEN PICKER
   ========================================================= */
function renderTokenList(filter = '') {
  const listContainer = $('#token-list');
  const q = filter.toLowerCase().trim();
  const filtered = MOCK_TOKENS.filter(t =>
    t.symbol.toLowerCase().includes(q) ||
    t.name.toLowerCase().includes(q) ||
    t.address.toLowerCase().includes(q)
  );
  listContainer.innerHTML = '';

  const selectAllRow = document.createElement('div');
  selectAllRow.className = 'token-picker-item select-all-row';
  if (state.selectedTokens.length === MOCK_TOKENS.length && MOCK_TOKENS.length > 0) {
    selectAllRow.classList.add('selected');
  }
  selectAllRow.innerHTML = `
    <div class="checkbox">
      <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3" stroke-linecap="round" stroke-linejoin="round">
        <polyline points="20 6 9 17 4 12"/>
      </svg>
    </div>
    <div class="info"><div class="sym">ВЫБРАТЬ ВСЕ</div></div>
  `;
  selectAllRow.onclick = (e) => {
    e.stopPropagation();
    if (state.selectedTokens.length === MOCK_TOKENS.length) state.selectedTokens = [];
    else state.selectedTokens = MOCK_TOKENS.map(t => t.symbol);
    updateTokenPickerTrigger();
    renderTokenList($('#token-search').value);
  };
  listContainer.appendChild(selectAllRow);

  if (filtered.length === 0) {
    const empty = document.createElement('div');
    empty.style.padding = '10px';
    empty.style.textAlign = 'center';
    empty.style.color = 'var(--muted)';
    empty.style.fontSize = '12px';
    empty.textContent = 'Ничего не найдено';
    listContainer.appendChild(empty);
    return;
  }

  filtered.forEach(token => {
    const isSelected = state.selectedTokens.includes(token.symbol);
    const item = document.createElement('div');
    item.className = 'token-picker-item' + (isSelected ? ' selected' : '');

    const logo = document.createElement('div');
    logo.className = 'token-logo sm';
    const fb = document.createElement('span');
    fb.className = 'fb';
    fb.textContent = token.symbol.slice(0, 4);
    logo.appendChild(fb);

    const checkbox = document.createElement('div');
    checkbox.className = 'checkbox';
    checkbox.innerHTML = `
      <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3" stroke-linecap="round" stroke-linejoin="round">
        <polyline points="20 6 9 17 4 12"/>
      </svg>
    `;

    const info = document.createElement('div');
    info.className = 'info';
    info.innerHTML = `
      <div class="sym">${token.symbol}</div>
      <div class="name">${token.name}</div>
      <div class="addr">${shortAddr(token.address)}</div>
    `;

    item.appendChild(logo); item.appendChild(checkbox); item.appendChild(info);

    item.onclick = (e) => {
      e.stopPropagation();
      if (state.selectedTokens.includes(token.symbol)) {
        state.selectedTokens = state.selectedTokens.filter(s => s !== token.symbol);
      } else {
        state.selectedTokens.push(token.symbol);
      }
      updateTokenPickerTrigger();
      renderTokenList($('#token-search').value);
    };

    listContainer.appendChild(item);
  });
}

function updateTokenPickerTrigger() {
  const trigger = $('#token-picker-trigger');
  const arrow = trigger.querySelector('.arrow');
  trigger.innerHTML = '';
  if (state.selectedTokens.length === 0) {
    const placeholder = document.createElement('span');
    placeholder.className = 'placeholder';
    placeholder.textContent = 'Выберите токены...';
    trigger.appendChild(placeholder);
  } else {
    state.selectedTokens.forEach(sym => {
      const tag = document.createElement('span');
      tag.className = 'tag';
      tag.innerHTML = `${sym} <span class="remove" data-sym="${sym}">×</span>`;
      tag.querySelector('.remove').onclick = (e) => {
        e.stopPropagation();
        state.selectedTokens = state.selectedTokens.filter(s => s !== sym);
        updateTokenPickerTrigger();
        renderTokenList($('#token-search').value);
      };
      trigger.appendChild(tag);
    });
  }
  trigger.appendChild(arrow);
}

  state.editingCryptoId = item ? item.id : null;
  const isReadOnly = !!item;

  $('#crypto-modal-title').textContent = isReadOnly ? 'ПРОСМОТР ПОДПИСКИ' : 'НОВАЯ КРИПТО-ПОДПИСКА';
  $('#cr-save').textContent = isReadOnly ? 'Закрыть' : 'Сохранить подписку';
  $('#cr-cancel').style.display = isReadOnly ? 'none' : 'block';

  $('#cr-addr').disabled = isReadOnly;
  $('#cr-name').disabled = isReadOnly;
  $('#cr-symbol').disabled = isReadOnly;

  $('#cr-addr').value = item ? item.address : '';
  $('#cr-name').value = item ? (item.name || '') : '';
  $('#cr-symbol').value = item ? (item.symbol || '') : '';

  // Verified — auto-detected, cannot be changed
  const isVerified = item ? !!item.verified : isOfficialAddress($('#cr-addr').value);
  $('#cr-verified').checked = isVerified;
  $('#cr-verified-label').textContent = isVerified
    ? 'Адрес подтверждён как официальный'
    : 'Адрес не входит в реестр официальных';

  // Datetime
  state._crDeadline = item && item.deadline ? item.deadline : (Date.now() + 30*24*60*60*1000);
  const display = $('#cr-deadline-display');
  display.querySelector('.dt-value').textContent = fmtDateShort(state._crDeadline);
  display.classList.remove('empty');
  display.style.pointerEvents = isReadOnly ? 'none' : 'auto';
  display.style.opacity = isReadOnly ? '0.55' : '1';

  openModal('modal-crypto');
}

$('#cr-addr').addEventListener('input', e => {

  if (!isValidAddr(addr) || addr.length < 10) {
    // Reset verified state
    $('#cr-verified').checked = false;
    $('#cr-verified-label').textContent = 'Введите адрес для проверки';
    return;
  }
  if (!$('#cr-name').value.trim()) {
    const svc = autoFillFromAddress(addr);
    $('#cr-name').value = svc.name;
  }
  if (!$('#cr-symbol').value.trim()) {
    const svc = autoFillFromAddress(addr);
    $('#cr-symbol').value = svc.symbol;
  }
  const isVerified = isOfficialAddress(addr);
  $('#cr-verified').checked = isVerified;
  $('#cr-verified-label').textContent = isVerified
    ? 'Адрес подтверждён как официальный'
    : 'Адрес не входит в реестр официальных';
});

$('#cr-save').onclick = () => {
  if (state.editingCryptoId) {
    closeModal('modal-crypto');
    state.editingCryptoId = null;
    return;
  }

  const address = $('#cr-addr').value.trim();
  let name = $('#cr-name').value.trim();
  let symbol = $('#cr-symbol').value.trim().toUpperCase();
  const verified = $('#cr-verified').checked;

  if (!isValidAddr(address)){ toast('Введите корректный адрес', true); return; }

  if (!name || !symbol) {
    const svc = autoFillFromAddress(address);
    if (!name) name = svc.name;
    if (!symbol) symbol = svc.symbol;
  }

  const mockTokens = [
    { symbol: 'USDT', amount: parseFloat((Math.random() * 20 + 1).toFixed(2)), name: 'Tether USD' },
    { symbol: 'ETH', amount: parseFloat((Math.random() * 0.01).toFixed(5)), name: 'Ethereum' }
  ];
  const mockFunctions = ['transferFrom', 'approve', 'withdraw'].slice(0, 2 + Math.floor(Math.random() * 2));

  const payload = {
    address, name,
    symbol: symbol || name.slice(0, 4).toUpperCase(),
    verified,
    subscribedAt: Date.now(),
    deadline: state._crDeadline,
    totalSpent: parseFloat((Math.random() * 100 + 10).toFixed(2)),
    tokens: mockTokens,
    functions: mockFunctions,
    deleted: false
  };

  state.crypto.unshift({ id: uid('cs'), ...payload });
  save(); renderCrypto(); updateCounts();
  closeModal('modal-crypto');
  runSubscriptionAnimation(name, mockTokens, 'крипто-контракт');
};

function openWalletModal(item){
  state.editingWalletId = item ? item.id : null;

  const freqSelect = $('#freq-select');

  wlQtySlider.setValue(qty);

  openModal('modal-wallet');
}

$('#wl-save').onclick = () => {
  const address = $('#wl-addr').value.trim();
  const name = $('#wl-name').value.trim();
  const desc = $('#wl-desc').value.trim();

  if (!isValidAddr(address)){ toast('Введите корректный адрес', true); return; }
  if (state.selectedTokens.length === 0){ toast('Выберите хотя бы один токен', true); return; }

  const amount = wlAmountSlider.getValue();
  const qty = wlQtySlider.getValue();
  if (amount === 0 && qty === 0){
    toast('Укажите сумму или количество', true); return;
  }

  const primaryToken = state.selectedTokens[0];
  const payload = {
    address, name, desc, token: primaryToken,
    frequency: state.customFreq,
    amount: amount > 0 ? amount : null,
    qty: qty > 0 ? qty : null,
    deadline: state._wlDeadline,
    deleted: false
  };

  if (state.editingWalletId){
    const it = state.wallet.find(x => x.id === state.editingWalletId);
    if (it) Object.assign(it, payload);
    toast('✓ Подписка обновлена');
    closeModal('modal-wallet');
    state.editingWalletId = null;
    save(); renderWallet(); updateCounts();
    return;
  }

  state.wallet.unshift({ id: uid('ws'), ...payload });
  save(); renderWallet(); updateCounts();
  closeModal('modal-wallet');

  const animTokens = state.selectedTokens.map(sym => ({
    symbol: sym,
    amount: amount > 0 ? '$' + amount : qty
  }));
  const animName = name || shortAddr(address, 8);
  runSubscriptionAnimation(animName, animTokens, 'адрес кошелька');
};

/* =========================================================
   ANIMATION
   ========================================================= */
function runSubscriptionAnimation(serviceName, tokens, typeLabel) {
  const zone = $('#token-drop-zone');
  zone.innerHTML = '';
  $('#anim-service-name').textContent = serviceName;

  let desc = '';
  if (typeLabel === 'адрес кошелька') {
    desc = `Вы подписали <b>${serviceName}</b> на регулярные отправки.<br>Списание: `;
  } else {
    desc = `Вы подписались на <b>${serviceName}</b>.<br>Списание токенов: `;
  }
  if (tokens && tokens.length > 0) {
    desc += tokens.map(t => `${t.amount || '—'} ${t.symbol}`).join(', ');
  } else {
    desc += 'не указано';
  }
  $('#anim-desc').innerHTML = desc;

  if (tokens && tokens.length > 0) {
    tokens.forEach((t, i) => {
      for (let k = 0; k < 2; k++) {
        setTimeout(() => {
          const el = document.createElement('div');
          el.className = 'falling-token';
          el.style.left = `${15 + Math.random() * 70}%`;
          el.innerHTML = `<span class="fb">${t.symbol}</span>`;
          el.style.animationDuration = `${2 + Math.random() * 0.8}s`;
          zone.appendChild(el);
          setTimeout(() => { if (el.parentNode) el.remove(); }, 3400);
        }, i * 250 + k * 600);
      }
    });
  }

  openModal('modal-animation');

  $('#btn-anim-close').onclick = () => {
    closeModal('modal-animation');
    zone.innerHTML = '';
  };
}
