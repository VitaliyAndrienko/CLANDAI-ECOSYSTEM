/* ================================================================

const COLD_HUE = 184;
const WARM_HUE = 24;
const HUE_PATH = 200;

function accentAt(p, dir){
  let h;
  if (dir === 'cold2warm') h = COLD_HUE + HUE_PATH * p;
  else                     h = WARM_HUE - HUE_PATH * p;
  h = ((h % 360) + 360) % 360;
  return `hsl(${h}, 100%, 55%)`;
}
function startAccentAt(dir){
  return dir === 'cold2warm'
    ? `hsl(${COLD_HUE}, 100%, 55%)`
    : `hsl(${WARM_HUE}, 100%, 55%)`;
}

function updateStats(){
  const count = state.selected.size;
  const sel = TOKENS.filter(t => state.selected.has(t.id));
  const totalValue = sel.reduce((s,t) => s + t.bal * t.price, 0);
  const sentValue = totalValue * state.pct / 100;

  const isCold = state.direction === 'cold2warm';
  $('#stat-count-wrap').className = 'stat ' + (isCold ? 'cold' : 'hot');

function applySlider(){
  const track = $('#slider-track');
  const thumb = $('#slider-thumb');
  if (!track) return;
  const trackW = track.clientWidth;
  const thumbW = thumb.offsetWidth;
  const pad = 5;
  const maxLeft = trackW - thumbW - pad * 2;
  const left = pad + (state.pct / 100) * maxLeft;

  thumb.style.left = left + 'px';
  $('#slider-fill').style.width = (left + thumbW) + 'px';
  thumb.textContent = Math.round(state.pct) + '%';
  track.classList.toggle('dragged', state.pct > 3);

  updateStats();
}

$('#slider-track').addEventListener('pointerdown', e => {
  if (e.button !== undefined && e.button !== 0) return;
  const track = $('#slider-track');
  const thumb = $('#slider-thumb');
  const rect = track.getBoundingClientRect();
  const thumbRect = thumb.getBoundingClientRect();

  state.dragging = true;
  thumb.classList.add('dragging');

  if (e.clientX < thumbRect.left || e.clientX > thumbRect.right){
    const trackW = track.clientWidth;
    const thumbW = thumb.offsetWidth;
    const pad = 5;
    const maxLeft = trackW - thumbW - pad * 2;
    const relX = e.clientX - rect.left - pad - thumbW / 2;
    state.pct = Math.max(0, Math.min(100, (relX / maxLeft) * 100));
    applySlider();
  }
  state.dragStartX = e.clientX;
  state.dragStartPct = state.pct;

  e.preventDefault();
  window.addEventListener('pointermove', onSliderMove);
  window.addEventListener('pointerup', onSliderUp);
  window.addEventListener('pointercancel', onSliderUp);
  if (navigator.vibrate) navigator.vibrate(10);
});

function onSliderMove(e){
  if (!state.dragging) return;
  const track = $('#slider-track');
  const thumb = $('#slider-thumb');
  const trackW = track.clientWidth;
  const thumbW = thumb.offsetWidth;
  const pad = 5;
  const maxLeft = trackW - thumbW - pad * 2;
  const dx = e.clientX - state.dragStartX;
  const deltaPct = (dx / maxLeft) * 100;
  state.pct = Math.max(0, Math.min(100, state.dragStartPct + deltaPct));
  applySlider();

  const p = Math.floor(state.pct / 5) * 5;
  if (p !== state._lastTick){
    state._lastTick = p;
    if (navigator.vibrate) navigator.vibrate(3);
  }
}

function onSliderUp(){
  if (!state.dragging) return;
  state.dragging = false;
  $('#slider-thumb').classList.remove('dragging');
  window.removeEventListener('pointermove', onSliderMove);
  window.removeEventListener('pointerup', onSliderUp);
  window.removeEventListener('pointercancel', onSliderUp);

  if (state.pct < 4) state.pct = 0;
  else if (state.pct > 96) state.pct = 100;
  applySlider();
}

/* ================================================================

   ================================================================ */
function buildTransferModal(){
  const dir = state.direction;
  const isCold2Warm = dir === 'cold2warm';

  const leftKind  = isCold2Warm ? 'snow' : 'fire';
  const rightKind = isCold2Warm ? 'fire' : 'snow';
  const leftNode  = $('#node-left');
  const rightNode = $('#node-right');
  leftNode.className  = 'node ' + (isCold2Warm ? 'cold' : 'warm');
  rightNode.className = 'node ' + (isCold2Warm ? 'warm' : 'cold');
  setIcon(leftNode, leftKind);
  setIcon(rightNode, rightKind);

  const rail = $('#rail');
  rail.innerHTML = '';

  const sel = TOKENS.filter(t => state.selected.has(t.id));
  const DURATION = 1.35;
  const STAGGER  = 0.32;

  sel.forEach((tok, i) => {
    const coin = document.createElement('div');
    coin.className = 'tcoin ' + (isCold2Warm ? 'cold' : 'warm');
    coin.innerHTML = `
      <span class="fb">${tok.sym}</span>
      <img src="${tokenLogo(tok)}" alt="${tok.sym}"
           onload="this.previousElementSibling.style.display='none'"
           onerror="this.remove()"/>`;
    const delay = i * STAGGER;
    const animName = isCold2Warm ? 'rollRight' : 'rollLeft';
    coin.style.animation = `${animName} ${DURATION}s cubic-bezier(.45,0,.55,1) ${delay}s forwards`;
    coin.style.opacity = '0';
    rail.appendChild(coin);
  });

  const totalMs = (DURATION + STAGGER * Math.max(0, sel.length - 1)) * 1000;

  state._pending = {
    sel, pct: state.pct / 100,
    totalValue: sel.reduce((s,t) => s + t.bal * t.price, 0) * state.pct / 100
  };

  $$('#modal-transfer .t-view').forEach(v => v.classList.remove('active'));
  $('#t-view-sending').classList.add('active');

  clearTimeout(state._txTimer);
  // Auto-advance to success view after animation completes
  state._txTimer = setTimeout(() => {
    if (!$('#modal-transfer').classList.contains('show')) return;
    if (!$('#t-view-sending').classList.contains('active')) return;
    if (navigator.vibrate) navigator.vibrate([20,40,20]);
    showSuccessView();
  }, totalMs + 250);
}

function showSuccessView(){
  const {sel, pct} = state._pending || {};
  if (!sel || !pct) return;

  const isCold2Warm = state.direction === 'cold2warm';

  // apply balance changes
  sel.forEach(t => {
    const moved = t.bal * pct;
    if (isCold2Warm) t.bal = Math.max(0, t.bal - moved);
    else t.bal += moved;
  });

  const dirLabel = isCold2Warm ? '❄ COLD → 🔥 WARM' : '🔥 WARM → ❄ COLD';
  const usdValue = state._pending.totalValue;

  $('#sum-dir').textContent = dirLabel;
  $('#sum-count').textContent = sel.length + ' шт.';
  $('#sum-amount').textContent = sel.length === 1
    ? num(sel[0].bal * pct) + ' ' + sel[0].sym
    : sel.length + ' видов';
  $('#sum-usd').textContent = money(usdValue);

  $$('#modal-transfer .t-view').forEach(v => v.classList.remove('active'));
  $('#t-view-success').classList.add('active');
  if (navigator.vibrate) navigator.vibrate([10,30,10]);

  renderTokenList();
  resetSlider();
  toast('✓ Перевод выполнен');
}
