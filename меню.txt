<script>
/* ==========================================================
   CLANDAI · MENU MODAL (demo)
   ========================================================== */
const state = {
  mode: 'warm',           // 'cold' | 'warm'
  quantum: 'quantum',     // 'classical' | 'quantum'
  quantumCredits: 60,
  whitelist: [],
  blocklist: [],
  sessions: 1,
};

const $ = s => document.querySelector(s);

function toast(msg){
  const el = $('#toast');
  el.textContent = msg;
  el.classList.add('show');
  clearTimeout(el._h);
  el._h = setTimeout(() => el.classList.remove('show'), 2200);
}
function vibrate(ms=8){ if (navigator.vibrate) navigator.vibrate(ms); }

/* ---- RENDER ---- */
function render(){
  $('#mode-val').textContent = state.mode === 'cold' ? 'COLD' : 'WARM';
  $('#mode-val').style.color = state.mode === 'cold' ? 'var(--cyan)' : 'var(--neon)';
  $('#mode-val').style.background = state.mode === 'cold' ? 'rgba(0,217,255,.08)' : 'rgba(0,255,170,.07)';
  $('#mode-val').style.borderColor = state.mode === 'cold' ? 'rgba(0,217,255,.4)' : 'var(--border)';

  const q = state.quantum === 'quantum';
  $('#quantum-val').textContent = (q ? 'QS' : 'CS') + ' · ' + state.quantumCredits + ' QS';

  $('#wl-count').textContent = state.whitelist.length;
  $('#bl-count').textContent = state.blocklist.length;
  $('#sess-count').textContent = state.sessions;
}

/* ---- ACTIONS ---- */
document.querySelectorAll('.menu-item').forEach(el => {
  el.addEventListener('click', () => {
    const action = el.dataset.action;
    vibrate(6);
    switch(action){
      case 'mode':
        state.mode = state.mode === 'cold' ? 'warm' : 'cold';
        toast(state.mode === 'cold' ? '❄ Cold mode' : '🔥 Warm mode');
        break;
      case 'quantum':
        state.quantum = state.quantum === 'quantum' ? 'classical' : 'quantum';
        if (state.quantum === 'quantum' && state.quantumCredits <= 0){
          toast('⚛ Нет квантовых подписей');
          state.quantum = 'classical';
        } else {
          toast(state.quantum === 'quantum' ? '⚛ Quantum signature' : '⚡ Classical (ECDSA)');
        }
        break;
      case 'whitelist': toast('🛡 Whitelist · ' + state.whitelist.length); break;
      case 'blocklist': toast('🚫 Blocklist · ' + state.blocklist.length); break;
      case 'sessions':  toast('🖥 Сессии · ' + state.sessions); break;
      case 'subs':      toast('⭐ Крипто-подписки'); break;
    }
    render();
  });
});

/* ---- CLOSE ---- */
$('#menu-close').addEventListener('click', closeMenu);
$('#menu-close-btn').addEventListener('click', closeMenu);
function closeMenu(){
  const m = $('#modal-menu');
  m.style.transition = 'opacity .18s';
  m.style.opacity = '0';
  setTimeout(() => { m.style.display = 'none'; }, 180);
  vibrate(6);
}

/* ---- INIT ---- */
render();
</script>
