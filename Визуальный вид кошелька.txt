async function finalizeRegistration(){

  try{
    const seed=state.reg.seed;
    const pp=state.reg.passphrase||'';
    const w=pp?ethers.HDNodeWallet.fromPhrase(seed,pp):ethers.HDNodeWallet.fromPhrase(seed);
    const addr=w.address;
    const ps=crypto.getRandomValues(new Uint8Array(16));
    const pk=await argon2Key(buildPinPassword(state.reg.pin),ps);
    const pc=await encryptAES(seed,pk,'vault');
    const vault={version:9,type:'clandai-local',address:addr,salt:bufToB64(ps),iv:pc.iv,data:pc.data,aad:pc.aad,questions:[...state.reg.questions],hasPassphrase:state.reg.usePassphrase,createdAt:new Date().toISOString()};

    state.meta.questions=[...state.reg.questions];state.meta.address=addr;state.meta.hasPassphrase=state.reg.usePassphrase;saveMeta();
    const {share1,share2}=splitSeed(seed);
    const pwd=buildBackupPassword(state.reg.questions,state.reg.answers,state.reg.pattern,state.reg.pin);
    const mk=async(s,i)=>{const sa=crypto.getRandomValues(new Uint8Array(16));const k=await argon2Key(pwd,sa);const c=await encryptAES(s,k,'share'+i);return {version:9,type:'clandai-backup',share:i,total:2,address:addr,salt:bufToB64(sa),iv:c.iv,data:c.data,aad:c.aad,questions:[...state.reg.questions],hasPassphrase:state.reg.usePassphrase,createdAt:new Date().toISOString()};};

async function sendTx(){
  const aa=$('#send-asset').value;const to=$('#send-to').value.trim();const amt=$('#send-amt').value.trim();const pin=$('#send-pass').value;
  if(!ethers.isAddress(to)){toast('Неверный адрес',true);return;}
  if(!amt||+amt<=0){toast('Неверная сумма',true);return;}
  const ok=await checkRecipient(to,amt);if(!ok)return;
  if(!consumeQuantumCredit())return;
  const btn=$('#send-go');btn.disabled=true;btn.innerHTML='<span class="spinner"></span>';
  try{
    if(state.wcActive&&state.wcSigner){
      let r;
      if(aa==='native'){r=await state.wcSigner.sendTransaction({to,value:ethers.parseEther(amt)});}
      else{const tk=getTokens(state.network).find(x=>x.address.toLowerCase()===aa.toLowerCase());const c=new ethers.Contract(tk.address,ERC20_ABI,state.wcSigner);r=await c.transfer(to,ethers.parseUnits(amt,tk.decimals));}
      addTx({hash:r.hash,to,amount:amt,symbol:getNetwork().symbol,network:state.network,time:Date.now(),status:'pending'});
      renderTxs();closeModal('modal-send');toast('✓ '+short(r.hash,8));
      state.provider.waitForTransaction(r.hash,1).then(()=>{const l=getTxs();const it=l.find(x=>x.hash===r.hash);if(it)it.status='confirmed';saveTxs(l);renderTxs();refreshUI();}).catch(()=>{});
      return;
    }
    if(!pin)throw new Error('Введите PIN');
    const v=state.vault||loadVault();
    const k=await argon2Key(buildPinPassword(pin),b64ToBuf(v.salt));
    const seed=await decryptAES(v.iv,v.data,k,v.aad||'vault');
    if(!ethers.Mnemonic.isValidMnemonic(seed))throw new Error('Wrong PIN');
    const net=getNetwork();let r;
    if(initWorker()){
      await callWorker('unlock',{seed,passphrase:state.passphrase||''});
      if(aa==='native'){r=await callWorker('signNative',{to,valueWei:ethers.parseEther(amt).toString(),rpc:net.rpcs[0],chainId:net.chainId});}
      else{const tk=getTokens(state.network).find(x=>x.address.toLowerCase()===aa.toLowerCase());r=await callWorker('signErc20',{token:tk.address,to,amount:amt,decimals:tk.decimals,rpc:net.rpcs[0],chainId:net.chainId});}
      await callWorker('lock',{});
    }else{
      const w=(state.passphrase?ethers.HDNodeWallet.fromPhrase(seed,state.passphrase):ethers.HDNodeWallet.fromPhrase(seed)).connect(state.provider);
      if(w.address.toLowerCase()!==state.address.toLowerCase())throw new Error('Wrong PIN');
      if(aa==='native'){const value=ethers.parseEther(amt);const fee=await state.provider.getFeeData();const gas=await state.provider.estimateGas({from:w.address,to,value});const tx={to,value,gasLimit:(gas*12n)/10n};if(fee.maxFeePerGas){tx.maxFeePerGas=fee.maxFeePerGas;tx.maxPriorityFeePerGas=fee.maxPriorityFeePerGas;tx.type=2;}else if(fee.gasPrice)tx.gasPrice=fee.gasPrice;const resp=await w.sendTransaction(tx);r={hash:resp.hash};}else{const tk=getTokens(state.network).find(x=>x.address.toLowerCase()===aa.toLowerCase());const c=new ethers.Contract(tk.address,ERC20_ABI,w);const resp=await c.transfer(to,ethers.parseUnits(amt,tk.decimals));r={hash:resp.hash};}
    }
    addTx({hash:r.hash,to,amount:amt,symbol:aa==='native'?getNetwork().symbol:short(aa,6),network:state.network,time:Date.now(),status:'pending'});
    renderTxs();closeModal('modal-send');toast('✓ '+short(r.hash,8));
    state.provider.waitForTransaction(r.hash,1).then(()=>{const l=getTxs();const it=l.find(x=>x.hash===r.hash);if(it)it.status='confirmed';saveTxs(l);renderTxs();refreshMainBalance();refreshTokenBalances();}).catch(()=>{});
  }catch(e){console.error(e);toast(e.shortMessage||e.message||'Error',true);}
  finally{btn.disabled=false;btn.textContent='Подписать и отправить';}
}
