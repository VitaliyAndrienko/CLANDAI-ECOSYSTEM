const Shamir = (() => {
  const EXP = new Uint8Array(512), LOG = new Uint8Array(256);
  (function(){ let x=1; for (let i=0;i<255;i++){ EXP[i]=x; LOG[x]=i; x = x ^ ((x<<1) ^ ((x&0x80)?0x11b:0)); x &= 0xff; } for (let i=255;i<512;i++) EXP[i]=EXP[i-255]; })();
  const mul=(a,b)=> (a===0||b===0)?0:EXP[(LOG[a]+LOG[b])%255];
  const div=(a,b)=> { if(b===0) throw new Error('div0'); if(a===0) return 0; return EXP[(LOG[a]-LOG[b]+255)%255]; };
  function split(secret, total, threshold){
    if (threshold > total) throw new Error('threshold > total');
    const shares = [];
    for (let t=0;t<total;t++) shares.push({ x: t+1, y: new Uint8Array(secret.length) });
    for (let i=0;i<secret.length;i++){
      const coefs = new Uint8Array(threshold);
      coefs[0] = secret[i];
      if (threshold>1){ const r=randomBytes(threshold-1); for(let j=1;j<threshold;j++) coefs[j]=r[j-1]||1; }
      for (let t=0;t<total;t++){
        let val=0, xp=1;
        for (let j=0;j<threshold;j++){ val ^= mul(coefs[j],xp); xp = mul(xp, shares[t].x); }
        shares[t].y[i] = val;
      }
    }
    return shares;
  }
  function combine(shares){
    const len = shares[0].y.length;
    const secret = new Uint8Array(len);
    for (let i=0;i<len;i++){
      let s = 0;
      for (let j=0;j<shares.length;j++){
        let num=1, den=1;
        for (let k=0;k<shares.length;k++){
          if (k===j) continue;
          num = mul(num, shares[k].x);
          den = mul(den, shares[j].x ^ shares[k].x);
        }
        s ^= mul(shares[j].y[i], div(num,den));
      }
      secret[i] = s;
    }
    return secret;
  }
  return { split, combine };
})();

const Core = (() => {
  const state = {
    unlocked: false, key3Unlocked: false,
    vaultKey: null, vault: null, seed: null,
    keys: { 1:null, 2:null, 3:null },
    unlockTimer: null, key3Timer: null,
    recoveryRequests: {},
  };

  const normalizeAnswer = s => String(s).normalize('NFKC').toLowerCase().trim().replace(/\s+/g,' ');
  const normalizePattern = arr => arr.join('-');

  async function deriveVaultKey(factors, salt){
    const combined = [
      'clandai-kdf-v1',
      normalizeAnswer(factors.a1),
      normalizeAnswer(factors.a2),
      normalizeAnswer(factors.a3),
      normalizePattern(factors.pattern),
      String(factors.pin || ''),
      String(factors.passphrase || ''),
    ].join('\u0001');
    const hash = await L.argon2id({
      password: combined, salt,
      parallelism: ARGON_PARAMS.parallelism,
      iterations: ARGON_PARAMS.iterations,
      memorySize: ARGON_PARAMS.memorySize,
      hashLength: ARGON_PARAMS.hashLength,
      outputType: 'binary',
    });
    return crypto.subtle.importKey('raw', hash, { name:'AES-GCM' }, false, ['encrypt','decrypt']);
  }

  async function aesEncrypt(key, plaintextBytes, aad){
    const iv = randomBytes(12);
    const ct = await crypto.subtle.encrypt({ name:'AES-GCM', iv, additionalData: aad, tagLength:128 }, key, plaintextBytes);
    return { iv, ct: new Uint8Array(ct) };
  }
  async function aesDecrypt(key, iv, ct, aad){
    const pt = await crypto.subtle.decrypt({ name:'AES-GCM', iv, additionalData: aad, tagLength:128 }, key, ct);
    return new Uint8Array(pt);
  }

  function deriveEvmKey(seed, path){
    const root = L.HDKey.fromMasterSeed(seed);
    const child = root.derive(path);
    const pubU = L.secp.getPublicKey(child.privateKey, false);
    const pubC = L.secp.getPublicKey(child.privateKey, true);
    const addrHex = bytesToHex(L.keccak_256(pubU.slice(1)).slice(-20));
    return { privateKey: child.privateKey, publicKey: pubC, address: '0x' + addrHex, path };
  }
  const KEY_PATHS = { 1: "m/44'/60'/0'/0/0", 2: "m/44'/60'/0'/0/1", 3: "m/44'/60'/0'/0/2" };

  function wipeBytes(b){ if (b) { const r = randomBytes(b.length); b.set(r); } }
  function wipeState(){
    if (state.seed) wipeBytes(state.seed);
    for (const k of [1,2,3]) {
      const kk = state.keys[k];
      if (kk && kk.privateKey) wipeBytes(kk.privateKey);
      state.keys[k] = null;
    }
    state.seed = null; state.vault = null; state.vaultKey = null;
    state.unlocked = false; state.key3Unlocked = false;
    clearTimeout(state.unlockTimer); clearTimeout(state.key3Timer);
  }
  function scheduleUnlockTimeout(ms){ clearTimeout(state.unlockTimer); state.unlockTimer = setTimeout(()=>Core.lock('idle'), ms); }
  function scheduleKey3Timeout(ms){ clearTimeout(state.key3Timer); state.key3Timer = setTimeout(()=>Core.lockKey3(), ms); }

  function loadRawVault(){ try { return JSON.parse(localStorage.getItem(STORAGE_KEY)); } catch { return null; } }
  function saveRawVault(v){ localStorage.setItem(STORAGE_KEY, JSON.stringify(v)); }

  function rcRequest(walletAddr){
    const id = bytesToHex(randomBytes(16));
    const now = Date.now();
    state.recoveryRequests[id] = {
      id, wallet: walletAddr, chainId: TRUSTED_CONTRACT.chainId,
      contract: TRUSTED_CONTRACT.address, requestedAt: now,
      readyAt: now + TRUSTED_CONTRACT.delayMs, cancelled: false,
      code: null, used: false, nonce: bytesToHex(randomBytes(8)),
      version: TRUSTED_CONTRACT.protocolVersion,
    };
    Audit.log('ok', `Recovery requested id=${id.slice(0,12)}…`);
    return { id, readyAt: state.recoveryRequests[id].readyAt, delayMs: TRUSTED_CONTRACT.delayMs };
  }
  function rcStatus(id){
    const r = state.recoveryRequests[id];
    if (!r) return { state: 'not_found' };
    if (r.cancelled) return { state: 'cancelled' };
    const remain = r.readyAt - Date.now();
    return { id: r.id, state: remain <= 0 ? 'ready' : 'pending', remainingMs: Math.max(0, remain), readyAt: r.readyAt, wallet: r.wallet };
  }
  function rcGetCode(id){
    const r = state.recoveryRequests[id];
    if (!r) throw new Error('Unknown recovery request');
    if (r.cancelled) throw new Error('Cancelled');
    if (Date.now() < r.readyAt) throw new Error('Delay not elapsed yet');
    if (r.used) throw new Error('Code already used');
    if (!r.code){ r.code = bytesToHex(randomBytes(24)).match(/.{1,4}/g).join('-'); }
    return { id: r.id, code: r.code };
  }
  function rcCancel(id){
    const r = state.recoveryRequests[id];
    if (!r) throw new Error('Unknown recovery request');
    r.cancelled = true;
    Audit.log('warn', `Recovery cancelled id=${id.slice(0,12)}…`);
    return { ok: true };
  }

  const API = {
    async init(){
      await loadLibs();
      const raw = loadRawVault();
      return { exists: !!raw, vaultVersion: raw?.vault_version, questions: raw?.questions || null };
    },

    async createVault({ mnemonic, passphrase, factors }){
      await loadLibs();
      mnemonic = mnemonic.trim().toLowerCase().replace(/\s+/g,' ');
      if (!L.bip39.validateMnemonic(mnemonic, L.wordlist)) throw new Error('Невалидная BIP-39 мнемоника');
      passphrase = passphrase || '';

      const salt = randomBytes(16);
      const vaultKey = await deriveVaultKey({ ...factors, passphrase }, salt);
      const seedBytes = L.bip39.mnemonicToSeedSync(mnemonic, passphrase);

      const subSalt = randomBytes(16);
      const subKey = await deriveVaultKey({ ...factors, passphrase, pin: (factors.pin||'') + '|k3' }, subSalt);

      const k1 = deriveEvmKey(seedBytes, KEY_PATHS[1]);
      const k2 = deriveEvmKey(seedBytes, KEY_PATHS[2]);
      const k3 = deriveEvmKey(seedBytes, KEY_PATHS[3]);
      const wrap = await aesEncrypt(subKey, k3.privateKey, enc.encode('k3'));

      const payload = {
        mnemonic, passphrase,
        keys: {
          1: { path: KEY_PATHS[1], privateKey: bytesToHex(k1.privateKey), publicKey: bytesToHex(k1.publicKey), address: k1.address },
          2: { path: KEY_PATHS[2], privateKey: bytesToHex(k2.privateKey), publicKey: bytesToHex(k2.publicKey), address: k2.address },
        },
        key3: {
          path: KEY_PATHS[3], address: k3.address, publicKey: bytesToHex(k3.publicKey),
          wrapped: { iv: bytesToHex(wrap.iv), ct: bytesToHex(wrap.ct) },
          subSalt: bytesToHex(subSalt),
        },
        createdAt: Date.now(),
      };

      const aad = enc.encode(`clandai-vault|v${VAULT_VERSION}`);
      const { iv, ct } = await aesEncrypt(vaultKey, enc.encode(JSON.stringify(payload)), aad);

      const raw = {
        vault_version: VAULT_VERSION, kdf_version: KDF_VERSION,
        normalization_version: NORM_VERSION, encryption_version: ENC_VERSION,
        kdf: { name: 'argon2id', ...ARGON_PARAMS },
        salt: bytesToHex(salt),
        cipher: { name: 'AES-256-GCM', iv: bytesToHex(iv), ct: bytesToHex(ct) },
        aad: 'clandai-vault|v1',
        trusted_contract: { ...TRUSTED_CONTRACT },
        recovery_code_hash: null,
        questions: { q1: factors.q1 || '', q2: factors.q2 || '', q3: factors.q3 || '' },
        has_passphrase: !!passphrase,
      };

      saveRawVault(raw);
      Audit.log('ok', 'Vault created');
      wipeBytes(seedBytes);
      return { ok: true, key1Address: k1.address, key2Address: k2.address, key3Address: k3.address };
    },

    async setRecoveryCodeHash(code){
      await loadLibs();
      const raw = loadRawVault();
      if (!raw) throw new Error('No vault');
      raw.recovery_code_hash = bytesToHex(L.sha256(enc.encode(code)));
      saveRawVault(raw);
    },

    async unlock(factors){
      await loadLibs();
      const raw = loadRawVault();
      if (!raw) throw new Error('Vault не создан');
      const salt = hexToBytes(raw.salt);
      const vaultKey = await deriveVaultKey({
        a1: factors.a1, a2: factors.a2, a3: factors.a3,
        pattern: factors.pattern, pin: factors.pin,
        passphrase: factors.passphrase || '',
      }, salt);
      const iv = hexToBytes(raw.cipher.iv);
      const ct = hexToBytes(raw.cipher.ct);
      const aad = enc.encode(raw.aad || 'clandai-vault|v1');
      let pt;
      try { pt = await aesDecrypt(vaultKey, iv, ct, aad); }
      catch { Audit.log('err', 'Vault unlock failed'); throw new Error('Неверные факторы или повреждённый vault'); }

      const vault = JSON.parse(dec.decode(pt));
      state.vault = vault; state.vaultKey = vaultKey;
      state.seed = L.bip39.mnemonicToSeedSync(vault.mnemonic, vault.passphrase || '');

      state.keys[1] = { privateKey: hexToBytes(vault.keys[1].privateKey), publicKey: hexToBytes(vault.keys[1].publicKey), address: vault.keys[1].address, path: vault.keys[1].path };
      state.keys[2] = { privateKey: hexToBytes(vault.keys[2].privateKey), publicKey: hexToBytes(vault.keys[2].publicKey), address: vault.keys[2].address, path: vault.keys[2].path };
      state.keys[3] = null;

      state.unlocked = true; state.key3Unlocked = false;
      Audit.log('ok', 'Vault unlocked');
      scheduleUnlockTimeout(10 * 60 * 1000);
      return { addresses: { 1: state.keys[1].address, 2: state.keys[2].address, 3: vault.key3.address }, key3Locked: true };
    },

    async unlockKey3(factors){
      if (!state.unlocked) throw new Error('Vault не разблокирован');
      await loadLibs();
      const v = state.vault;
      const subSalt = hexToBytes(v.key3.subSalt);
      const subKey = await deriveVaultKey({
        a1: factors.a1, a2: factors.a2, a3: factors.a3,
        pattern: factors.pattern, pin: factors.pin,
        passphrase: v.passphrase || '',
      }, subSalt);
      const iv = hexToBytes(v.key3.wrapped.iv);
      const ct = hexToBytes(v.key3.wrapped.ct);
      let priv;
      try { priv = await aesDecrypt(subKey, iv, ct, enc.encode('k3')); }
      catch { Audit.log('err', 'Key3 unlock failed'); throw new Error('Неверные факторы для Key 3'); }
      state.keys[3] = { privateKey: priv, publicKey: hexToBytes(v.key3.publicKey), address: v.key3.address, path: v.key3.path };
      state.key3Unlocked = true;
      Audit.log('ok', 'Key 3 unlocked');
      scheduleKey3Timeout(5 * 60 * 1000);
      return { ok: true, address: state.keys[3].address };
    },

    lockKey3(){
      if (state.keys[3]?.privateKey) wipeBytes(state.keys[3].privateKey);
      state.keys[3] = null; state.key3Unlocked = false;
      clearTimeout(state.key3Timer);
      Audit.log('warn', 'Key 3 locked');
    },
    lock(reason='manual'){ wipeState(); Audit.log('warn', `Vault locked (${reason})`); },

    getState(){
      return {
        unlocked: state.unlocked, key3Unlocked: state.key3Unlocked,
        addresses: { 1: state.keys[1]?.address || null, 2: state.keys[2]?.address || null, 3: state.vault?.key3?.address || null },
        paths: { 1: KEY_PATHS[1], 2: KEY_PATHS[2], 3: KEY_PATHS[3] },
      };
    },

    _checkPolicy(tx){
      const errors = []; const warnings = [];
      if (!tx.chainId || Number.isNaN(Number(tx.chainId))) errors.push('chainId отсутствует');
      if (!/^0x[0-9a-fA-F]{40}$/.test(tx.to || '')) errors.push('Некорректный "to" адрес');
      if (tx.to && tx.to.toLowerCase() === TRUSTED_CONTRACT.address.toLowerCase() && tx.data && tx.data.length > 10) warnings.push('Транзакция направлена на trusted recovery contract');
      if (tx.value && BigInt(tx.value || 0) > 0n) warnings.push(`Value > 0: ${tx.value} wei`);
      if (tx.data && tx.data.length > 2 && tx.data.length < 10) errors.push('data слишком короткий');
      if (tx.gasLimit && (Number(tx.gasLimit) < 21000)) errors.push('gasLimit < 21000');
      return { ok: errors.length === 0, errors, warnings };
    },

    async signTransaction(tx, keyId=1){
      if (!state.unlocked) throw new Error('Vault не разблокирован');
      if (keyId === 3 && !state.key3Unlocked) throw new Error('Key 3 заблокирован: требуется 3-факторный unlock');
      const k = state.keys[keyId];
      if (!k) throw new Error(`Key ${keyId} недоступен`);

      const policy = this._checkPolicy(tx);
      Audit.log(policy.ok ? 'info' : 'err', `Policy check key=${keyId} ok=${policy.ok} errors=${policy.errors.join(';')}`);
      if (!policy.ok) throw new Error('Policy check failed: ' + policy.errors.join('; '));

      await loadLibs();
      const chainId = BigInt(tx.chainId);
      const nonce = BigInt(tx.nonce || 0);
      const gasPrice = BigInt(Math.floor(Number(tx.maxFeeGwei || 20) * 1e9));
      const gasLimit = BigInt(tx.gasLimit || 21000);
      const to = hexToBytes(tx.to);
      const value = BigInt(tx.value || 0);
      const data = hexToBytes(tx.data && tx.data !== '0x' ? tx.data : '0x');

      const toRlp = to.length ? to : new Uint8Array(0);
      const unsigned = [nonce, gasPrice, gasLimit, toRlp, value, data, chainId, 0, 0];
      const rlpEncoded = L.rlp.encode(unsigned);
      const msgHash = L.keccak_256(rlpEncoded);

      const sig = L.secp.sign(msgHash, k.privateKey, { lowS: true });
      const sigBytes = sig.toCompactRawBytes ? sig.toCompactRawBytes() : sig;
      const r = sigBytes.slice(0, 32);
      const s = sigBytes.slice(32, 64);
      const v = BigInt(sig.recovery ?? 0);
      const vEip155 = v + 35n + chainId * 2n;

      const signed = [nonce, gasPrice, gasLimit, toRlp, value, data, vEip155, r, s];
      const rawTx = '0x' + bytesToHex(L.rlp.encode(signed));

      Audit.log('ok', `Tx signed key=${keyId} to=${tx.to.slice(0,10)}…`);
      return { rawTransaction: rawTx, txHash: '0x' + bytesToHex(L.keccak_256(L.rlp.encode(signed))), signer: k.address, policy };
    },

    async createFullBackup(){
      const raw = loadRawVault();
      if (!raw) throw new Error('Vault не найден');
      Audit.log('info', 'Full backup created');
      return { type: 'clandai-full-backup', version: 1, createdAt: Date.now(), vault: raw };
    },

    async createShamirBackup(total, threshold){
      const raw = loadRawVault();
      if (!raw) throw new Error('Vault не найден');
      if (threshold < 2 || threshold > total) throw new Error('threshold должен быть 2..total');
      const vaultBytes = enc.encode(JSON.stringify(raw));
      const shares = Shamir.split(vaultBytes, total, threshold);
      const meta = { type: 'clandai-shamir-share', version: 1, createdAt: Date.now(), total, threshold, vault_version: raw.vault_version };
      const out = shares.map(s => ({ ...meta, x: s.x, y: bytesToHex(s.y) }));
      Audit.log('info', `Shamir backup created ${threshold}/${total}`);
      return out;
    },

    async restoreFromFull(backup, factors){
      await loadLibs();
      if (!backup || backup.type !== 'clandai-full-backup') throw new Error('Неверный формат backup');
      const raw = backup.vault;
      if (!raw || !raw.cipher || !raw.salt) throw new Error('Повреждённый backup');
      const salt = hexToBytes(raw.salt);
      const vaultKey = await deriveVaultKey({
        a1: factors.a1, a2: factors.a2, a3: factors.a3,
        pattern: factors.pattern, pin: factors.pin,
        passphrase: factors.passphrase || '',
      }, salt);
      const iv = hexToBytes(raw.cipher.iv);
      const ct = hexToBytes(raw.cipher.ct);
      const aad = enc.encode(raw.aad || 'clandai-vault|v1');
      let pt;
      try { pt = await aesDecrypt(vaultKey, iv, ct, aad); }
      catch { throw new Error('Не удалось расшифровать backup: неверные факторы или повреждение'); }
      const parsed = JSON.parse(dec.decode(pt));
      if (!parsed.mnemonic || !parsed.keys) throw new Error('Vault payload невалиден');
      saveRawVault(raw);
      Audit.log('ok', 'Vault restored from full backup');
      return { ok: true, key1Address: parsed.keys[1].address, key2Address: parsed.keys[2].address };
    },

    async restoreFromShares(shares, factors){
      await loadLibs();
      if (!Array.isArray(shares) || shares.length < 2) throw new Error('Нужно минимум 2 shares');
      const threshold = shares[0].threshold;
      if (shares.length < threshold) throw new Error(`Нужно минимум ${threshold} shares (у вас ${shares.length})`);
      const seen = new Set();
      for (const s of shares){
        if (s.type !== 'clandai-shamir-share') throw new Error('Неверный тип share');
        if (seen.has(s.x)) continue;
        seen.add(s.x);
      }
      const picked = shares.slice(0, threshold).map(s => ({ x: s.x, y: hexToBytes(s.y) }));
      let vaultBytes;
      try { vaultBytes = Shamir.combine(picked); }
      catch { throw new Error('Не удалось объединить shares'); }
      let raw;
      try { raw = JSON.parse(dec.decode(vaultBytes)); }
      catch { throw new Error('Восстановленный payload повреждён'); }
      return this.restoreFromFull({ type: 'clandai-full-backup', version: 1, vault: raw }, factors);
    },

    rcRequest(){ return rcRequest(this.getState().addresses[1] || '0x0'); },
    rcStatus(id){ return rcStatus(id); },
    rcGetCode(id){ return rcGetCode(id); },
    rcCancel(id){ return rcCancel(id); },

    async revealSeed(recoveryCode){
      await loadLibs();
      if (!state.unlocked) throw new Error('Сначала разблокируйте vault');
      const raw = loadRawVault();
      if (!raw.recovery_code_hash) throw new Error('Recovery code не установлен');
      const h = bytesToHex(L.sha256(enc.encode(recoveryCode)));
      if (h !== raw.recovery_code_hash){ Audit.log('err', 'Secret reveal failed (bad code)'); throw new Error('Неверный recovery code'); }
      Audit.log('ok', 'Secret reveal');
      return { mnemonic: state.vault.mnemonic };
    },

    audit(){ return Audit.all(); },
    auditClear(){ Audit.clear(); },
    _wipe(){ wipeState(); },
  };

  setInterval(()=>{
    for (const id in state.recoveryRequests){
      const r = state.recoveryRequests[id];
      if (!r.cancelled && r.code && !r.used && Date.now() - r.readyAt > 24*60*60*1000) delete state.recoveryRequests[id];
    }
  }, 60_000);

  return API;
})();

function mountPattern(wrapEl, onChange){
  wrapEl.innerHTML = '';
  wrapEl.classList.add('pattern-wrap');

  const grid = document.createElement('div');
  grid.className = 'pattern-grid';
  wrapEl.appendChild(grid);

  const svg = document.createElementNS('http://www.w3.org/2000/svg','svg');
  svg.setAttribute('class','pattern-svg');
  svg.setAttribute('preserveAspectRatio','none');
  const pathDone = document.createElementNS('http://www.w3.org/2000/svg','path');
  const pathLive = document.createElementNS('http://www.w3.org/2000/svg','path');
  pathLive.setAttribute('class','live');
  svg.appendChild(pathDone);
  svg.appendChild(pathLive);
  wrapEl.appendChild(svg);

  const dots = [];
  for (let i = 0; i < 9; i++){
    const d = document.createElement('div');
    d.className = 'pattern-dot';
    d.dataset.idx = i;
    grid.appendChild(d);
    dots.push(d);
  }

  const seq = [];
  let dragging = false;
  let startDot = null;

  function dotCenter(idx){
    const d = dots[idx];
    const r = d.getBoundingClientRect();
    const w = wrapEl.getBoundingClientRect();
    return { x: r.left - w.left + r.width/2, y: r.top - w.top + r.height/2 };
  }

  function updateSvg(){
    const w = wrapEl.clientWidth, h = wrapEl.clientHeight;
    svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
    svg.setAttribute('width', w);
    svg.setAttribute('height', h);

    if (seq.length >= 2){
      const pts = seq.map(i => dotCenter(i));
      pathDone.setAttribute('d', pts.map((p,k)=> (k===0?'M':'L') + p.x.toFixed(1) + ' ' + p.y.toFixed(1)).join(' '));
    } else {
      pathDone.setAttribute('d','');
    }
    pathLive.setAttribute('d','');
  }

  function hitDot(clientX, clientY){
    for (let i = 0; i < dots.length; i++){
      const r = dots[i].getBoundingClientRect();
      const cx = r.left + r.width/2, cy = r.top + r.height/2;
      const rr = Math.max(r.width, r.height) * 0.75;
      const dx = clientX - cx, dy = clientY - cy;
      if (dx*dx + dy*dy <= rr*rr) return i;
    }
    return -1;
  }

  function addDot(i){
    if (seq.includes(i)) return;
    seq.push(i);
    dots[i].classList.add('active');
    onChange(seq.slice());
    updateSvg();
  }

  function onDown(e){
    e.preventDefault();
    wrapEl.setPointerCapture?.(e.pointerId);
    const i = hitDot(e.clientX, e.clientY);
    if (i < 0) return;
    dragging = true;
    startDot = i;
    seq.length = 0;
    dots.forEach(d=>d.classList.remove('active'));
    addDot(i);
  }

  function onMove(e){
    if (!dragging) return;
    e.preventDefault();
    const i = hitDot(e.clientX, e.clientY);
    if (i >= 0 && !seq.includes(i)) addDot(i);
    // live line
    if (seq.length){
      const w = wrapEl.getBoundingClientRect();
      const last = dotCenter(seq[seq.length-1]);
      const cur = { x: e.clientX - w.left, y: e.clientY - w.top };
      const pts = seq.map(k => dotCenter(k));
      const d = pts.map((p,k)=> (k===0?'M':'L') + p.x.toFixed(1) + ' ' + p.y.toFixed(1)).join(' ')
              + ` L ${cur.x.toFixed(1)} ${cur.y.toFixed(1)}`;
      pathLive.setAttribute('d', d);
    }
  }

  function onUp(e){
    if (!dragging) return;
    dragging = false;
    pathLive.setAttribute('d','');
    try { wrapEl.releasePointerCapture?.(e.pointerId); } catch(_){}
  }

  wrapEl.addEventListener('pointerdown', onDown);
  wrapEl.addEventListener('pointermove', onMove);
  wrapEl.addEventListener('pointerup', onUp);
  wrapEl.addEventListener('pointercancel', onUp);
  wrapEl.addEventListener('pointerleave', onUp);
  wrapEl.addEventListener('contextmenu', e=>e.preventDefault());

  // Поправка размеров при resize/первом layout
  requestAnimationFrame(updateSvg);
  window.addEventListener('resize', updateSvg);

  const reset = () => {
    seq.length = 0;
    dots.forEach(d=>d.classList.remove('active'));
    pathDone.setAttribute('d','');
    pathLive.setAttribute('d','');
    onChange(seq.slice());
  };

  return { reset, value: () => seq.slice() };
}
