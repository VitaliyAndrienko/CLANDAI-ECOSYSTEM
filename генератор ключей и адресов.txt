  const mods = await Promise.all(LIB_URLS.map(u => import(/* @vite-ignore */ u)));
  const [bip39, bip39en, bip32, sha256m, ripemd160m, sha512m, hmacm,
         base, secp, ed, sha3, nacl, starknet, cardano, tonweb, bip39v3] = mods;
  libs = {
    bip39, wordlist: bip39en.wordlist, HDKey: bip32.HDKey,
    sha256: sha256m.sha256, ripemd160: ripemd160m.ripemd160,
    sha512: sha512m.sha512, hmac: hmacm.hmac,
    base58: base.base58, base58check: base.base58check,
    bech32: base.bech32, bech32m: base.bech32m,
    secp, ed25519: ed.ed25519, keccak_256: sha3.keccak_256,
    nacl, starknet, cardano, tonweb, bip39v3,
  };
  return libs;
}

const bytesToHex = b => Array.from(b, x => x.toString(16).padStart(2, '0')).join('');
const enc = new TextEncoder();

function toChecksumEvmAddress(addrHex, keccak_256) {
  const lower = addrHex.toLowerCase().replace(/^0x/, '');
  const hash = bytesToHex(keccak_256(enc.encode(lower)));
  let out = '0x';
  for (let i = 0; i < 40; i++) out += parseInt(hash[i], 16) >= 8 ? lower[i].toUpperCase() : lower[i];
  return out;
}

function slip10Ed25519(seed, indices, hmac, sha512) {
  let I = hmac(sha512, enc.encode('ed25519 seed'), seed);
  let key = I.slice(0, 32);
  let chain = I.slice(32, 64);
  for (const idx of indices) {
    if (idx < 0x80000000) throw new Error('ed25519 path must be fully hardened');
    const data = new Uint8Array(37);
    data.set(key, 1);
    new DataView(data.buffer).setUint32(33, idx, false);
    I = hmac(sha512, chain, data);
    key = I.slice(0, 32);
    chain = I.slice(32, 64);
  }
  return key;
}

function parseEdPath(path, index) {
  return path.replace('{i}', index).split('/').slice(1).map(p => {
    const hardened = p.endsWith("'");
    const n = parseInt(hardened ? p.slice(0, -1) : p, 10);
    return hardened ? (n + 0x80000000) >>> 0 : n;
  });
}

function btcAddress(L, pubkey, opts) {
  const { sha256, ripemd160, base58check, bech32, bech32m, secp } = L;
  const b58 = base58check(sha256);
  const h160 = d => ripemd160(sha256(d));
  const p2pkhVer = opts.p2pkhVer ?? 0x00;
  const p2shVer  = opts.p2shVer  ?? 0x05;
  if (opts.addrType === 'p2pkh') return b58.encode(new Uint8Array([p2pkhVer, ...h160(pubkey)]));
  if (opts.addrType === 'p2sh') {
    const h = h160(pubkey);
    const redeem = new Uint8Array([0x00, 0x14, ...h]);
    return b58.encode(new Uint8Array([p2shVer, ...h160(redeem)]));
  }
  if (opts.addrType === 'p2wpkh') return bech32.encode(opts.hrp, [0, ...bech32.toWords(h160(pubkey))]);
  if (opts.addrType === 'p2tr') {
    const internalXOnly = pubkey.slice(1);
    const tagHash = sha256(enc.encode('TapTweak'));
    const tweakInput = new Uint8Array(96);
    tweakInput.set(tagHash, 0); tweakInput.set(tagHash, 32); tweakInput.set(internalXOnly, 64);
    const tweak = sha256(tweakInput);
    const P = secp.ProjectivePoint;
    let internalPoint = P.fromHex(pubkey);
    if (pubkey[0] === 0x03) internalPoint = internalPoint.negate();
    const tweakScalar = BigInt('0x' + bytesToHex(tweak));
    const outputPoint = internalPoint.add(P.BASE.multiply(tweakScalar));
    const outXOnly = outputPoint.toRawBytes(true).slice(1);
    return bech32m.encode(opts.hrp, [1, ...bech32m.toWords(outXOnly)]);
  }
  throw new Error('Unknown addrType: ' + opts.addrType);
}

function cashaddrEncode(prefix, type, hash) {
  const CHARSET = 'qpzry9x8gf2tvdw0s3jn54khce6mua7l';
  const GENERATOR = [0x98f2bc8e61, 0x79b76d99e2, 0xf33e5fb3c4, 0xae2eabe2a8, 0x1e4f43e470];
  function polymod(values) {
    let chk = 1;
    for (const v of values) {
      const top = chk >> 35;
      chk = ((chk & 0x07ffffffff) << 5) ^ v;
      for (let i = 0; i < 5; i++) if ((top >> i) & 1) chk ^= GENERATOR[i];
    }
    return chk;
  }
  const data = [type, ...hash];
  const payload = [];
  for (const b of data) for (let i = 4; i >= 0; i--) payload.push((b >> i) & 1);
  const words = [];
  for (let i = 0; i < payload.length; i += 5) {
    let w = 0;
    for (let j = 0; j < 5; j++) w = (w << 1) | (payload[i + j] || 0);
    words.push(w);
  }
  const prefixVals = [...prefix].map(c => c.charCodeAt(0) & 0x1f);
  const checksumData = [...prefixVals, 0, ...words, 0, 0, 0, 0, 0, 0, 0, 0];
  const chk = polymod(checksumData) ^ 1;
  const checksum = [];
  for (let i = 0; i < 8; i++) checksum.push((chk >> (5 * (7 - i))) & 0x1f);
  return prefix + ':' + [...words, ...checksum].map(w => CHARSET[w]).join('');
}

const RIPPLE_ALPHABET = 'rpshnaf39wBUDNEGHJKLM4PQRST7VWXYZ2bcdeCg65jkm8oFqi1tuvAxyz';
function rippleBase58Encode(payload, sha256) {
  const data = new Uint8Array(payload.length + 4);
  data.set(payload, 0);
  data.set(sha256(sha256(payload)).slice(0, 4), payload.length);
  let num = 0n;
  for (const b of data) num = num * 256n + BigInt(b);
  let out = '';
  while (num > 0n) { out = RIPPLE_ALPHABET[Number(num % 58n)] + out; num /= 58n; }
  let zeros = 0;
  for (const b of data) { if (b === 0) zeros++; else break; }
  return RIPPLE_ALPHABET[0].repeat(zeros) + out;
}

const MONERO_ALPHABET = '123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz';
const MONERO_ENCODED_BLOCK_SIZES = [0, 2, 3, 5, 6, 7, 9, 10, 11];
const ED25519_L = BigInt('0x1000000000000000000000000000000014def9dea2f79cd65812631a5cf5d3ed');

function moneroBase58Encode(data) {
  let out = '';
  for (let i = 0; i < data.length; i += 8) {
    const block = data.slice(i, i + 8);
    const encodedSize = MONERO_ENCODED_BLOCK_SIZES[block.length];
    let num = 0n;
    for (const b of block) num = (num << 8n) | BigInt(b);
    let str = '';
    for (let j = 0; j < encodedSize; j++) {
      str = MONERO_ALPHABET[Number(num % 58n)] + str;
      num /= 58n;
    }
    out += str;
  }
  return out;
}

function bytesToBigIntLE(bytes) {
  let n = 0n;
  for (let i = bytes.length - 1; i >= 0; i--) n = (n << 8n) | BigInt(bytes[i]);
  return n;
}
function bigIntToBytesLE(n, length) {
  const out = new Uint8Array(length);
  for (let i = 0; i < length; i++) { out[i] = Number(n & 0xffn); n >>= 8n; }
  return out;
}

function moneroAddress(L, mnemonic, passphrase, index) {
  const { hmac, sha512, keccak_256, ed25519 } = L;
  const seed = L.bip39.mnemonicToSeedSync(mnemonic, passphrase);
  const indices = parseEdPath(`m/44'/128'/{i}'`, index);
  const spendKeyRaw = slip10Ed25519(seed, indices, hmac, sha512);
  let spendInt = bytesToBigIntLE(spendKeyRaw) % ED25519_L;
  if (spendInt === 0n) spendInt = 1n;
  const spendKey = bigIntToBytesLE(spendInt, 32);
  const viewHash = keccak_256(spendKey);
  let viewInt = bytesToBigIntLE(viewHash) % ED25519_L;
  if (viewInt === 0n) viewInt = 1n;
  const viewKey = bigIntToBytesLE(viewInt, 32);
  const P = ed25519.ExtendedPoint;
  const spendPub = P.BASE.multiply(spendInt).toRawBytes();
  const viewPub  = P.BASE.multiply(viewInt).toRawBytes();
  const payload = new Uint8Array([18, ...spendPub, ...viewPub]);
  const checksum = keccak_256(payload).slice(0, 4);
  const full = new Uint8Array([...payload, ...checksum]);
  return {
    address: moneroBase58Encode(full),
    privateKey: bytesToHex(spendKey),
    viewKey: bytesToHex(viewKey),
    publicKey: bytesToHex(spendPub),
    viewPublicKey: bytesToHex(viewPub),
  };
}

async function starknetAddress(L, seed, index) {
  const { starknet, HDKey } = L;
  const root = HDKey.fromMasterSeed(seed);
  const child = root.derive(`m/44'/9004'/0'/0/${index}`);
  const N = BigInt('0x0800000000000010ffffffffffffffffffffffffffffffffffffffffffffffff');
  let key = BigInt('0x' + bytesToHex(child.privateKey)) % N;
  if (key === 0n) key = 1n;
  const privateKeyHex = '0x' + key.toString(16).padStart(64, '0');
  let pubKey;
  try { pubKey = starknet.ec.getStarkKey(privateKeyHex); }
  catch { pubKey = privateKeyHex; }
  const OZ_CLASS_HASH = '0x05b4b537eaa2399e3aa99c4e2e0208ebd6c71bc1467938cd52c798c601e43564';
  let address;
  try {
    const addr = starknet.hash.calculateContractAddressFromHash(pubKey, OZ_CLASS_HASH, [pubKey], 0);
    address = '0x' + BigInt(addr).toString(16).padStart(64, '0');
  } catch {
    address = '0x' + BigInt(pubKey).toString(16).padStart(64, '0');
  }
  return { privateKey: privateKeyHex, publicKey: pubKey, address };
}

// Cardano
// ============================================================
async function cardanoAddress(L, mnemonic, passphrase, index) {
  const { cardano, bip39v3 } = L;
  const CSL = cardano;
  const entropyHex = bip39v3.mnemonicToEntropy(mnemonic);
  const entropy = new Uint8Array(entropyHex.match(/.{1,2}/g).map(b => parseInt(b, 16)));
  const passBytes = passphrase ? enc.encode(passphrase) : new Uint8Array(0);
  const rootKey = CSL.Bip32PrivateKey.from_bip39_entropy(entropy, passBytes);
  const harden = n => 0x80000000 + n;
  const accountKey = rootKey.derive(harden(1852)).derive(harden(1815)).derive(harden(index));
  const utxoPubKey = accountKey.derive(0).derive(0).to_public();
  const stakeKey   = accountKey.derive(2).derive(0).to_public();
  const baseAddr = CSL.BaseAddress.new(
    CSL.NetworkInfo.mainnet().network_id(),
    CSL.StakeCredential.from_keyhash(utxoPubKey.hash()),
    CSL.StakeCredential.from_keyhash(stakeKey.hash())
  );
  return {
    address: baseAddr.to_address().to_bech32(),
    privateKey: bytesToHex(accountKey.to_raw_key().as_bytes()),
    publicKey: bytesToHex(utxoPubKey.as_bytes()),
  };
}

// TON (V4R2)
// ============================================================
async function tonAddress(L, mnemonic, index) {
  const { tonweb, hmac, sha512, ed25519 } = L;
  const TonWeb = tonweb.default || tonweb;
  const seed = L.bip39.mnemonicToSeedSync(mnemonic, '');
  const indices = parseEdPath(`m/44'/607'/{i}'`, index);
  const key32 = slip10Ed25519(seed, indices, hmac, sha512);
  const pub = ed25519.getPublicKey(key32);
  try {
    const tonwebInstance = new TonWeb();
    const WalletClass = tonwebInstance.wallet.all.v4R2;
    const wallet = new WalletClass(tonwebInstance.provider, { publicKey: pub });
    const addr = await wallet.getAddress();
    return { address: addr.toString({ bounceable: false, testOnly: false }),
             privateKey: bytesToHex(key32), publicKey: bytesToHex(pub) };
  } catch {
    const wc = new Uint8Array([0]);
    const hash = L.sha256(new Uint8Array([...wc, ...pub]));
    const stateInitHash = L.sha256(new Uint8Array([...hash.slice(0, 32)]));
    const payload = new Uint8Array([0x11, ...stateInitHash]);
    return { address: L.base58check(L.sha256).encode(payload),
             privateKey: bytesToHex(key32), publicKey: bytesToHex(pub) };
  }
}

// Kaspa, Injective
// ============================================================
function kaspaAddress(L, pubkey) {
  const h = L.sha256(pubkey).slice(0, 20);
  return L.bech32.encode('kaspa', L.bech32.toWords(h));
}

function injectiveCosmosAddress(L, childPublicKey) {
  const pubU = L.secp.getPublicKey(childPublicKey, false).slice(1);
  const h = L.keccak_256(pubU).slice(-20);
  return L.bech32.encode('inj', L.bech32.toWords(h));
}

async function deriveAll(mnemonic, passphrase, index, selectedIds, showPriv) {
  const L = await loadLibs();
  const mn = mnemonic.trim().toLowerCase().replace(/\s+/g, ' ');
  if (!L.bip39.validateMnemonic(mn, L.wordlist)) {
    throw new Error('Невалидная BIP-39 мнемоника (проверьте checksum и слова)');
  }
  const seed = L.bip39.mnemonicToSeedSync(mn, passphrase);
  const root = L.HDKey.fromMasterSeed(seed);
  const results = [];
  const evmCache = new Map();

  for (const net of NETWORKS) {
    if (!selectedIds.has(net.id)) continue;
    try {
      if (net.type === 'evm') {
        const src = net.sameAs ? NETWORKS.find(n => n.id === net.sameAs) : net;
        const path = src.path.replace('{i}', index);
        let e = evmCache.get(path);
        if (!e) {
          const child = root.derive(path);
          const pubU = L.secp.getPublicKey(child.privateKey, false);
          const pubC = L.secp.getPublicKey(child.privateKey, true);
          const addrHex = bytesToHex(L.keccak_256(pubU.slice(1)).slice(-20));
          e = {address: toChecksumEvmAddress(addrHex, L.keccak_256), path,
               privateKey: child.privateKey, publicKey: bytesToHex(pubC)};
          evmCache.set(path, e);
        }
        results.push({id:net.id, name:net.name, group:net.group, address:e.address, path:e.path,
          publicKey: '0x' + e.publicKey,
          privateKey: showPriv ? '0x' + bytesToHex(e.privateKey) : null,
          note: net.note || (net.sameAs ? '= ' + src.name : null)});
      }
      else if (net.type === 'btc') {
        const path = net.path.replace('{i}', index);
        const child = root.derive(path);
        const address = btcAddress(L, child.publicKey, {
          addrType: net.addrType, hrp: net.hrp,
          p2pkhVer: net.p2pkhVer, p2shVer: net.p2shVer,
        });
        results.push({id:net.id, name:net.name, group:net.group, address, path,
          publicKey: bytesToHex(child.publicKey),
          privateKey: showPriv ? bytesToHex(child.privateKey) : null});
      }
      else if (net.type === 'bch') {
        const path = net.path.replace('{i}', index);
        const child = root.derive(path);
        const h160 = L.ripemd160(L.sha256(child.publicKey));
        const prefix = net.id === 'bch' ? 'bitcoincash' : 'ecash';
        const address = cashaddrEncode(prefix, 0, h160);
        results.push({id:net.id, name:net.name, group:net.group, address, path,
          publicKey: bytesToHex(child.publicKey),
          privateKey: showPriv ? bytesToHex(child.privateKey) : null});
      }
      else if (net.type === 'kaspa') {
        const path = net.path.replace('{i}', index);
        const child = root.derive(path);
        const address = kaspaAddress(L, child.publicKey);
        results.push({id:net.id, name:net.name, group:net.group, address, path,
          publicKey: bytesToHex(child.publicKey),
          privateKey: showPriv ? bytesToHex(child.privateKey) : null, note: net.note});
      }
      else if (net.type === 'ed25519') {
        const indices = parseEdPath(net.path, index);
        const key32 = slip10Ed25519(seed, indices, L.hmac, L.sha512);
        const pub = L.ed25519.getPublicKey(key32);
        const address = (net.id === 'solana' || net.id === 'near' || net.id === 'algorand')
          ? L.base58.encode(pub) : '0x' + bytesToHex(pub);
        results.push({id:net.id, name:net.name, group:net.group, address,
          path: net.path.replace('{i}', index),
          publicKey: bytesToHex(pub),
          privateKey: showPriv ? bytesToHex(key32) : null});
      }
      else if (net.type === 'ton') {
        const r = await tonAddress(L, mn, index);
        results.push({id:net.id, name:net.name, group:net.group, address: r.address,
          path: net.path.replace('{i}', index),
          publicKey: r.publicKey,
          privateKey: showPriv ? r.privateKey : null, note: net.note});
      }
      else if (net.type === 'stellar') {
        const indices = parseEdPath(net.path, index);
        const key32 = slip10Ed25519(seed, indices, L.hmac, L.sha512);
        const pub = L.ed25519.getPublicKey(key32);
        const address = L.base58check(L.sha256).encode(new Uint8Array([0x30, ...pub]));
        results.push({id:net.id, name:net.name, group:net.group, address,
          path: net.path.replace('{i}', index),
          publicKey: bytesToHex(pub),
          privateKey: showPriv ? bytesToHex(key32) : null});
      }
      else if (net.type === 'algorand') {
        const indices = parseEdPath(net.path, index);
        const key32 = slip10Ed25519(seed, indices, L.hmac, L.sha512);
        const pub = L.ed25519.getPublicKey(key32);
        results.push({id:net.id, name:net.name, group:net.group, address: L.base58.encode(pub),
          path: net.path.replace('{i}', index),
          publicKey: bytesToHex(pub),
          privateKey: showPriv ? bytesToHex(key32) : null});
      }
      else if (net.type === 'tezos') {
        const indices = parseEdPath(net.path, index);
        const key32 = slip10Ed25519(seed, indices, L.hmac, L.sha512);
        const pub = L.ed25519.getPublicKey(key32);
        const pkh = L.sha256(pub).slice(0, 20);
        const address = L.base58check(L.sha256).encode(new Uint8Array([0x06, 0xa1, 0x9f, ...pkh]));
        results.push({id:net.id, name:net.name, group:net.group, address,
          path: net.path.replace('{i}', index),
          publicKey: bytesToHex(pub),
          privateKey: showPriv ? bytesToHex(key32) : null});
      }
      else if (net.type === 'hedera') {
        const indices = parseEdPath(net.path, index);
        const key32 = slip10Ed25519(seed, indices, L.hmac, L.sha512);
        const pub = L.ed25519.getPublicKey(key32);
        const address = '0.0.' + (BigInt('0x' + bytesToHex(pub)) % 1000000n + 1n);
        results.push({id:net.id, name:net.name, group:net.group, address,
          path: net.path.replace('{i}', index),
          publicKey: bytesToHex(pub),
          privateKey: showPriv ? bytesToHex(key32) : null});
      }
      else if (net.type === 'monero') {
        const r = moneroAddress(L, mn, passphrase, index);
        results.push({id:net.id, name:net.name, group:net.group, address: r.address,
          path: "m/44'/128'/{i}' (Ledger-style)",
          publicKey: r.publicKey,
          privateKey: showPriv ? r.privateKey : null,
          viewKey: showPriv ? r.viewKey : null,
          viewPublicKey: r.viewPublicKey,
          note: net.note});
      }
      else if (net.type === 'substrate') {
        const indices = parseEdPath(net.path, index);
        const key32 = slip10Ed25519(seed, indices, L.hmac, L.sha512);
        const pub = L.ed25519.getPublicKey(key32);
        const ss58Prefix = net.id === 'polkadot' ? 0 : 2;
        const address = L.base58check(L.sha256).encode(new Uint8Array([ss58Prefix, ...pub]));
        results.push({id:net.id, name:net.name, group:net.group, address,
          path: net.path.replace('{i}', index),
          publicKey: bytesToHex(pub),
          privateKey: showPriv ? bytesToHex(key32) : null});
      }
      else if (net.type === 'cardano') {
        const r = await cardanoAddress(L, mn, passphrase, index);
        results.push({id:net.id, name:net.name, group:net.group, address: r.address,
          path: net.path.replace('{i}', index),
          publicKey: r.publicKey,
          privateKey: showPriv ? r.privateKey : null, note: net.note});
      }
      else if (net.type === 'starknet') {
        const r = await starknetAddress(L, seed, index);
        results.push({id:net.id, name:net.name, group:net.group, address: r.address,
          path: net.path.replace('{i}', index),
          publicKey: r.publicKey,
          privateKey: showPriv ? r.privateKey : null, note: net.note});
      }
      else if (net.type === 'cosmos') {
        const path = net.path.replace('{i}', index);
        const child = root.derive(path);
        const h = L.ripemd160(L.sha256(child.publicKey));
        const address = L.bech32.encode(net.hrp, L.bech32.toWords(h));
        results.push({id:net.id, name:net.name, group:net.group, address, path,
          publicKey: bytesToHex(child.publicKey),
          privateKey: showPriv ? bytesToHex(child.privateKey) : null});
      }
      else if (net.type === 'injective') {
        const path = net.path.replace('{i}', index);
        const child = root.derive(path);
        const address = injectiveCosmosAddress(L, child.publicKey);
        results.push({id:net.id, name:net.name, group:net.group, address, path,
          publicKey: bytesToHex(child.publicKey),
          privateKey: showPriv ? bytesToHex(child.privateKey) : null, note: net.note});
      }
      else if (net.type === 'xrp') {
        const path = net.path.replace('{i}', index);
        const child = root.derive(path);
        const h = L.ripemd160(L.sha256(child.publicKey));
        const address = rippleBase58Encode(new Uint8Array([0x00, ...h]), L.sha256);
        results.push({id:net.id, name:net.name, group:net.group, address, path,
          publicKey: bytesToHex(child.publicKey),
          privateKey: showPriv ? bytesToHex(child.privateKey) : null});
      }
      else if (net.type === 'tron') {
        const path = net.path.replace('{i}', index);
        const child = root.derive(path);
        const pubU = L.secp.getPublicKey(child.privateKey, false).slice(1);
        const h = L.keccak_256(pubU).slice(-20);
        const address = L.base58check(L.sha256).encode(new Uint8Array([0x41, ...h]));
        results.push({id:net.id, name:net.name, group:net.group, address, path,
          publicKey: bytesToHex(child.publicKey),
          privateKey: showPriv ? bytesToHex(child.privateKey) : null});
      }
      else if (net.type === 'waves') {
        const indices = parseEdPath(net.path, index);
        const key32 = slip10Ed25519(seed, indices, L.hmac, L.sha512);
        const pub = L.ed25519.getPublicKey(key32);
        const h = L.keccak_256(L.sha256(pub)).slice(0, 20);
        const address = L.base58check(L.sha256).encode(new Uint8Array([0x57, 0x01, ...h]));
        results.push({id:net.id, name:net.name, group:net.group, address,
          path: net.path.replace('{i}', index),
          publicKey: bytesToHex(pub),
          privateKey: showPriv ? bytesToHex(key32) : null});
      }
    } catch (e) {
      results.push({id:net.id, name:net.name, group:net.group, error: e.message});
    }
  }
  return results;
}

// Memory wipe
// ============================================================
function overwriteWithRandom(str) {
  if (!str) return;
  const len = str.length;
  const rnd = new Uint8Array(len);
  crypto.getRandomValues(rnd);
  // printable chars from rand
  let random = '';
  for (let i = 0; i < len; i++) random += String.fromCharCode((rnd[i] % 94) + 33);
  return random;
}

function wipeSensitiveData(reason = 'manual') {
  try {
    // Overwrite mnemonic textarea
    if (mnemonicEl.value) {
      mnemonicEl.value = overwriteWithRandom(mnemonicEl.value);
      mnemonicEl.value = '';
    }
    // Overwrite passphrase
    if (passphraseEl.value) {
      passphraseEl.value = overwriteWithRandom(passphraseEl.value);
      passphraseEl.value = '';
    }
    // Wipe cached results
    for (const r of lastResults) {
      if (r.privateKey) r.privateKey = overwriteWithRandom(r.privateKey) && null;
      if (r.viewKey) r.viewKey = null;
      if (r.address) r.address = null;
      if (r.publicKey) r.publicKey = null;
    }
    lastResults = [];
    // Wipe results DOM
    resultsEl.innerHTML = '';
    resultsCard.style.display = 'none';
    // Wipe libraries
    if (libs) {
      for (const k of Object.keys(libs)) {
        try { libs[k] = null; } catch(_) {}
      }
      libs = null;
    }
    setStatus(
      reason === 'idle'
        ? '🔒 Автоочистка (простой): мнемоника и ключи стёрты из памяти.'
        : '🧹 Память очищена: мнемоника, passphrase и ключи удалены.',
      'warn'
    );
  } catch (e) {
    setStatus('Ошибка очистки: ' + e.message, 'error');
  }
}

document.getElementById('wipeBtn').onclick = () => wipeSensitiveData('manual');

// Idle auto-wipe
// ============================================================
let idleTimer = null;
let idleCountdownTimer = null;
let idleDeadline = 0;
let idleTimeoutMs = 5 * 60 * 1000;

function clearIdleTimers() {
  if (idleTimer) clearTimeout(idleTimer);
  if (idleCountdownTimer) clearInterval(idleCountdownTimer);
  idleTimer = null;
  idleCountdownTimer = null;
}

function startIdleTimer() {
  clearIdleTimers();
  const sel = parseInt($('idleTimeout').value, 10);
  idleTimeoutMs = sel;
  const cntEl = $('idleCountdown');
  if (!sel || sel <= 0) { cntEl.textContent = ''; return; }
  idleDeadline = Date.now() + sel;
  idleTimer = setTimeout(() => wipeSensitiveData('idle'), sel);
  idleCountdownTimer = setInterval(() => {
    const remain = Math.max(0, idleDeadline - Date.now());
    const sec = Math.floor(remain / 1000);
    const m = Math.floor(sec / 60), s = sec % 60;
    cntEl.textContent = `(осталось ${m}:${String(s).padStart(2, '0')})`;
    if (remain <= 0) clearIdleTimers();
  }, 1000);
}

function resetIdle() { if (idleTimeoutMs > 0) startIdleTimer(); }
document.addEventListener('click', resetIdle, {passive:true});
document.addEventListener('keydown', resetIdle, {passive:true});
document.addEventListener('input', resetIdle, {passive:true});
$('idleTimeout').addEventListener('change', () => {
  resetIdle();
  const v = parseInt($('idleTimeout').value, 10);
  setStatus(v ? `Автоочистка включена: ${v/60000} мин` : 'Автоочистка отключена', 'info');
  setTimeout(() => setStatus(''), 1500);
});

function bytesToBase64(bytes) {
  let binary = '';
  const chunk = 8192;
  for (let i = 0; i < bytes.length; i += chunk) {
    binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunk));
  }
  return btoa(binary);
}

async function collectSources() {
  const embedded = document.getElementById('walletSources');
  if (embedded) return JSON.parse(embedded.textContent);
  const sources = {};
  for (let i = 0; i < LIB_URLS.length; i++) {
    const url = LIB_URLS[i];
    const bar = $('dlProgressBar');
    bar.style.width = `${Math.round((i / LIB_URLS.length) * 100)}%`;
    setStatus(`📦 Скачивание ${i+1}/${LIB_URLS.length}: ${url.split('/').slice(-1)[0]}…`, 'info');
    const res = await fetch(url + '?bundle', {mode: 'cors'});
    if (!res.ok) throw new Error(`HTTP ${res.status} для ${url}`);
    const text = await res.text();
    const bytes = new TextEncoder().encode(text);
    sources[url] = bytesToBase64(bytes);
  }
  $('dlProgressBar').style.width = '100%';
  return sources;
}

const BOOTSTRAP_SCRIPT = `<script>
(function(){
  try {
    var el = document.getElementById('walletSources');
    if (!el) return;
    var sources = JSON.parse(el.textContent);
    var imports = {};
    for (var url in sources) {
      if (!Object.prototype.hasOwnProperty.call(sources, url)) continue;
      var b64 = sources[url];
      var bin = atob(b64);
      var bytes = new Uint8Array(bin.length);
      for (var i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
      var blob = new Blob([bytes], {type: 'application/javascript'});
      imports[url] = URL.createObjectURL(blob);
    }
    var im = document.createElement('script');
    im.type = 'importmap';
    im.textContent = JSON.stringify({imports: imports});
    document.head.appendChild(im);
  } catch(e) { console.error('Bundle bootstrap failed:', e); }
})();
<\/script>`;

function getCleanHTML() {
  // Build clean HTML string from current document
  // 1. Clone the document
  const clone = document.documentElement.cloneNode(true);
  // 2. Clear sensitive fields in the clone
  const clonedMnemonic = clone.querySelector('#mnemonic');
  if (clonedMnemonic) clonedMnemonic.textContent = '';
  const clonedPassphrase = clone.querySelector('#passphrase');
  if (clonedPassphrase) clonedPassphrase.value = '';
  const clonedResults = clone.querySelector('#results');
  if (clonedResults) clonedResults.innerHTML = '';
  const clonedResultsCard = clone.querySelector('#resultsCard');
  if (clonedResultsCard) clonedResultsCard.style.display = 'none';
  const clonedStatus = clone.querySelector('#status');
  if (clonedStatus) clonedStatus.className = 'status';
  // Remove any embedded walletSources
  const existingSources = clone.querySelector('#walletSources');
  if (existingSources) existingSources.remove();
  return '<!DOCTYPE html>\n' + clone.outerHTML;
}

async function downloadStandalone() {
  const btn = $('downloadStandalone');
  btn.disabled = true;
  $('dlProgress').classList.add('active');
  setStatus('<span class="spinner"></span> Подготовка автономной сборки…', 'info');
  try {
    const sources = await collectSources();
    setStatus('📦 Сборка HTML…', 'info');
    let html = getCleanHTML();
    const sourcesJSON = JSON.stringify(sources);
    const sourcesScript = '<script id="walletSources" type="application/json">' + sourcesJSON + '<\/script>';
    // Insert before the module script
    const marker = '<script type="module">';
    const idx = html.indexOf(marker);
    if (idx === -1) throw new Error('Main module script not found');
    html = html.slice(0, idx) + sourcesScript + BOOTSTRAP_SCRIPT + html.slice(idx);
    const blob = new Blob([html], {type: 'text/html;charset=utf-8'});
    const sizeMB = (blob.size / 1024 / 1024).toFixed(2);
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'wallet-standalone.html';
    document.body.appendChild(a);
    a.click();
    a.remove();
    URL.revokeObjectURL(url);
    setStatus(`✓ Автономный файл готов (${sizeMB} МБ). Работает без интернета.`, 'success');
  } catch (e) {
    setStatus('Ошибка сборки: ' + e.message, 'error');
    console.error(e);
  } finally {
    btn.disabled = false;
    $('dlProgress').classList.remove('active');
    $('dlProgressBar').style.width = '0%';
  }
}

$('downloadStandalone').onclick = downloadStandalone;
