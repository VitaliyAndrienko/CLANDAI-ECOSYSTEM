function formatNum(n){
  if (!isFinite(n)) return '0';
  if (n >= 1e9) return (n/1e9).toFixed(2) + 'B';
  if (n >= 1e6) return (n/1e6).toFixed(2) + 'M';
  if (n >= 1e3) return n.toLocaleString('en-US', { maximumFractionDigits: 2 });
  if (n >= 1) return n.toLocaleString('en-US', { maximumFractionDigits: 4 });
  return n.toFixed(6).replace(/\.?0+$/, '');
}
function formatMoney(n){
  const s = Math.abs(n).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  return (n < 0 ? '-$' : '$') + s;
}

function setupSlider(trackEl, fillEl, thumbEl, onChange){
  let pct = 0, dragging = false, startX = 0, startPct = 0, suppress = true;
  function apply(){
    const w = trackEl.clientWidth;
    const thumbW = thumbEl.offsetWidth;
    const pad = 5;
    const maxLeft = Math.max(1, w - thumbW - pad * 2);
    const left = pad + (pct / 100) * maxLeft;
    thumbEl.style.left = left + 'px';
    fillEl.style.width = (left + thumbW) + 'px';
    if (!suppress) onChange(pct);
  }
  function onDown(e){
    if (e.button !== undefined && e.button !== 0) return;
    dragging = true;
    thumbEl.classList.add('dragging');
    startX = e.clientX; startPct = pct;
    e.preventDefault();
    window.addEventListener('pointermove', onMove);
    window.addEventListener('pointerup', onUp);
    window.addEventListener('pointercancel', onUp);
    vibrate(8);
  }
  function onMove(e){
    if (!dragging) return;
    const w = trackEl.clientWidth;
    const thumbW = thumbEl.offsetWidth;
    const pad = 5;
    const maxLeft = Math.max(1, w - thumbW - pad * 2);
    const dx = e.clientX - startX;
    const newLeft = Math.max(0, Math.min(maxLeft, (startPct / 100) * maxLeft + dx));
    pct = (newLeft / maxLeft) * 100;
    apply();
  }
  function onUp(){
    if (!dragging) return;
    dragging = false;
    thumbEl.classList.remove('dragging');
    window.removeEventListener('pointermove', onMove);
    window.removeEventListener('pointerup', onUp);
    window.removeEventListener('pointercancel', onUp);
    if (pct >= 97) pct = 100;
    else if (pct <= 3) pct = 0;
    apply();
  }
  trackEl.addEventListener('pointerdown', e => {
    const thumbRect = thumbEl.getBoundingClientRect();
    if (e.clientX >= thumbRect.left && e.clientX <= thumbRect.right){
      onDown(e);
    } else {
      const rect = trackEl.getBoundingClientRect();
      const w = trackEl.clientWidth;
      const thumbW = thumbEl.offsetWidth;
      const pad = 5;
      const maxLeft = Math.max(1, w - thumbW - pad * 2);
      const relX = e.clientX - rect.left - pad - thumbW / 2;
      pct = Math.max(0, Math.min(100, (relX / maxLeft) * 100));
      apply();
      onDown(e);
    }
  });
  window.addEventListener('resize', () => { if (!suppress) apply(); });
  return {
    getPct(){ return pct; },
    set(p){ pct = p; apply(); },
    apply,
    enable(){ suppress = false; }
  };
}

/* FILTERING */
function getFilteredTokens(){
  const q = state.search.toLowerCase().trim();
  let amountMax = Infinity;
  let qtyMax = Infinity;
  if (state.appliedAmountPct !== null && state.appliedAmountPct < 100){
    amountMax = (state.appliedAmountPct / 100) * state.globalMaxAmount;
  } else if (state.appliedAmountPct === 0){
    amountMax = 0;
  }
  if (state.appliedQtyPct !== null && state.appliedQtyPct < 100){
    qtyMax = (state.appliedQtyPct / 100) * state.globalMaxQuantity;
  } else if (state.appliedQtyPct === 0){
    qtyMax = 0;
  }
  const result = [];
  state.favoriteIds.forEach(id => {
    const t = ALL_TOKENS.find(x => x.id === id);
    if (!t) return;
    if (state.deletedAt[id]){ result.push(t); return; }
    const value = t.balance * t.price;
    if (value > amountMax) return;
    if (t.balance > qtyMax) return;
    if (q){
      const haystack = (t.name + ' ' + t.symbol + ' ' + t.address).toLowerCase();
      if (!haystack.includes(q)) return;
    }
    result.push(t);
  });
  return result;
}

function updateSliderMaxes(){
  let maxAmt = 0, maxQty = 0;
  state.favoriteIds.forEach(id => {
    const t = ALL_TOKENS.find(x => x.id === id);

    if (v > maxAmt) maxAmt = v;
    if (t.balance > maxQty) maxQty = t.balance;

  state.globalMaxQuantity = Math.max(1, maxQty);
}

function togglePanel(btn, targetWin, otherBtn, otherWin){
  const isActive = targetWin.classList.contains('active');
  otherWin.classList.remove('active');
  otherBtn.classList.remove('active');
  targetWin.classList.toggle('active', !isActive);
  btn.classList.toggle('active', !isActive);
}

function updateArrowStates(){
  document.querySelectorAll('.token-card').forEach(el => {
    const id = el.dataset.id;
    const idx = state.favoriteIds.indexOf(id);
    if (idx < 0) return;
    const upBtn = el.querySelector('[data-act="up"]');
    const downBtn = el.querySelector('[data-act="down"]');
    if (upBtn) upBtn.disabled = (idx === 0);
    if (downBtn) downBtn.disabled = (idx === state.favoriteIds.length - 1);
  });
}

/* MOVE TOKEN with FLIP animation */
function moveToken(id, dir){
  if (isAnimatingMove) return;
  const idx = state.favoriteIds.indexOf(id);
  if (idx < 0) return;
  const newIdx = idx + dir;
  if (newIdx < 0 || newIdx >= state.favoriteIds.length) return;

  const list = $('#token-list');
  const movingEl = list.querySelector(`.token-card[data-id="${id}"]`);
  const targetId = state.favoriteIds[newIdx];
  const targetEl = list.querySelector(`.token-card[data-id="${targetId}"]`);

  [state.favoriteIds[idx], state.favoriteIds[newIdx]] = [state.favoriteIds[newIdx], state.favoriteIds[idx]];
  saveFavorites();

  if (!movingEl || !targetEl){
    render();
    vibrate(6);
    return;
  }

  isAnimatingMove = true;

  const beforeMoving = movingEl.getBoundingClientRect();
  const beforeTarget = targetEl.getBoundingClientRect();

  if (dir < 0) list.insertBefore(movingEl, targetEl);
  else list.insertBefore(movingEl, targetEl.nextSibling);

  const afterMoving = movingEl.getBoundingClientRect();
  const afterTarget = targetEl.getBoundingClientRect();

  const dmX = beforeMoving.left - afterMoving.left;
  const dmY = beforeMoving.top - afterMoving.top;
  const dtX = beforeTarget.left - afterTarget.left;
  const dtY = beforeTarget.top - afterTarget.top;

  movingEl.classList.add('moving');
  targetEl.classList.add('shifting');

  movingEl.style.transition = 'none';
  movingEl.style.transform = `translate(${dmX}px, ${dmY}px)`;
  targetEl.style.transition = 'none';
  targetEl.style.transform = `translate(${dtX}px, ${dtY}px)`;

  void movingEl.offsetWidth;
  void targetEl.offsetWidth;

  const duration = 420;
  const easing = 'cubic-bezier(.34,1.15,.64,1)';

  movingEl.style.transition = `transform ${duration}ms ${easing}`;
  movingEl.style.transform = '';
  targetEl.style.transition = `transform ${duration}ms ${easing}`;
  targetEl.style.transform = '';

  vibrate(6);

  setTimeout(() => {
    movingEl.style.transition = '';
    movingEl.style.transform = '';
    targetEl.style.transition = '';
    targetEl.style.transform = '';
    movingEl.classList.remove('moving');
    targetEl.classList.remove('shifting');
    updateArrowStates();
    isAnimatingMove = false;
  }, duration + 30);
}

function startDelete(id){
  if (state.deletedAt[id]) return;
  state.deletedAt[id] = Date.now();
  saveDeletes();
  render();
  vibrate([12, 30, 12]);
  toast('Удаление через 60 секунд');
}

function cancelDelete(id){
  if (!state.deletedAt[id]) return;
  if (state.deleteTimers.has(id)){
    clearInterval(state.deleteTimers.get(id));
    state.deleteTimers.delete(id);
  }
  delete state.deletedAt[id];
  saveDeletes();
  render();
  toast('✓ Удаление отменено', 'ok');
}

function startCountdown(id){
  if (state.deleteTimers.has(id)){
    clearInterval(state.deleteTimers.get(id));
    state.deleteTimers.delete(id);
  }
  const item = document.querySelector(`.token-card[data-id="${id}"]`);
  if (!item) return;
  const banner = item.querySelector('.delete-banner');
  if (!banner) return;
  const fillEl = banner.querySelector('.fill');
  const secondsEl = banner.querySelector('.cd-seconds');
  const started = state.deletedAt[id];
  const tick = () => {
    const elapsed = Date.now() - started;
    const remaining = Math.max(0, DELETE_TIMEOUT - elapsed);
    const pct = remaining / DELETE_TIMEOUT;
    const secs = Math.ceil(remaining / 1000);
    if (fillEl) fillEl.style.width = (pct * 100) + '%';
    if (secondsEl) secondsEl.textContent = secs;
    if (remaining <= 0){
      const iid = state.deleteTimers.get(id);
      if (iid){ clearInterval(iid); state.deleteTimers.delete(id); }
      finalizeDelete(id);
    }
  };
  tick();
  const iid = setInterval(tick, 100);
  state.deleteTimers.set(id, iid);
}

function finalizeDelete(id){
  delete state.deletedAt[id];
  saveDeletes();
  const idx = state.favoriteIds.indexOf(id);
  if (idx >= 0){
    state.favoriteIds.splice(idx, 1);
    saveFavorites();
  }
  const item = document.querySelector(`.token-card[data-id="${id}"]`);
  if (item){
    item.classList.add('removing');
    setTimeout(() => render(), 450);
  } else {
    render();
  }
  toast('Токен удалён из избранного');
}
